    \section{Fundamentals}

    \begin{definition}
        Let $\Omega$ be a set and $\mathcal F$ is a collection of subsets of $\Omega$. We call $\mathcal F$ a \emph{$\sigma$-algebra} if
        \begin{itemize}
            \item $\Omega \in \mathcal F$;
            \item if $A \in \mathcal F$, then $A^\comp \in \mathcal F$;
            \item if $(A_n)_{n \in \NN}$ is a countable collection of sets in $\mathcal F$, then
            \begin{align*}
                \bigcup A_n \in \mathcal F
            \end{align*}
        \end{itemize}
    \end{definition}
    
    \begin{definition}
        Let $\mathcal F$ be a $\sigma$-algebra on $\Omega$. A function
        \begin{align*}
            \PP: \mathcal F \to [0, 1]
        \end{align*}
        is called a \emph{probability measure} if
        \begin{itemize}
            \item $\PP(\Omega) = 1$;
            \item for any countable disjoint collection $(A_n)_{n \in \NN}$ in $\mathcal F$,
            \begin{align*}
                \PP \left(\bigcup A_n\right) = \sum \PP(A_n)
            \end{align*}
        \end{itemize}
        We call $\PP(A)$ the \emph{probability} of $A$.
    \end{definition}
    
    Then $(\Omega, \mathcal F, \PP)$ is a \emph{probability space}.
    
    \begin{definition}
        The elements of $\Omega$ are called \emph{outcomes}, and the elements of the $\sigma$-algebra $\mathcal F$ are called \emph{events}.
    \end{definition}
    
    Importantly, we talk about probabilities of \emph{events}, never probabilities of outcomes -- the domain of $\PP$ is $\mathcal F$, not $\Omega$.

    \begin{remark*}
        When $\Omega$ is countable, we take $\mathcal F$ to be all subsets of $\Omega$. However, this framework will become useful when considering uncountable $\Omega$.
    \end{remark*}
    
    Let us note some important properties of probability measures that follow from the definition.

    \begin{itemize}
        \item $\PP(A^\comp) = 1 - \PP(A)$;
        \item $\PP(\emptyset) = 0$;
        \item $\PP(A) \leq \PP(B)$ if $A \subseteq B$;
        \item $\PP(A \cup B) = \PP(A) + \PP(B) - \PP(A \cap B)$.
    \end{itemize}
    
    \begin{example}
        \begin{itemize}
            \item \emph{Rolling a fair die}. Then $\Omega = \{1, 2, 3, 4, 5, 6\}$, $\mathcal F = \mathcal P(\Omega)$, and
            \begin{align*}
                \PP(A) = \frac{\mid A\mid }{6}
            \end{align*}
            since all outcomes are equally likely.
    
            \item $\Omega = \{\omega_1, \dots, \omega_n\}$, $\mathcal F = \mathcal P(\Omega)$, $\PP(A) = \frac{\mid A\mid }{\mid \Omega\mid }$ models the experiment of picking an element of $\Omega$ uniformly at random.
            
            \item \emph{Picking balls from a bag}. Suppose we have $n$ balls labelled $\{1, \dots, n\}$, and we pick $k \leq n$ balls uniformly at random without replacement. Then $\Omega = \{A:  A \subseteq \{1, \dots, n\}, \mid A\mid  = k\}$, so $\mid \Omega\mid  = \binom n k$ and $\PP(\{\omega\}) = \binom n k ^{-1}$ for any $\omega \in \Omega$.
            
            \item \emph{A deck of 52 cards}. Take a well-shuffled deck of 52 cards. Then $\Omega = \{\text{all permutations of 52 cards}\}$, so $\mid \Omega\mid  = 52!$.
            
            What is, for example, $\PP(\text{top 2 cards are aces})$? It is $\frac{4 \cdot 3 \cdot 50!}{52!}$.

            \item \emph{Largest digit}. Consider a string of $n$ random digits from 0 up to 9. Then $\Omega = \{0, \dots, 9\}^n$, $\mid \Omega\mid  = 10^n$.
            
            Consider the events $A_k = \{\text{no digit exceeds $k$}\}$, $B_k = \{\text{largest digit is $k$}\}$. Then
            \begin{align*}
                \PP(A_k) &= \frac{(k+1)^n}{10^n} \\
                \implies \PP(B_k) = \PP(A_k \setminus A_{k-1}) &= \frac{(k+1)^n-k^n}{10^n}
            \end{align*}

            \item \emph{Birthday problem}. There are $n$ people. What is the probability that at least 2 of them share the same birthday?
            
            Assume each birthday is equally likely to be one of $\{1, \dots, 365\}$. Then $\Omega = \{1, \dots, 365\}^n$, $\mid \Omega\mid  = 365^n$, and $\PP(\{\omega\}) = \frac{1}{365^n}$.

            Let $A = \{\text{at least 2 people have the same birthday}\}$. Instead we count $A^\comp = \{\text{all birthdays different}\}$.

            \begin{align*}
                \PP(A^\comp) &= \frac{365 \cdot 364 \cdots (365-n+1)}{365^n} \\
                \implies \PP(A) &= 1 - \frac{365 \cdot 364 \cdots (365-n+1)}{365^n}
            \end{align*}
            
            Perhaps surprisingly, if $n=22$, $\PP(A) = 0.476$, while if $n=23$, $\PP(A) = 0.507$.
        \end{itemize}
    \end{example}
    
    \subsection{Combinatorial analysis}

    Let us introduce some notions.

    \begin{enumerate}
        \item Let $\Omega$ be a finite set with $\mid \Omega\mid  = n$. We want to partition $\Omega$ into $k$ disjoint subsets $\Omega_1, \dots, \Omega_k$ with $\mid \Omega_i\mid  = n_i$ and $\sum n_i = n$. How many ways are there to do this? It is

        \begin{align*}
            M &= \binom n {n_1} \cdot \binom {n-n_1} {n_2} \dots \binom{n-(n_1 + \dots + n_{k-1})}{n_k} \\
            &= \frac{n!}{n_1!n_2! \cdots n_k!} \\
            &= \binom{n}{n_1, n_2, \dots, n_k}
        \end{align*}
        
        This is the \emph{multinomial coefficient}.

        \item Suppose $f: \{1, \dots, k\} \to \{1, \dots, n\}$ (with $k \leq n$). $f$ is \emph{strictly increasing} if $x < y \implies f(x) < f(y)$, and $f$ is \emph{increasing} if $x < y \implies f(x) \leq f(y)$.
        
        How many strictly increasing $f: \{1, \dots, k\} \to \{1, \dots, n\}$ are there? Such a function is determined uniquely by its range, so there are $\binom n k$ of them.

        What about increasing functions? Let's define a bijection
        \begin{align*}
            \{f: \{1, \dots, k\} \to \{1, \dots, n\}\text{ inc.}\} &\to \{f: \{1, \dots, k\} \to \{1, n+k-1\} \text{ strictly inc.}\}
        \end{align*}
        which takes $f$ to $g$ with $g(i) = f(i) + i - 1$. Hence we have $\binom{n+k-1}{k}$ such functions.
    \end{enumerate}

    \begin{definition}[Asymptotic notation]
        If $(a_n)$ and $(b_n)$ are sequences, we write $a_n \sim b_n$ if
        \begin{align*}
            \lim_{n \to \infty} \frac{a_n}{b_n} = 1
        \end{align*}
    \end{definition}
    
    \subsection{Stirling's formula}
    

    \begin{theorem}[Stirling]
        \[
            n! \sim n^n \sqrt{2\pi n} e^{-n}
        \]
        as $n \to \infty$.        
    \end{theorem}
    
    \begin{proof}[Weaker statement]
        We shall first show a weaker statement, that $\log n! \sim n \log n$.

        Define $l_n = \log n! = \log 2 + \dots + \log n$. We have
        \begin{align*}
            \log \lfloor x \rfloor \leq \log x \leq \log \lfloor x+1 \rfloor
        \end{align*}
        Integrating from 1 to $n$ gives
        \begin{align*}
            \sum_{k=1}^{n-1} \log k &\leq \int_1^n \log x \; dx \leq \sum_{k=1}^{n} \log k \\
            \implies l_{n-1} &\leq n \log n - n + 1 \leq l_n \\
            \implies n \log n - n + 1 &\leq l_n \leq (n+1) \log (n+1) - (n+1) + 1
        \end{align*}
        Now the result follows by dividing through by $n \log n$ and taking the limit.
    \end{proof}
    
    \begin{proof}[Non-examinable proof of Stirling]
        For every twice-differentiable $f: \RR \to \RR$, for all $a < b$, we have
        \begin{align*}
            \int_a^b f(x) \; dx = \frac{f(a) + f(b)}{2}(b-a) - \frac{1}{2} \int_a^b (x-a)(b-x)f''(x) \; dx
        \end{align*}
        obtained by integrating twice by parts. Put $f = \log$, $a = k$, $b = k+1$ for $k \in \NN$, to get
        \begin{align*}
            \int_k^{k+1} \log x \; dx = \frac{\log k + \log(k+1)}{2} + \frac{1}{2} \int_{k}^{k+1} \frac{(x-k)(k+1-x)}{x^2} \; dx
        \end{align*}
        Summing from $k=1$ to $n-1$, we have
        \begin{align*}
            \int_1^n \log x \; dx = \frac{\log(n-1)! + \log n!}{2} + \sum_{k=1}^{n-1} a_k
        \end{align*}
        where $a_k = \frac 1 2 \int_0^1 \frac{x(1-x)}{(x+k)^2} \; dx$. Hence
        \begin{align*}
            n \log n - n + 1 = \log n! - \frac{\log n}{2} + \sum_{k=1}^{n-1} a_k
        \end{align*}
        and so
        \begin{align*}
            \log n! &= n \log n - n + 1 + \frac{\log n}{2} - \sum_{k=1}^{n-1} a_k \\
            \implies n! &= n^n \cdot e^{-n} \cdot \sqrt{n} \cdot \exp\left(1-\sum_{k=1}^{n-1} a_k\right)
        \end{align*}
        Let us now consider
        \begin{align*}
            a_k &= \frac{1}{2} \int_0^1 \frac{x(1-x)}{(x+k)^2} \; dx \\
            &\leq \frac{1}{2} \int_0^1 \frac{1}{2k^2} \int_0^1 x(1-x) \; dx \\
            &= \frac{1}{12k^2}
        \end{align*}
        and so $\sum_{k=1}^\infty a_k < \infty$. Now put
        \begin{align*}
            A &= \exp\left(1 - \sum_{k=1}^\infty a_k\right) \\
            \implies n! &= n^n \cdot e^{-n} \cdot \sqrt{n} \cdot A \cdot \underbrace{\exp\left(\sum_{k=n}^\infty a_k\right)}_{\to 1 \text{ as } n \to \infty} \\
            \implies \frac{n!}{n^ne^{-n}\sqrt{n}} &\to A \text{ as } n \to \infty
        \end{align*}
        It remains to show that $A = \sqrt{2\pi}$. Consider
        \begin{align*}
            2^{-2n}\binom{2n}{n} &= 2^{-2n} \frac{(2n)!}{(n!)^2} \\
            &\sim \frac{2^{-2n} (2n)^{2n} e^{-2n} \sqrt{2n} A}{n^{2n}e^{-2n}nA^2} \\
            &= \frac{\sqrt 2}{A \sqrt n}
        \end{align*}
        Now, we shall use a different method to show
        \begin{align*}
            2^{-2n} \binom{2n}{n} \sim \frac{1}{\sqrt{\pi n}} \text{ as }n \to \infty
        \end{align*}
        Consider
        \begin{align*}
            I_n &= \int_0^{\pi/2} \cos^n \theta \; d\theta
        \end{align*}
        We have $I_0 = \frac{\pi}{2}$, $I_1 = 1$, and, by integrating by parts, we have
        \begin{align*}
            I_n = \frac{n-1}{n} I_{n-2} && (*)
        \end{align*}
        so that
        \begin{align*}
            I_{2n} &= \frac{2n-1}{2n} I_{2n-2} \\
            &= \dots = \frac{(2n-1)(2n-3)\cdots 3 \cdot 1}{(2n)(2n-2) \cdots 2} \cdot \frac{\pi}{2} \\
            &= \frac{2n(2n-1)(2n-2) \cdots 3 \cdot 2 \cdot 1}{2n \cdot 2n \cdot (2n-2) \cdot (2n-2) \cdots 2 \cdot 2} \cdot \frac{\pi}{2} \\
            &= \frac{(2n)!}{2^{2n}(n!)^2} \cdot \frac{\pi}{2} \\
            &= \frac{\pi}{2} \cdot 2^{-2n} \binom{2n}{n}
        \end{align*}
        and we can similarly obtain
        \begin{align*}
            I_{2n+1} &= \frac{2n \cdots 4 \cdot 2}{(2n+1) \cdots 3 \cdot 1} \cdot 1 \\
            &= \frac{1}{2n+1} \left(2^{-2n} \binom{2n}{n}\right)^{-1}
        \end{align*}
        Now, we want to show that $\frac{I_{2n}}{I_{2n+1}} \to 1$ as $n \to \infty$. By $(*)$, we have
        \begin{align*}
            \frac{I_n}{I_{n-2}} \to 1 \text{ as } n \to \infty
        \end{align*}
        But $I_n = \int_0^{\pi/2} \cos^n \theta \; d\theta$, so $I_n$ is decreasing in $n$. This gives
        \begin{align*}
            \frac{I_{2n}}{I_{2n-2}} \leq \frac{I_{2n}}{I_{2n+1}} \leq \frac{I_{2n-1}}{I_{2n+1}}
        \end{align*}
        which shows the result by sandwich. Hence, we get, as $n \to \infty$,
        \begin{align*}
            \frac{\frac{\pi}2 \cdot 2^{-2n} \cdot \binom{2n}{n}}{\frac{1}{2n+1} \cdot \left(2^{-2n} \binom{2n}{n}\right)^{-1}} &\to 1 \\ \implies \left(2^{-2n} \binom{2n}{n}\right)^2 &\sim \frac{1}{\pi n}
        \end{align*}        
    \end{proof}
    
    \begin{proposition}[Countable subadditivity]
        Let $(A_n)$ be a collection of events in $\mathcal F$. Then
        \begin{align*}
            \PP\left(\bigcup A_n\right) \leq \sum \PP(A_n)
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Let $B_1 = A_1$, and for $n \geq 2$, let $B_n = A_n \setminus (A_1 \cup \cdots \cup A_{n-1})$. By definition, it's immediate that $(B_n)$ is a disjoint collection of sets in $\mathcal F$, and $\bigcup B_n = \bigcup A_n$, so
        \begin{align*}
            \PP\left(\bigcup A_n\right) = \PP\left(\bigcup B_n\right) &= \sum \PP(B_n) \\
            &\leq \sum \PP(A_n)
        \end{align*}
        since $B_n \subseteq A_n$ for all $n$.
    \end{proof}
    
    \begin{proposition}[Continuity of probability measures]
        \label{continuity}
        Take a sequence of events $(A_n)_{n \in \NN}$ such that $A_n \subseteq A_{n+1}$ for all $n$ -- we say $(A_n)$ is an \emph{increasing} sequence.

        Then $\PP(A_n)$ is increasing and converges. In particular,
        \begin{align*}
            \PP(A_n) \to \PP\left(\bigcup A_n\right)
        \end{align*}
        as $n \to \infty$.
    \end{proposition}
    
    \begin{proof}
        Define $B_1 = A_1$, and for $n \geq 2$, let $B_n = A_n \setminus (A_1 \cup \cdots \cup A_n)$. Then
        \begin{align*}
            \bigcup_{k=1}^n B_k = A_n
        \end{align*}
        and the $B_n$'s are disjoint.Then
        \begin{align*}
            \PP(A_n) &= \PP\left(\bigcup_{k=1}^n B_k\right) \\
            &= \sum_{k=1}^n \PP(B_k) \to \sum_{k=1}^\infty \PP(B_k)
        \end{align*}
        Also,
        \begin{align*}
            \sum_{k=1}^\infty \PP(B_k) &= \PP\left(\bigcup B_k\right) \\
            &= \PP\left(A_k\right)
        \end{align*}
    \end{proof}
    
    \begin{corollary}
        Analogously, take a sequence of events $(A_n)$ such that $A_{n+1} \subseteq A_n$ for all $n$ -- we say $(A_n)$ is a decreasing sequence. Then
        \begin{align*}
            \PP(A_n) \to \PP\left(\bigcap A_n\right)
        \end{align*}
    \end{corollary}
    
    \begin{proof}
        Take complements and use the previous claim.
    \end{proof}
    
    \begin{proposition}[Inclusion-exclusion formula for probability measures]
        Let $A_1, \dots, A_n \in \mathcal F$. Then
        \begin{align*}
            \PP\left(\bigcup A_i\right) = \sum_{k=1}^n (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n} \PP(A_{i_1} \cap \cdots \cap A_{i_k})
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        We proceed by induction.

        \emph{Base case}. ADD

        \emph{Inductive step}. We have
        \begin{align*}
            \PP\left(\bigcup_{i=1}^n A_i\right) &= \PP\left(\left(\bigcup_{i=1}^{n-1} A_i\right) \cup A_n\right) \\
            &= \PP\left(\bigcup_{i=1}^{n-1} A_i\right) + \PP(A_n) - \PP\left(\bigcup_{i=1}^{n-1} A_i \cap A_n\right) \\
            &= \PP\left(\bigcup_{i=1}^{n-1} A_i\right) + \PP(A_n) - \PP\left(\left(\bigcup_{i=1}^{n-1} A_i\right) \cap A_n\right) \\
            &= \PP\left(\bigcup_{i=1}^{n-1} A_i\right) + \PP(A_n) - \PP\left(\bigcup_{i=1}^{n-1} \underbrace{\left(A_i \cap A_n\right)}_{B_i}\right)
        \end{align*}
        Applying the inductive hypothesis, we get
        \begin{align*}
            \PP\left(\bigcup_{i=1}^{n-1} A_i\right) &= \sum_{k=1}^{n-1} (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n-1} \PP(A_{i_1} \cap \dots \cap A_{i_k}) \\
            \PP\left(\bigcup_{i=1}^{n-1} B_i\right) &= \sum_{k=1}^{n-1} (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n-1} \PP(A_{i_1} \cap \dots \cap A_{i_k} \cap A_n)
        \end{align*}
        and subbing this in, we will find that everything works out.
    \end{proof}
    
    \begin{proposition}[Bonferroni inequalities]
        Let $A_1, \dots, A_n$ be events. Then
        \begin{align*}
            \PP\left(\bigcup A_i\right) \begin{cases}
                \displaystyle \leq \sum_{k=1}^r (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n} \PP(A_{i_1} \cap \cdots \cap A_{i_k}) & \text{if } r \text{ odd} \\
                \displaystyle \geq \sum_{k=1}^r (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n} \PP(A_{i_1} \cap \cdots \cap A_{i_k}) & \text{if } r \text{ even} 
            \end{cases}
        \end{align*}
        where those sums are just truncated expansions of inclusion-exclusion.
    \end{proposition}
    
    \begin{proof}
        We induct again.

        \emph{Base case}. $\PP(A \cup B) = \PP(A) + \PP(B) - \PP(A \cap B) \leq \PP(A) + \PP(B)$

        \emph{Inductive step}. Suppose $r$ is odd. Then
        \begin{align*}
            \PP\left(\bigcup_{i=1}^n A_i\right) &= \PP\left(\bigcup_{i=1}^{n-1} A_i\right) + \PP(A_n) - \PP\left(\bigcup_{i=1}^{n-1} \underbrace{\left(A_i \cap A_n\right)}_{B_i}\right)
        \end{align*}
        By the inductive hypotheses,
        \begin{align*}
            \PP\left(\bigcup_{i=1}^{n-1} A_i\right) &\leq \sum_{k=1}^{r} (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n-1} \PP(A_{i_1} \cap \dots \cap A_{i_k}) \\
            \PP\left(\bigcup_{i=1}^{n-1} (A_i \cap A_n)\right) &\geq \sum_{k=1}^{r-1} (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n-1} \PP(A_{i_1} \cap \dots \cap A_{i_k} \cap A_n)
        \end{align*}
        where in the former, we choose to apply the identity for odd $r$, and in the latter, we apply it for even $r-1$. Substituting much as for the proof of the inclusion-exclusion formula, we obtain the result. The even case is analogous.
    \end{proof}
    
    Let us now consider the special case where $\Omega$ is a finite set and $\PP(A) = \mid A\mid /\mid \Omega\mid $. Then we have

    \begin{corollary}[Inclusion-exclusion principle for sets]
        Suppose $\Omega$ is finite. Then
        \begin{align*}
            \mid A_1 \cup \dots \cup A_n\mid  = \sum_{k=1}^n (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n} \mid A_{i_1} \cap \dots \cap A_{i_n}\mid 
        \end{align*}
    \end{corollary}
    
    \begin{proof}
        Consider the probability space $(\Omega, \mathcal F, \PP)$ where $\PP(A) = \mid A\mid /\mid \Omega\mid $. Then, apply inclusion-exclusion and cancel $1/\mid \Omega\mid $ from both sides.
    \end{proof}

    \subsection{Counting surjections}

    Define a probability space as follows:
    \begin{align*}
        \Omega &= \{f: \{1, \dots, n\} \to \{1, \dots, m\}\} \\
        A &= \{f \in \Omega: f \text{ is a surjection}\}
    \end{align*}
    What is $\mid A\mid $? Let us define a set of events
    \begin{align*}
        A_i = \{f \in \Omega: i \notin \{f(1), \dots, f(n)\}\}, \, i \in \{1, \dots, m\}
    \end{align*}
    so that
    \begin{align*}
        A &= A_1^\comp \cap \dots \cap A_m^\comp = (A_1 \cup \dots \cup A_m)^\comp \\
        \implies \mid A\mid  &= \mid \Omega\mid  - \mid A_1 \cup \dots \cup A_m\mid  \\
        &= \mid \Omega\mid  - \sum_{k=1}^m (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq m} \mid A_{i_1} \cap \dots \cap A_{i_k}\mid  \\
        &= \underbrace{\mid \Omega\mid }_{m^n} - \sum_{k=1}^m (-1)^{k+1} \binom{m}{k} (m-k)^n \\
        &= \sum_{k=0}^m (-1)^k \binom{m}{k} (m-k)^n
    \end{align*}

    \subsection{Counting derangements}

    \begin{definition}
        A \emph{derangement} is a permutation with no fixed points.
    \end{definition}
    Let
    \begin{align*}
        \Omega &= \{\text{permutations of } \{1, \dots, n\}\} \\
        A &= \{\text{derangements}\}

    \begin{align*}
        A_i = \{f \in \Omega: f(i) = i\}
    \end{align*}
    so that
    \begin{align*}
        A = A_1^\comp \cap \dots \cap A_n^\comp = (A_1 \cup \dots \cup A_n)^\comp \\
        \implies \PP(A) = 1 - \PP\left(\bigcup A_i\right) &= 1 - \sum_{k=1}^n (-1)^{k+1} \sum_{1 \leq i_1 < \dots < i_k \leq n} \underbrace{\PP(A_{i_1} \cap \dots \cap A_{i_k})}_{\frac{(n-k)!}{n!}} \\

        &= 1 - \sum_{k=1}^n (-1)^{k+1} \frac{1}{k!} \\
        &= \sum_{k=0}^n \frac{(-1)^k}{k!} \to e^{-1}
    \end{align*}
    
    \section{Independence and conditional probability}
    
    \begin{definition}[Independence]
        Let $A$, $B \in \mathcal F$. We say $A$ and $B$ are \emph{independent} and write $A \perp B$ if
        \begin{align*}
            \PP(A \cap B) = \PP(A) \PP(B)
        \end{align*}
        Likewise, a countable collection of events $(A_n)_{n \in \NN}$ is said to be independent if, for distinct $i_1, \dots, i_k$,
        \begin{align*}
            \PP(A_{i_1} \cap \dots \cap A_{i_k}) = \prod_{j=1}^k \PP(A_{i_j})
        \end{align*}
    \end{definition}
    
    \begin{remark*}
        Pairwise independence of events does not imply independence.
    \end{remark*}
    
    \begin{example*}
        Toss a fair coin twice. We have
        \begin{align*}
            \Omega &= \{(0, 0), (0, 1), (1, 0), (1, 1)\} \\
            \PP(\{\omega\}) &= \frac 1 4
        \end{align*}
        Let
        \begin{align*}
            A &= \{(0, 0), (0, 1)\} \\
            B &= \{(0, 0), (1, 0)\} \\
            C &= \{(1, 0), (0, 1)\}
        \end{align*}
        We have $\PP(A) = \PP(B) = \PP(C)$, and furthermore we can check that $A \perp B$, $A \perp C$ and $B \perp C$. But $\\P(A \cap B \cap C) = \PP(\emptyset) = 0$, so they are not independent.
    \end{example*}
    
    \begin{proposition}
        If $A$ is independent of $B$, then $A$ is independent of $B^\comp$.
    \end{proposition}
    
    \begin{proof}
        We have
        \begin{align*}
            \PP(A \cap B^\comp) &= \PP(A) - \PP(A \cap B) \\
            &= \PP(A)(1 - \PP(B)) \\
            &= \PP(A)\PP(B^\comp)
        \end{align*}
    \end{proof}
    
    \begin{definition}[Conditional probability]
        Take a probability space $(\Omega, \mathcal F, \PP)$ and $B \in \mathcal F$ with $\PP(B) > 0$. For $A \in \mathcal F$ we write the \emph{conditional probability of $A$ given $B$} 
        \begin{align*}
            \PP(A \mid  B) = \frac{\PP(A \cap B)}{\PP(B)}
        \end{align*}
    \end{definition}
    
    \begin{remark*}
        Notably, if $A \perp B$, then $\PP(A \mid  B) = \PP(A)$, as we would expect.
    \end{remark*}
    
    \begin{proposition}[Countable additivity for conditional probability]
        Suppose $(A_n)_{n \in \NN}$ is a disjoint sequence in $\mathcal F$. Then
        \begin{align*}
            \PP\left(\bigcup A_n \mid  B \right) = \sum \PP(A_n \mid  B)
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        We have
        \begin{align*}
            \PP\left(\bigcup A_n \mid  B\right) &= \frac{\PP\left(\bigcup A_n \cap B\right)}{\PP(B)} \\
            &= \frac{\PP\left(\bigcup (A_n \cap B)\right)}{\PP(B)} \\
            &= \sum \frac{\PP(A_n \cap B)}{\PP(B)} \\
            &= \sum \PP(A_n \mid  B)
        \end{align*}
    \end{proof}
    
    \begin{proposition}[Law of total probability]
        Suppose $(B_n)_{n \in \NN}$ is a disjoint collection in $\mathcal F$, $\bigcup B_n = \Omega$, and $\PP(B_n) > 0$ for every $n$. Let $A \in \mathcal F$, then
        \begin{align*}
            \PP(A) = \sum_n \PP(A \mid  B_n) \PP(B_n)
        \end{align*}
    \end{proposition}

    \begin{proof}
        We have
        \begin{align*}
            \PP(A) &= \PP\left(A \cap \left(\bigcup B_n\right)\right) \\
            &= \PP\left(\bigcup (B_n \cap A)\right) \\
            &= \sum \PP(B_n \cap A) \\
            &= \sum \PP(A \mid  B_n) \PP(B_n)
        \end{align*}
    \end{proof}
    
    \begin{proposition}[Bayes' theorem]
        Suppose $(B_n)$ are disjoint, $\bigcup B_n = \Omega$ and $P(B_n) > 0$ for every $n$. Let $A \in \mathcal F$, then
        \begin{align*}
            \PP(B_n \mid  A) = \frac{\PP(A \mid  B_n) \cdot \PP(B_n)}{\sum_k \PP(A \mid  B_k)\PP(B_k)}
        \end{align*}
    \end{proposition}
    
    \begin{proof}[Proof sketch]
        Use the definitions of $\PP(B_n \mid  A) $ and $\PP(A \mid  B_n)$, and expand the bottom using the law of total probability.
    \end{proof}
    
    This formula is the basis of so-called Bayesian statistics. We know the probability of $\PP(B_n)$ and we have a model which gives us $\PP(A \mid  B_n)$. With the formula, we find the \emph{posterior} probability of $B_n$ given that $A$ occurred.
    
    \begin{example*}[False positives for a rare condition]
        Suppose that condition $A$ affects $0.1\%$ of the population. We have a medical test which is positive for $98\%$ of the affected population and $1\%$ of the unaffected population. What is the probability that a random individual who tests positive has the disease?

        Put $A = \{\text{individual suffers from }A\}$, $P = \{\text{individual tests positive}\}$. We have $\PP(A) = 0.001$, $\PP(P \mid  A) = 0.98$, $\PP(P \mid  A^\comp) = 0.01$, so
        \begin{align*}
            \PP(A \mid  P) &= \frac{\PP(P \mid  A) \PP(A)}{\PP(P \mid  A)\PP(A) + \PP(P\mid A^\comp)\PP(A^\comp)} \\
            &= 0.089
        \end{align*}
        In particular, we can see
        \begin{align*}
            \PP(A \mid  P) = \frac{1}{1 + \frac{\PP(P\mid A^\comp)\PP(A^\comp)}{\PP(P \mid  A)\PP(A)}}
        \end{align*}
        Since $\PP(A^\comp)$ and $\PP(P\mid A)$ are both very close to one, the relevant ratio is
        \begin{align*}
            \frac{\PP(P \mid  A^\comp)}{\PP(A)}
        \end{align*}
        But since $\PP(P \mid  A^\comp) \gg \PP(A)$, this causes a very small value for $\PP(A \mid  P)$.
    \end{example*}

    \begin{example*}[Simpson's paradox]
        Consider the following data:
        \begin{center}
            \begin{tabular}{l | c c c}
                \emph{All applicants} & Admitted & Rejected & \% Admitted \\
                \hline
                State & 25 & 25 & 50\% \\
                Indep. & 28 & 22 & 56\%
            \end{tabular}
        \end{center}
        \begin{center}        
            \begin{tabular}{l | c c c}
                \emph{London schools} & Admitted & Rejected & \% Admitted \\
                \hline
                State & 15 & 22 & 41\% \\
                Indep. & 5 & 8 & 38\%
            \end{tabular}
        \end{center}
        \begin{center}    
            \begin{tabular}{l | c c c}
                \emph{Cambridge schools} & Admitted & Rejected & \% Admitted \\
                \hline
                State & 10 & 3 & 77\% \\
                Indep. & 23 & 14 & 62\%
            \end{tabular}
        \end{center}
        -- paradoxically, the state admission rate is higher when the school locations are taken individually, but the independent admission right is higher for the combined data. This phenomenon is called \emph{confounding}, and arises when we aggregate data from disparate populations.

        Let 
        \begin{align*}
            A &= \{\text{individual is admitted}\} \\
            B &= \{\text{individual from London}\} \\ 
            C &= \{\text{individual from state school}\}
        \end{align*}
        Then
        \begin{align*}
            \PP(A \mid B \cap C) &> \PP(A \mid B \cap C^\comp) \\
            \PP(A \mid B^\comp \cap C) &> \PP(A \mid B^\comp \cap C^\comp) \\
            \text{but } \PP(A \mid C) &< \PP(A \mid C^\comp)
        \end{align*}
        Let's examine how this can happen. We have
        \begin{align*}
            \PP(A \mid C) &= \PP(A \cap B \mid C) + \PP(A \cap B^\comp \mid C) \\
            &= \frac{\PP(A \cap B \cap C)}{\PP(C)} + \frac{\PP(A \cap B^\comp \cap C)}{\PP(C)} \\
            &= \frac{\PP(A \mid B \cap C)\PP(B \cap C)}{\PP(C)} + \frac{\PP(A \mid B^\comp \cap C)\PP(B^\comp \cap C)}{\PP(C)} \\
            &= \PP(A \mid B \cap C)\PP(B \mid C) + \PP(A \mid B^\comp \cap C^\comp)\PP(B^\comp \mid C^\comp) \\
            &> P(A \mid B \cap C^\comp)\PP(B \mid C) + \PP(A \mid B^\comp \cap C^\comp)\PP(B^\comp \mid C)
        \end{align*}
        by the first two inequalities from above. Now, if $\PP(B \mid C) = \PP(B \mid C^\comp)$, then we would get $\PP(A \mid C) > \PP(A \mid C^\comp)$, which is the intuitive expectation -- but here this does not hold, and so we can make the `paradox' hold.

        Even more simply, this is just the fact that, if $\frac{a_1}{a_2} > \frac{b_1}{b_2}$ and $\frac{c_1}{c_2} > \frac{d_1}{d_2}$, it does not necessarily hold that $\frac{a_1+c_1}{a_2+c_2} > \frac{b_1+d_1}{b_2+d_2}$.       
    \end{example*}
    
    \section{Probability distributions}

    \subsection{Discrete probability distributions}

    Take a probability space $(\Omega, \mathcal F, \PP)$ with $\Omega$ either finite or countable -- that is, $\Omega = \{\omega_1, \omega_2, \dots\}$. As normal, we take $\mathcal F = \mathcal P(\Omega)$, and so we need only care about $\PP(\{\omega_i\})$ for every $i$.

    \begin{definition}[Discrete probability distributions]
        For such a setup, we call the sequence of probabilities $\left(\PP(\{\omega_i\})\right)_i$ a \emph{discrete probability distribution}. We may also write $p_i = \PP(\{\omega_i\})$.
    \end{definition}

    Let us consider some examples of discrete probability distributions.
    
    \begin{enumerate}[label=\protect\circled{\arabic*}]
        \item \emph{Bernoulli distribution}. The Bernoulli distribution models the outcome of a toss of a biased coin. We have $\Omega = \{0, 1\}$, $p_0 = \PP(\{0\}) = 1 - p$, $p_1 = \PP(\{1\}) = p$ for some $p$ with $p \in [0, 1]$.
        \item \emph{Binomial distribution}. We denote this $\Bin(n, p)$, with $n \in \NN$, $p \in [0, 1]$. It models the number of heads seen for $n$ independent tosses of a $p$-biased coin. We have
        \begin{align*}
            \Omega &= \{0, 1, \dots, n\} \\
            p_k &= \binom n k p^k (1-p)^{n-k} 
        \end{align*}
        \item \emph{Multinomial distribution}. We denote this $\M(n, p_1, \dots, p_k)$, where $p_i \geq 0$, $\sum p_i = 1$. Given $k$ boxes and $n$ balls placed in box $i$ with $p_i$, the multinomial distribution gives the probability of $n_l$ balls in box $l$ for every $n$. We have
        \begin{align*}
            \Omega &= \left\{(n_1, \dots, n_k) \in \NN^k: \sum_{i=1}^k n_i = n\right\} \\
            \PP(n_1, \dots, n_k) &= \binom n {n_1} n_1^{p_1} \cdot \binom {n-n_1}{n_2} p_2^{n_2} \cdots \binom{n-n_1-\dots-n_{k-1}}{n_k} p_k^{n_k} \\
            &= \binom n {n_1, n_2, \dots, n_k} p_1^{n_1} p_2^{n_2} \dots p_k^{n_k}
        \end{align*}
        \item \emph{Geometric distribution}. The geometric distribution gives the probabiity that the first heads achieved on a $p$-coin is after $k$ tosses:
        \begin{align*}
            \Omega &= \{1, 2, \dots\} \\
            p_k &= (1-p)^{k-1}p
        \end{align*}
        Alternatively, we might want to model the probability that we see $k$ tails before the first heads:
        \begin{align*}
            \Omega &= \{0, 1, \dots\} \\
            p_k' &= (1-p)^kp
        \end{align*}
        \item \emph{Poisson distribution}. Suppose customers arrive into a shop in $[0, 1]$. Discretise $[0, 1]$ into $\left\{\left[\frac{i-1}{N}, \frac{i}{N}\right]\right\}, i = 1, \dots, N$, $N \in \NN$. Independently, in each interval, a customer arrives with probability $p$, and doesn't arrive with probability $1-p$.
        
        Then
        \begin{align*}
            \PP(k \text{ customers have arrived}) = \binom N k p^k (1-p)^{N-k}
        \end{align*}
        -- in other words, this follows a binomial distribution. Now, take $p = \frac{\lambda} N$, so
        \begin{align*}
            \PP(k \text{ customers have arrived}) &= \frac{N!}{k!(N-k)!}\left(\frac{\lambda}{N}\right)^k\left(1-\frac{\lambda}{N}\right)^{N-k} \\
            &= \frac{\lambda^k}{k!} \cdot \underbrace{\frac{N(N-1)\dots(N-k+1)}{N^k}}_{\to 1} \underbrace{\left(1-\frac{\lambda}{N}\right)^{N-k}}_{\to e^{-\lambda}} \\
            &\to e^{-\lambda}\frac{\lambda^k}{k!}
        \end{align*}
        as $N \to \infty$. Hence we have
        \begin{align*}
            \Omega &= \{0, 1, 2, \dots\} \\
            p_k &= e^{-\lambda}\frac{\lambda^k}{k!}
        \end{align*}
        and we denote this $\Poi(\lambda)$.
    \end{enumerate}
    
    \subsection{Random variables}

    \begin{definition}
        \label{random-variables}
        Given a probability space $(\Omega, \mathcal F, \PP)$, a \emph{random variable} is a function
        \begin{align*}
            X: \Omega \to \RR
        \end{align*}
        satisfying
        \begin{align*}
            \{\omega \in \Omega : X(\omega) \leq x\} \in \mathcal F
        \end{align*}
        for any $x$. We have the shorthand notation $\{X \leq x\}$ to refer to $\{\omega \in \Omega: X(\omega) \leq x\}$, and $\{X \in A\} = \{\omega \in \Omega: X(\omega) \in A\}$.
    \end{definition}
    
    \begin{definition}
        Let $A \in \mathcal F$. Define the \emph{indicator} of $A$
        \begin{align*}
            1_A(\omega) = 1(\omega \in A) = \begin{cases}
                1 & \omega \in A \\
                0 & \omega \notin A
            \end{cases}
        \end{align*}
        Since $A \in \mathcal F$, $1_A$ is a random variable.        
    \end{definition}

    \begin{definition}
        \label{probability-distribution-function}
        Suppose $X$ is a random variable. The \emph{probability distribution function} of $X$ is given by
        \begin{align*}
            F_X(x): \RR &\to [0, 1] \\
            x &\mapsto \PP(\{\omega \in \Omega: X(\omega) \leq x\}) = \PP(X \leq x)
        \end{align*}
    \end{definition}
    
    \begin{definition}
        $X = (X_1, \dots, X_n)$ is called a \emph{random variable in $\RR^n$} if
        \begin{align*}
            (X_1, \dots, X_n): \Omega \to \RR^n
        \end{align*}
        is a function such that, for any $x_1, \dots, x_n$,
        \begin{align*}
            \{\omega: X_1(\omega) \leq x_1, \dots, X_n(\omega) \leq x_n\} \in \mathcal F
        \end{align*}
        which we denote with the shorthand $\{X_1 \leq x_1, \dots, X_n \leq x_n\}$.
    \end{definition}
    
    Note that $(X_1, \dots, X_n)$ is a random variable if and only if all of $X_1, \dots, X_n$ are.

    \begin{definition}
        A random variable $X$ is called \emph{discrete} if it takes values in some countable set $S$.
    \end{definition}
    
    \begin{definition}
        For a discrete random variable, we are able to write
        \begin{align*}
            p_x = \PP(X = x) = \PP({\omega: X(\omega) = x})
        \end{align*}
        Then, $(p_x)_{x \in S}$ is called the \emph{probability mass function} of $X$.        
    \end{definition}
    
    Now we are able to say things like
    \begin{align*}
        \text{for any }A \subseteq S, \PP(X \in A) = \sum_{x \in A}p_x
    \end{align*}
    If $p_x$ follows a given distribution, then we say $X$ is a random variable with that distribution or a random variable following that distribution.
    
    \begin{definition}
        Let $X_1, \dots, X_n$ be discrete random variables with values in $S_1, \dots, S_n$. They are called \emph{independent} if
        \begin{align*}
            \PP(X_1 = x_1, \dots, X_n = x_n) = \PP(X_1 = x_1) \dots \PP(X_n = x_n)
        \end{align*}
        for all $x_1 \in S_1, \dots, x_n \in S_n$.
    \end{definition}
    
    \begin{example}
        Toss a $p$-coin $N$ times independently. Then
        \begin{align*}
            \Omega = \{0, 1\}^N
        \end{align*}
        and every $\omega \in \Omega$ is of the form $(\omega_1, \dots, \omega_N)$, where $\omega_i$ is the $i^\text{th}$ outcome. Hence
        \begin{align*}
            p_{\omega} = \PP(\{\omega\}) = \prod_{k=1}^N p^{\omega_k} (1-p)^{1-\omega_k}
        \end{align*}
        Let us also define random variables $X_k$ for each $k$, satisfying $X_k(\omega) = \omega_k$. Clearly $X_k$ is a Bernoulli r.v. of parameter $p$. We claim $X_1, \dots, X_N$ are independent: we have
        \begin{align*}
            P(X_1 = x_1, \dots, X_N = x_N) &= \prod_{k=1}^N p^{x_\omega}(1-p)^{1-x_\omega} \\
            &= \prod_{k=1}^N \PP(X_i = x_i)
        \end{align*}
    \end{example}

    \subsection{Expectation}

    \begin{definition}
        Suppose $\Omega$ is finite or countable, and let $X: \Omega \to \RR$ be a discrete random variable. We say $X$ is \emph{non-negative} if $X \geq 0$ -- then we define the \emph{expectation} of $X$ by
        \begin{align*}
            \EE[X] = \sum_{\omega \in \Omega} \PP(\{\omega\}) \cdot X(\omega)
        \end{align*}
    \end{definition}
    
    We can equivalently let
    \begin{align*}
        \Omega_X = \{X(\omega): \omega \in \Omega\}
    \end{align*}
    so that
    \begin{align*}
        \EE[X] &= \sum_{\omega \in \Omega}X(\omega) \cdot \PP(\{\omega\}) \\
        &= \sum_{x \in \Omega_X} \sum_{\omega \in \{X = x\}} X(\omega) \PP(\{\omega\}) \\
        &= \sum_{x \in \Omega_X} \sum_{\omega \in \{X = x\}} x \PP(\{\omega\}) \\
        &= \sum_{x \in \Omega_X} x \sum_{\omega \in \{X = x\}} \PP(\{\omega\}) \\
        &= \sum_{x \in \Omega_X} x \cdot  \PP(X = x)
    \end{align*}
    or in other words the average of the values taken by $X$, weighted by their probabilities.
    
    \begin{example}
        Suppose $X \sim \Bin(n, p)$. Then
        \begin{align*}
            \EE[X] &= \sum_{k=0}^N k\PP(X = k) \\
            &= \sum_{k=0}^N k \binom N k p^k (1-p)^{N-k} \\
            &= \sum_{k=0}^N k \frac{N!}{(N-k)!k!}p^k(1-p)^{N-k} \\
            &= \sum_{k=0}^{N-1} N \cdot \binom{N-1}{k} p^{k-1} (1-p)^{N-k} \cdot p \\
            &= Np \cdot (p+1-p)^{N-1} = Np
        \end{align*}
    \end{example}
    
    \begin{example}
        Suppose $X \sim \Poi(\lambda)$. Then
        \begin{align*}
            \EE[X] &= \sum_{k=0}^\infty k \PP(X = k) \\
            &= \sum_{k=0}^\infty ke^{-\lambda}\frac{\lambda^k}{k!} \\
            &= e^{-\lambda} \cdot \lambda \sum_{k=0}^\infty \frac{\lambda^{k-1}}{(k-1)!} = \lambda
        \end{align*}
    \end{example}
    
    Let $X$ be a random variable -- then we define
    \begin{align*}
        X^+ &= \max(X, 0) \\
        X^- &= \max(-X, 0)
    \end{align*}
    so $X = X^+ - X^-$ and $|X| = X^+ + X^-$. Supposing $X$ is discrete, we can define the expectations of $\EE[X^+]$ and $\EE[X^-]$, since they are both non-negative.

    If at least one of $\EE[X^+]$ and $\EE[X^-]$ is finite, we define
    \begin{align*}
        \EE[X] = \EE[X^+] - \EE[X^-]
    \end{align*}
    If both are infinite, the expectation of $X$ is undefined.

    Also, if $\EE[|X|]$ (which we define in the obvious way) is $< \infty$, we say $X$ is integrable. When $\EE[X]$ is well-defined, we have
    \begin{align*}
        \EE[X] = \sum_{x \in \Omega_X} x \cdot \PP(X = x)
    \end{align*}
    
    We have some properties of expectations as follows:
    \begin{itemize}
        \item If $X \geq 0$, then $\EE[X] \geq 0$.
        \item If $X \geq 0$ and $\EE[X] = 0$, then $\PP(X = 0) = 1$.
        \item If $c \in \RR$, then $\EE[cX] = c\EE[X]$ and $\EE[c + X] = c + \EE[X]$.
        \item If $X$ and $Y$ are random variables (and $X$, $Y$ integrable), then $\EE[X + Y] = \EE[X] + \EE[Y]$.
        \item In the countable case, we take $c_1, \dots, c_n \in \RR$ and $X_1, \dots, X_n$ integrable, and then we have
        \begin{align*}
            \EE\left[\sum_{i=1}^n c_i X_i\right] = \sum_{i=1}^n c_i \EE[X_i]
        \end{align*}
    \end{itemize}
    
    Let us give a proof of the final identity in the case where $\Omega$ is countable. We have
    \begin{align*}
        \EE\left[\sum_n X_n\right] &= \sum_{\omega \in \Omega} \left(\sum_n X_n(\omega)\right) \PP(\{\omega\}) \\
        &= \sum_n \sum_{\omega \in \Omega} X_n(\omega) \PP(\{\omega\}) \\
        &= \sum_n \EE[X_n]
    \end{align*}
    where we can interchange sums by the assumption that the random variables are integrable.

    We also have the useful result that, if $X = 1(A)$ for some event $A$,
    \begin{align*}
        \EE[X] = \PP(A)
    \end{align*}
    
    \begin{proposition}
        Let $g: \RR \to \RR$, and define the random variable $g(X)$ by
        \begin{align*}
            g(X)(\omega) = g(X(\omega))
        \end{align*}
        Then
        \begin{align*}
            \EE[g(X)] = \sum_{x \in \Omega_X} g(x) \PP(X = x)
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Put $Y = g(X)$, then
        \begin{align*}
            \{Y = y\} &= \{\omega: Y(\omega) = y\} \\
            &= \{\omega: g(X(\omega)) = y\} \\
            &= \{\omega: X(\omega) \in g^{-1}(\{y\})\} \\
            &= \{X \in g^{-1}(\{y\})\}
        \end{align*}
        Hence
        \begin{align*}
            \EE[Y] &= \sum_{y \in \Omega_Y} y \cdot \PP(Y = y) \\
            &= \sum_{y \in \Omega_Y} y \cdot \PP(X \in g^{-1}(\{y\})) \\
            &= \sum_{y \in \Omega_Y} y \sum_{x \in g^{-1}(\{y\})} \PP(X = x) \\
            &= \sum_{y \in \Omega_Y} \sum_{x \in g^{-1}(\{y\})} g(x) \PP(X = x) \\
            &= \sum_{x \in \Omega_X} g(x) \cdot \PP(X = x)
        \end{align*}
    \end{proof}
    
    \begin{proposition}
        Suppose that $X \geq 0$ and takes integer values. Then
        \begin{align*}
            \EE[X] = \sum_{k=1}^\infty \PP(X \geq k) = \sum_{k=0}^\infty \PP(X > k)
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Since $X \geq 0$ and $X \in \NN$, we have
        \begin{align*}
            X = \sum_{k=1}^\infty 1(X \geq k) = \sum_{k=0}^\infty 1(X > k)
        \end{align*}
        Taking expectations, we obtain
        \begin{align*}
            \EE[X] &= \EE\left[\sum_{k=1}^\infty 1(X \geq k)\right] &&= \EE\left[\sum_{k=0}^\infty 1(X > k)\right] \\
            &= \sum_{k=1}^\infty \EE[1(X \geq k)] &&= \sum_{k=0}^\infty \EE[1(X > k)] \\
            &= \sum_{k=1}^\infty \PP(X \geq k) &&= \sum_{k=0}^\infty \PP(X > k)
        \end{align*}
    \end{proof}

    \subsection{Moments, variance and covariance}

    \begin{definition}
        Let $X$ be a random variable and $r \in \NN$. We call $\EE[X^r]$, as long as it is well-defined, the \emph{$r^\text{th}$ moment} of $X$.
    \end{definition}
    
    \begin{definition}
        The \emph{variance} of $X$ is given by
        \begin{align*}
            \Var(X) = \EE[\left(X-\EE[X]\right)^2]
        \end{align*}
        Furthermore, the \emph{standard deviation} of $X$ is given by $\sqrt{\Var(X)}$.        
    \end{definition}
    
    We have some properties of variance as follows:
    \begin{itemize}
        \item $\Var(X) \geq 0$, since it is the expectation of a squared quantity;
        \item If $\Var(X) = 0$, then $\PP(X = \EE[X]) = 1$;
        \item If $c \in \RR$, then $\Var(cX) = c^2\Var(X)$ and $\Var(X + c) = \Var(X)$.
    \end{itemize}
    
    \begin{lemma}
        The variance of $X$ is also given by
        \begin{align*}
            \Var(X) = \EE[X^2] - \EE[X]^2
        \end{align*}
    \end{lemma}
    
    \begin{proof}
        We have
        \begin{align*}
            \Var(X) &= \EE[X^2 - 2X\EE[X] + (\EE[X])^2] \\
            &= \EE[X^2] - 2\EE[X]\EE[X] + (\EE[X])^2 \\
            &= \EE[X^2] - \EE[X]^2
        \end{align*}
    \end{proof}
    
    \begin{lemma}
        The function $\EE[(X-c)^2]$ is minimised when $c=\EE[X]$, and at that point is equal to $\Var(X)$.
    \end{lemma}
    
    \begin{proof}
        We have
        \begin{align*}
            f(c) &= \EE[(X-c)^2] \\
            &= \EE[X^2] - 2c\EE[X] + c^2 \\
            \implies f'(c) &= -2\EE[X] + 2c = 0 \\
            \implies c &= \EE[X]
        \end{align*}
    \end{proof}
    
    \begin{example}
        \begin{enumerate}[label=\protect\circled{\arabic*}]
            \item Suppose $X \sim \Bin(n, p)$. Recall $\EE[X] = np$. We have
            \begin{align*}
                \Var(X) = \EE[X(X-1)] + \EE[X] - \EE[X]^2
            \end{align*}
            Then
            \begin{align*}
                \EE[X(X-1)] &= \sum_{k=2}^n k(k-1) \frac{n!}{k!(n-k)!}p^k(1-p)^{n-k} \\
                &= (n-1)np^2 \sum_{k=0}^{n-2} \binom{n-2}{k} p^k (1-p)^{n-2-k} \\
                &= (n-1)np^2
            \end{align*}
            so that $\Var(X) = (n-1)np^2 + np - n^2p^2 = np(1-p)$.
            \item Suppose $X \sim \Poi(\lambda)$. Recall $\EE[X] = \lambda$. We have
            \begin{align*}
                \EE[X(X-1)] &= \sum_{k=2}^\infty k(k-1) e^{-\lambda}\frac{\lambda^k}{k!} \\
                &= e^{-\lambda}l^2 \sum_{k=2}^\infty \frac{\lambda^{k-2}}{(k-2)!} = \lambda^2
            \end{align*}
            So $\Var(X) = \lambda^2 + \lambda - \lambda^2 = \lambda$.
        \end{enumerate}
    \end{example}
    
    \begin{definition}
        Let $X$ and $Y$ be random variables. Then the \emph{covariance} of $X$ and $Y$ is
        \begin{align*}
            \Cov(X, Y) = \EE[(X-\EE[X])(Y-\EE[Y])]
        \end{align*}
    \end{definition}
    We have some properties:
    \begin{itemize}
        \item $\Cov(X, Y) = \Cov(Y, X)$;
        \item $\Cov(X, X) = \Var(X)$;
        \item for any $c \in \RR$, $\Cov(cX, Y) = c\Cov(X, Y)$ and $\Cov(X + c, Y) = \Cov(X, Y)$;
        \item for any $c \in \RR$, $\Cov(c, X) = 0$.
    \end{itemize}
    
    \begin{lemma}
        The covariance of $X$ and $Y$ is given by
        \begin{align*}
            \Cov(X, Y) = \EE[XY] - \EE[X]\EE[Y]
        \end{align*}
    \end{lemma}
    
    \begin{proof}
        Expand.
    \end{proof}

    \begin{lemma}
        For random variables $X$ and $Y$, we have
        \begin{align*}
            \Var(X+Y) = \Var(X) + \Var(Y) + 2\Cov(X, Y)
        \end{align*}
    \end{lemma}
    
    \begin{proof}
        We have
        \begin{align*}
            \Var(X + Y) &= \EE[((X-\EE[X])+(Y-\EE[Y]))^2] \\
            &= \Var(X) + 2\Cov(X, Y) + \Var(Y)
        \end{align*}
    \end{proof}
    
    \begin{lemma}
        For $c_1, \dots, c_n$ and $d_1, \dots, d_n \in \RR$,
        \begin{align*}
            \Cov\left(\sum_{i=1}^n c_iX_i, \sum_{j=1}^n d_iY_i\right) = \sum_{i,j=1}^n c_id_j \Cov(X_i, Y_j)
        \end{align*}
    \end{lemma}
    
    \begin{corollary}
        We have
        \begin{align*}
            \Var\left(\sum_{i=1}^n X_i\right) &= \Cov\left(\sum_{i=1}^n X_i, \sum_{i=1}^n X_i\right) \\
            &= \sum_{i=1}^n \Var(X_i) + \sum_{i \neq j}\Cov(X_i, X_j)
        \end{align*}
    \end{corollary}
    
    Let us consider some results on independence.

    \begin{lemma}
        Suppose $X_1$, $X_2$ and $X_3$ are independent -- then $X_1$ and $X_2$ are independent.
    \end{lemma}

    \begin{proof}
        We have
        \begin{align*}
            \PP(X_1 = x_1, X_2 = x_2) &= \sum_{x_3} \PP(X_1 = x_1, X_2 = x_2, X_3 = x_3) \\
            &= \sum_{x_3} \PP(X_1 = x_1) \PP(X_2 = x_2) \PP(X_3 = x_3) \\
            &= \PP(X_1 = x_1)\PP(X_2 = x_2)
        \end{align*}
    \end{proof}
    
    \begin{remark*}
        The result obviously generalises to any subset of a set of independent events being independent.
    \end{remark*}
    
    \begin{lemma}
        Let $X$ and $Y$ be two independent random variables, and let $f$, $g: \RR \to \RR_{+}$. Then
        \begin{align*}
            \EE[f(X)g(Y)] &= \EE[f(X)]\EE[g(Y)]
        \end{align*}
    \end{lemma}
    
    \begin{proof}
        Let the random variable $Z = (X, Y)$, so $h(Z) = f(X)g(Y)$. Then
        \begin{align*}
            \EE[h(Z)] &= \sum_{(x, y)}h(x, y) \PP(Z = (x, y)) \\
            &= \sum_{(x, y)}f(x)g(y) \PP(X = x, Y = y) \\
            &= \sum_{(x, y)}f(x)g(y) \PP(X = x)\PP(Y = y) \\
            &= \EE[f(X)] \cdot \EE[g(Y)]
        \end{align*}
    \end{proof}
    
    \begin{proposition}
        If $X$ and $Y$ are independent, $\Cov(X, Y) = 0$.
    \end{proposition}
    
    \begin{remark*}
        The converse is not true.
    \end{remark*}
    
    \begin{example}
        Let $X_1, X_2, X_3$ be independent Bernoulli random variables with $p=\frac{1}{2}$. Let $Y_1 = 2X_1-1$, $Y_2 = 2X_2-1$. Also let $Z_1 = Y_1X_3$, $Z_2 = Y_2X_3$. Then $\EE[Y_1] = \EE[Y_2] = 0$, and $\EE[Z_1] = \EE[Z_2] = 0$ by independence. Then
        \begin{align*}
            \Cov(Z_1, Z_2) = \EE[Z_1Z_2] = \EE[Y_1Y_2X_3^2]
        \end{align*}
        But we claim $Z_1$ and $Z_2$ are not independent. Observe $\PP(Z_1 = 0, Z_2 = 0) = \PP(X_3 = 0) = \frac{1}{2}$. But $\PP(Z_1 = 0) = \PP(Z_2 = 0) = \frac 1 2$.
    \end{example}
    
    \subsection{Inequalities on random variables}

    \begin{proposition}[Markov's inequality]
        Let $X$ be a non-negative random variable and $a > 0$. Then
        \begin{align*}
            \PP(X \geq a) \leq \frac{\EE[X]}{a}
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        We have
        \begin{align*}
            X &\geq a \cdot 1(X \geq a) \\
            \implies E[X] &\geq a \cdot \PP(X \geq a)
        \end{align*}
        by taking expectations.
    \end{proof}
    
    \begin{proposition}[Chebyshev's inequality]
        Let $X$ be a random variable with $\EE[X] < \infty$. Then, for any $a > 0$,
        \begin{align*}
            \PP(|X - \EE[X]| \geq a) \leq \frac{\Var(X)}{a^2}
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Observe
        \begin{align*}
            \PP(|X-\EE[X]| \geq a) &= \PP((X-\EE[X])^2 \geq a^2) \\
            &\leq \frac{\EE[(X-\EE[X])^2]}{a^2} && \text{by Markov}\\
            &= \frac{\Var(X)}{a^2}
        \end{align*}
    \end{proof}
    
    \begin{proposition}[Cauchy-Schwarz]
        Let $X$ and $Y$ be random variables, then
        \begin{align*}
            \EE[|XY|] \leq \sqrt{\EE[X^2] \cdot \EE[Y^2]}
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Assume that $\EE[X^2]$ and $\EE[Y^2]$ are finite, otherwise there is nothing to prove. Then, $|XY| \leq \frac 1 2 (X^2 + Y^2)$, so, taking expectations, $\EE[|XY|]$ is also finite. We may also assume $\EE[X^2] > 0$, $\EE[Y^2] > 0$. Let us also assume for the time being that $X, Y \geq 0$.

        Let $t \in \RR$ and consider $(X-tY)^2$. We have
        \begin{align*}
            0 &\leq \EE[(X-tY)^2] \\
            &= \EE[X^2] - 2t\EE[XY] + t^2\EE[Y^2] := f(t)
        \end{align*}
        Then $f'(t) = -2\EE[XY] + 2t\EE[Y^2]$, so $\min f(t)$ is achieved at $t^* = \frac{\EE[XY]}{\EE[Y^2]}$. We hence have
        \begin{align*}
            0 &\leq \EE[X^2] - 2\frac{\EE[XY]^2}{\EE[Y^2]} + \frac{\EE[XY]^2}{\EE[Y^2]} \\
            \implies \EE[XY]^2 &\leq \EE[X^2] \cdot \EE[Y^2]
        \end{align*}
    \end{proof}
    
    \begin{remark*}
        We could also have just put in our value for $t*$ from the beginning -- the minimisation approach is purely to provide rationale.
    \end{remark*}
    
    \begin{remark*}
        When is equality achieved? We would need $f(t^*) = 0$, that is
        \begin{align*}
            \EE[(X - t^*Y)^2] &= 0 \\
            \implies P(X = t^*Y) &= \PP(X - t^*Y = 0) = 1
        \end{align*}
        so $X$ is just a multiple of $Y$.        
    \end{remark*}
    
    \begin{definition}[Convexity]
        A function $f: \RR \to \RR$ is called \emph{convex} if, for any $x$, $y \in \RR$ and any $t \in [0, 1]$,
        \begin{align*}
            f(tx + (1-t)y) \leq tf(x) + (1-t)f(y)
        \end{align*}
    \end{definition}
    
    \begin{proposition}[Jensen's inequality]
        Let $X$ be a random variable, and $X: \RR \to \RR$ a convex function. Then
        \begin{align*}
            \EE[f(X)] \geq f(\EE[X])
        \end{align*}
    \end{proposition}
    
    \begin{remark*}
        We can remember the direction of the inequality by putting $f(x) = x^2$ and using non-negativity of the variance.
    \end{remark*}
    
    \begin{lemma}
        Let $f: \RR \to \RR$ be a convex function. Then $f$ is the supremum of all the lines lying below it. In other words, for all $m$ in $\RR$, there exist $a$, $b$ in $\RR$ such that $f(m) = am + b$ and $f(x) \geq ax + b$ for all $x$.
    \end{lemma}
    
    \begin{proof}[Proof of lemma]
        Let $m \in \RR$, and suppose $x < m < y$. Then $m = tx + (1-t)y$ for some $t \in [0, 1]$, or, equivalently,
        \begin{align*}
            t(m-x) = (1-t)(y-m) && (*)
        \end{align*}
        By convexity, we have
        \begin{align*}
            f(m) = f(tx + (1-t)y) &\leq tf(x) + (1-t)f(y) \\
            \implies t(f(m) - f(x)) &\leq (1-t)(f(y)-f(m)) \\
            (*) \implies \frac{f(m)-f(x)}{m-x} &\leq \frac{f(y)-f(m)}{y-m}
        \end{align*}
        Then set
        \begin{align*}
            a = \sup_{x < m} \frac{f(m)-f(x)}{m-x}
        \end{align*}
        so that, for any $x < m < y$,
        \begin{align*}
             \frac{f(m)-f(x)}{m-x} \leq a \leq \frac{f(y)-f(m)}{y-m}
        \end{align*}
        Hence, by rearrangement, we conclude $f(z) \geq a(z-m) + f(m)$, for any $z$ (depending on whether $z$ is greater than $m$ or less than $m$, we use either the left or right inequality.)        
    \end{proof}
    
    \begin{proof}[Proof of proposition]
        Let $m = \EE[X]$. By the claim, we have $a$, $b \in \RR$ such that $f(m) = am + b$ and $f(x) \geq ax + b$ for all $x$. We have
        \begin{align*}
            f(X) &\geq aX + b \\
            \implies \EE[f(X)] &\geq a\EE[X] + b = am + b = f(m) = f(\EE[X])
        \end{align*}
    \end{proof}
    
    \begin{remark*}
        When is equality achieved?
        
        Suppose $f$ convex with the extra property that, for $n = \EE[X]$, there exist $a$, $b \in \RR$ with $f(m) = am + b$ but $f(x) > ax + b$, note the strict inequality, for $x \neq m$. When should $X$ satisfy $\EE[f(X)] = f(\EE[X])$?

        Since $f(X) \geq aX + b$, we have $f(X) - (aX + b) \geq 0$. Taking expectations, we have
        \begin{align*}
            \EE[f(X)] \geq a\EE[X] + b = am + b = f(m) = f(\EE[X])
        \end{align*}
        Since we want $\EE[f(X)] = f(\EE[X])$, we must have
        \begin{align*}
            \EE[f(X) - (aX + b)] &= 0 \\
            \implies \PP(f(X) = aX + b) &= 1
        \end{align*}
        but then, since $f(x) > ax + b$ for $x \neq m$, we conclude $\PP(X = m) = 1$ (if we didn't have that extra part, we'd conclude $f$ is just a linear function).
        
    \end{remark*}

    \begin{proposition}[AM-GM inequality]
        Let $x_1, \dots, x_n > 0$. Then 
        \begin{align*}
            \left(\prod_{k=1}^n x_k\right)^{\frac 1 n} \leq \frac 1 n \sum_{k=1}^n x_k
        \end{align*}
    \end{proposition}
    
    \begin{lemma}
        Let $f$ be a convex function. Then, for any $x_1, \dots, x_n \in \RR$ we have
        \begin{align*}
            \frac 1 n \sum_{k=1}^n f(x_k) \geq f\left(\frac 1 n \sum_{k=1}^n x_k\right)
        \end{align*}
    \end{lemma}
    
    \begin{proof}[Proof of lemma]
        Let $X$ be a random variable with $\PP(X = x_i) = \frac 1 n$, $i=1, \dots, n$. Then
        \begin{align*}
            \frac 1 n \sum_{k=1}^n f(x_k) = \sum_x f(x) \PP(X = x) = \EE[f(X)]
        \end{align*}
        so Jensen's gives
        \begin{align*}
            \EE[f(X)] \geq f(\EE[X]) = f\left(\frac 1 n \sum_{k=1}^n x_k\right)
        \end{align*}
        as required.        
    \end{proof}
    
    \begin{proof}[Proof of proposition]
        Let $f(x) = -\log x$ and suppose $x_1, \dots, x_n > 0$. Then
        \begin{align*}
            -\frac 1 n \sum_{k=1}^n \log x_k &\geq -\log \left(\frac 1 n \sum_{k=1}^n x_n\right) \\
            \implies \left(\prod_{k=1}^n x_k\right)^{\frac 1 n} &\leq \frac 1 n \sum_{k=1}^n x_k
        \end{align*}
        as required.        
    \end{proof}
    
    \subsection{Joint distribution and conditional expectation}

    Recall if $X$ is a discrete random variable, then the disttribution of $X$ is $(\PP(X = x))_x$.

    \begin{definition}
        Let $X_1, \dots, X_n$ be discrete random variables. Their \emph{joint distribution} is defined to be the set of values of
        \begin{align*}
            \PP(X_1 = x_1, \dots, X_n = x_n)
        \end{align*}
        for any $x_1, \dots, x_n$.        
    \end{definition}
    
    \begin{remark*}
        We can recover the distribution of any one of the variables from such a joint distribution with
        \begin{align*}
            \PP(X_1 = x_1) = \sum_{x_2, \dots, x_n} \PP(X_1 = x_1, X_2 = x_2, \dots, X_n = x_n)
        \end{align*}
        In this context the distribution of the single variable is sometimes called its \emph{marginal} distribution.        
    \end{remark*}
    
    \begin{definition}
        Let $X$ and $Y$ be discrete random variables. The \emph{conditional distribution} of $X$ given $Y = y$ is
        \begin{align*}
            \PP(X = x \mid Y = y) = \frac{\PP(X = x, Y = y)}{\PP(Y = y)}
        \end{align*}
    \end{definition}
    By the law of probability we of course have
    \begin{align*}
        \PP(X = x) = \sum_y \PP(X = x\mid Y = y)\PP(Y = y)
    \end{align*}
    
    % \subsubsection{Distribution of the sum of independent random variables}

    Let $X$ and $Y$ be independent -- what is the distribution of $X+Y$? We have
    \begin{align*}
        \PP(X + Y = z) &= \sum_y \PP(X + Y = z, Y = y) \\
        &= \sum_y \PP(X = z-y, Y = y) \\
        &= \sum_y \PP(X = z-y)\PP(Y = y)
    \end{align*}
    where the last line is by independence. This is of course a convolution.

    \begin{example*}
        Let $X \sim \Poi(\lambda)$ and $Y \sim \Poi(\mu)$, with $\lambda, \mu > 0$ and $X \perp Y$. Then
        \begin{align*}
            \PP(X + Y = n) &= \sum_{k=0}^n \PP(X = n-k)\PP(Y = k) \\
            &= e^{-\lambda}\frac{\lambda^{n-k}}{(n-k)!} \cdot e^{-\mu}\frac{\mu^k}{k!} \\
            &= \frac{e^{-(\lambda+\mu)}}{n!} \sum_{k=0}^n \binom n k \lambda^{n-k} \mu^k \\
            &= e^{-(\lambda+\mu)}\frac{(\lambda+\mu)^n}{n!}
        \end{align*}
        so in particular $X + Y \sim \Poi(\lambda + \mu)$.        
    \end{example*}
    Let us derive conditional expectation for random variables. First, take $B \in \mathcal F$ to be an event with $\PP(B) > 0$. We have
    \begin{align*}
        \EE[X \mid B] = \frac{\EE[X \cdot 1(B)]}{\PP(B)}
    \end{align*}
    and, if $X = 1(A)$, then
    \begin{align*}
        \EE[1(A) \mid B] = \frac{\EE[1(A) \cdot 1(B)]}{\PP(B)} = \frac{\EE[1(A \cap B)]}{\PP(B)} = \PP(A \mid B)
    \end{align*}
    We can now formulate the \emph{law of total expectation}.
    \begin{proposition}[Law of total expectation]
        Let $X$ be a random variable and $(\Omega_n)$ a partition of $\Omega$ with $\PP(\Omega_n) > 0$ for any $n$. Then
        \begin{align*}
            \EE[X] = \sum_n \EE[X \mid \Omega_n] \PP(\Omega_n)
        \end{align*}
    \end{proposition}
    \begin{proof}
        Use indicators. We have
        \begin{align*}
            \EE[X] = \EE[X \cdot 1(\Omega)] = \EE\left[X \cdot \sum_n 1(\Omega_n)\right] = \sum_n \EE[X \mid \Omega_n]\PP(\Omega_n)
        \end{align*}
    \end{proof}
    Now, we want to condition expectation not on an event but on a particular random variable. We have
    \begin{align*}
        \EE[X \mid Y = y] &= \frac{\EE[X \cdot 1(Y=y)]}{\PP(Y=y)} \\
        &= \sum_x x \cdot \frac{\PP(X = x, Y = y)}{\PP(Y = y)} \\
        &= \sum_x x \cdot \PP(X = x \mid Y = y)
    \end{align*}
    Let us now define $g(y) = \EE[X \mid Y = y]$.
    \begin{definition}
        The expectation of $X$ given $Y$, $\EE[X \mid Y]$, is a random variable which is a function of $Y$. In particular,
        \begin{align*}
            \EE[X \mid Y] = g(Y)
        \end{align*}
    \end{definition}
    
    \begin{example*}
        Toss a $p$-coin independently $n$ times, and let $X_i = 1(i^\text{th} \text{ toss is H})$. Let $Y_n = X_1 + \dots + X_n$. Let us attempt to calculate $\EE[X_1 \mid Y_n]$. First, let
        \begin{align*}
            g(y) = \EE[X_1 \mid Y_n = y] &= \frac{\EE[X_1 \cdot 1(Y_n=y)]}{\PP(Y_n=y)} \\
            &= \frac{\PP(X_1 = 1, Y_n = y)}{\PP(Y_n=y)} \\
            &= \frac{p \binom{n-1}{y-1} p^{y-1} (1-p)^{n-y}}{\binom{n}{y} p^y (1-p)^{n-y}} \\
            &= \frac{y}{n}
        \end{align*}
        Hence, $\EE[X_1 \mid Y_n] = \frac{Y_n}{n}$.        
    \end{example*}
    We have some simple and uninspiring properties:
    \begin{itemize}
        \item $\EE[cX \mid Y] = c\EE[X \mid Y]$;
        \item $\EE[c + X \mid Y] = c + \EE[X \mid Y]$;
        \item $\EE[c \mid Y] = c$;
        \item If $X_1, \dots, X_n$ are random variables,
        \begin{align*}
            \EE\left[\sum_{i=1}^n X_i \mid Y\right] = \sum_{i=1}^n \EE[X_i \mid Y]
        \end{align*}
    \end{itemize}
    and some more interesting properties. We have
    \begin{align*}
        \EE[X \mid Y] &= \sum_y 1(Y = y) \EE[X \mid Y = y] \\
        \implies \EE[\EE[X \mid Y]] &= \sum_y \EE[X \mid Y = y]\PP(Y = y)= \EE[X]
    \end{align*}
    Suppose now $X$ and $Y$ are independent random variables. Then
    \begin{align*}
        \EE[X \mid Y] &= \sum_y 1(Y = y) \EE[X \mid Y = y] \\
        &= \sum_y 1(Y = y) \sum_x x \cdot \PP(X = x \mid Y = y) \\
        &= \sum_y 1(Y = y) \sum_x x \cdot \PP(X = x) \\
        &= \sum_y 1(Y = y) \EE[X] = \EE[X]
    \end{align*}
    
    \begin{proposition}
        Let $Y$ and $Z$ be independent. Then
        \begin{align*}
            \EE[\EE[X \mid Y] \mid Z] = \EE[X]
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Let $g(Y) = \EE[X \mid Y]$, $g(y) = \EE[X \mid Y = y]$. We claim $g(Y)$ is independent of $\ZZ$. Observe
        \begin{align*}
            \PP(g(Y) = w, Z = z) &= \sum_{y \in g^{-1}(\{\omega\})} \PP(Y = y, Z = z) \\
            &= \sum_{y \in g^{-1}(\{\omega\})} \PP(Y = y) \PP(Z = z) \\
            &= \PP(g(Y) = w) \PP(Z = z)
        \end{align*}
        Hence, previous results give $\EE[g(Y) \mid Z] = \EE[g(Y)]$, and this is $\EE[\EE[X \mid Y]] = \EE[X]$.        
    \end{proof}
    
    \begin{proposition}
        Let $X$, $Y$ be two random variables, and $h: \RR \to \RR$. Then
        \begin{align*}
            \EE[h(Y) \cdot X \mid Y] = h(Y) \cdot \EE[X \mid Y]
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        We have
        \begin{align*}
            \EE[h(Y) \cdot X \mid Y] &= \sum_y 1(Y = y) \EE[h(Y) \cdot X \mid Y = y] \\
            &= \sum_y 1(Y = y) h(y) \EE[X \mid Y = y] \\
            &= h(Y) \sum_y 1(Y = y) \EE[X \mid Y = y] \\
            &= h(Y) \EE[X \mid Y]
        \end{align*}
    \end{proof}
    
    \begin{corollary}
        \[\EE[\EE[X \mid Y] \mid Y] = \EE[X \mid Y]\]
    \end{corollary}
    
    \begin{example*}
        Toss a $p$-coin independently $n$ times, and let $X_i = 1(i^\text{th} \text{ toss is H})$. Let $Y_n = X_1 + \dots + X_n$. What is $\EE[X_1 \mid Y_n]$?

        By symmetry, $\EE[X_1 \mid Y_n] = \EE[X_i \mid Y_n]$ for any $i$, $1 \leq i \leq n$. Then
        \begin{align*}
            Y_n = \EE[Y_n \mid Y_n] = \EE[X_1 + \dots + X_n \mid Y_n] = n \EE[X_1 \mid Y_n] 
        \end{align*}
        which was what we wanted.
    \end{example*}
    
    \section{Random walks}

    \begin{definition}[Random process]
        A \emph{random process} or \emph{stochastic process} is a sequence of random variables $(X_n)$.
    \end{definition}
    
    \begin{definition}[Random walks]
        A \emph{random walk} is a random process that can be expressed as
        \begin{align*}
            X_n = x + Y_1 + \dots + Y_n
        \end{align*}
        where $x$ is a number and $(Y_i)$ are independent identically distributed (i.i.d.) random variables. We call the $Y_i$ the \emph{steps} of the random variable.         
    \end{definition}
    
    A \emph{simple random walk} has
    \begin{align*}
        Y_i = \begin{cases}
            1 & \text{with probability }p \\
            -1 & \text{with probability }q = 1-p
        \end{cases}
    \end{align*}
    We will write
    \begin{itemize}
        \item $\PP_x(A) = \PP(A \mid X_0 = x)$;
        \item $h(x) = \PP_x((X_n) \text{ reaches } a \text{ before }0)$, where $a$ is some arbitrary constant at which the game ends.
    \end{itemize}
    so $h(0) = 0$, $h(a) = 1$. Then
    \begin{align*}
        h(x) &= \PP_x((X_n) \text{ hits } a \text{ bef. }0) \\
        &= \PP_x((X_n) \text{ hits } a \text{ bef. }0 \mid Y_1 = 1) \cdot p + \PP_x((X_n) \text{ hits } a \text{ bef. }0 \mid Y_1 = -1) \cdot q \\
        &= p \cdot h(x+1) + q \cdot h(x-1)
    \end{align*}

    \lecture{Lecture 12 -- 18/02/26}
    
    \begin{itemize}
        \item \emph{Case $p=q=\frac 1 2$}, then $X_n$ is a \emph{simple symmetric random walk}. We have
        \begin{align*}
            h(x) &= \frac 1 2 h(x+1) + \frac 1 2 h(x-1) \\
            \implies h(x+1) - h(x) &= h(x) - h(x-1) \\
            \implies \Aboxed{h(x) &= cx}
        \end{align*}
        for some constant $c$. Since $h(a) = 1$, $c = \frac 1 a$, and hence $h(x) = \frac h a$.
        \item \emph{Case $p\neq q$}. Let us try a solution of the form $\lambda^x$ for some lambda -- we have
        \begin{align*}
            \lambda^x &= p\lambda^{x+1} + q\lambda^{x-1} \\
            \implies \lambda &= 1, \lambda = \frac q p
        \end{align*}
        so the general solution has the form
        \begin{align*}
            h(x) = A + B\left(\frac q p\right)^x
        \end{align*}
        which, applying boundary conditions, gives
        \begin{align*}
            \Aboxed{h(x) = \frac{\left(\frac q p\right)^x - 1}{\left(\frac q p\right)^a-1}}
        \end{align*}
    \end{itemize}
    
    These results are called the \emph{gambler's ruin estimates}.

    Now let us consider a new quantity -- let $T = \min\{n \geq 0: X_n \in \{0, a\}\}$ -- the first $n$ such that $X_n$ is either 0 or $a$, i.e. the duration of the game. We want to find $\EE[T \mid X_0 = x] = \EE_x[T] = k(x)$. We have from the law of total expectation
    \begin{align*}
        k(x) &= \EE_x[T \mid Y_1 = 1]\cdot\PP(Y_1 = 1) + \EE_x[T \mid Y_1 = -1]\cdot\PP(Y_1 = -1) \\
        &= p(1 + k(x+1)) + q(1 + k(x-1)) \\
        &= 1 + pk(x+1) + qk(x-1)
    \end{align*}
    with $k(0) = k(a) = 0$.

    \begin{itemize}
        \item \emph{Case $p=q=\frac 1 2$}. Try a solution of the form $Ax^2$, to get
        \begin{align*}
            Ax^2 &= 1 + pA(x+1)^2 + qA(x-1)^2 \\
            \implies A &= -1
        \end{align*}
        giving a general solution of the form $k(x) = -x^2 + Bx + C$. Applying boundary conditions, we obtain
        \begin{align*}
            \Aboxed{k(x) = x(a-x)}
        \end{align*}
        \item \emph{Case $p\neq q$}. Try $k(x) = Cx$ to get
        \begin{align*}
            Cx &= 1 + pC(x+1) + qC(x-1) \\
            \implies c &= \frac 1 {q-p}
        \end{align*}
        Adding this to the solution of the homogeneous equation, we obtain
        \begin{align*}
            k(x) = A + \frac 1 {q-p} x + B\left(\frac q p\right)^x
        \end{align*}
        which, after boundary conditions, gives
        \begin{align*}
            \Aboxed{k(x) = \frac 1 {q-p} x - \frac a {q-p}\cdot\frac{\left(\frac q p\right)^x - 1}{\left(\frac q p\right)^a-1}}
        \end{align*}
    \end{itemize}
    
    \section{Probability generating functions}

    Let $X$ be a random variable with values in $\NN$ -- recall $p_r = \PP(X=r)$ for $r \in \NN$ constitutes the probability mass function.

    \begin{definition}[Probability generating function]
        The \emph{probability generating function (PGF)} for a random variable $X$ with values in $\NN$ and probability masses $(p_r)$ is
        \begin{align*}
            p(z) = \EE[z^X] = \sum_{r=0}^\infty p_r z^r
        \end{align*}
        with $|z| \leq 1$.        
    \end{definition}
    
    \begin{remark*}
        This is well-defined because
        \begin{align*}
            |p(z)| = \left|\sum_{r=0}^\infty p_rz^r\right| \leq \sum_{r=0}^\infty p_r |z|^r \leq \sum_{r=0}^\infty p_r = 1
        \end{align*}
        so $p$ converges absolutely for $|z| \leq 1$.        
    \end{remark*}
    
    \begin{theorem}
        The pgf uniquely determines the distribution of $X$. \label{pgf-unique}
    \end{theorem}
    
    \begin{proof}
        Let $(p_r)$ and $(q_r)$ be two probability distributions with the same pgf's -- for all $|z| \leq 1$, $\sum p_r z^r = \sum q_r z^r$.

        Plugging in $z=0$ gives $p_0 = q_0$. We now proceed by strong induction. Suppose $p_r = q_r$ for all $r \leq n$, so
        \begin{align*}
            \sum_{r=n+1}^\infty p_r z^r &= \sum_{r=n+1}^\infty q_rz^r
        \end{align*}
        Clearly dividing by $z^{n+1}$ and setting $z=0$ gives $p_{n+1} = q_{n+1}$ as required.        
    \end{proof}

    \begin{proposition}
        Suppose $X_1, \dots, X_n$ are independent random variables with pgf's $q_i(z)$, and let $X = X_1 + \dots + X_n$ have pgf $p$. Then
        \begin{align*}
            p(z) = q_1(z)\dots q_n(z)
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Clearly
        \begin{align*}
            p(z) = \EE[z^{X_1+\dots+X_n}] = \EE[X_1] \dots \EE[X_n] = q_1(z) \dots q_n(z)
        \end{align*}
        by independence.        
    \end{proof}

    \begin{example*}
        Let us calculate some pgf's for important distributions.
        \begin{enumerate}[label=\protect\circled{\arabic*}]
            \item \emph{Binomial}. Suppose $X \sim \Bin(n, p)$. Then
            \begin{align*}
                p(z) &= \sum_{k=0}^n z^k \binom n k p^k (1-p)^{n-k} \\
                &= (pz + 1-p)^n
            \end{align*}
            \item \emph{Geometric}. Suppose $X \sim \Geo(p)$. Then
            \begin{align*}
                \EE[z^X] &= \sum_{r=0}^\infty z^r (1-p)^r p \\
                &= \frac p {1-z(1-p)}
            \end{align*}
            \item \emph{Poisson}. Suppose $X \sim \Poi(\lambda)$. Then
            \begin{align*}
                \EE[z^X] &= \sum_{r=0}^\infty z^r e^{\lambda} \frac{\lambda^r}{r!} \\
                &= e^{\lambda(z-1)}
            \end{align*}
        \end{enumerate}
    \end{example*}
    
    \begin{example*}
        This fact can make calculating the pmf of sums of random variables easier. For example, suppose $X \sim \Bin(n, p)$ and $Y \sim \Bin(m, p)$ with $X \perp Y$. Then
        \begin{align*}
            \EE[z^{X+Y}] = \EE[z^X] \EE[z^Y] &= (pz+1-p)^n \cdot (pz+1-p)^m \\
            &= (pz+1-p)^{n+m}
        \end{align*}
        from which it follows that $X + Y \sim \Bin(n + m, p)$.

        What about Poisson? Suppose $X \sim \Poi(\lambda)$, $Y \sim \Poi(\mu)$, and $X \perp Y$. Then
        \begin{align*}
            \EE[z^{X+Y}] = e^{\lambda(z-1)}e^{\mu(z-1)} = e^{(\lambda+\mu)(z-1)}
        \end{align*}
        from which it follows that $X + Y \sim \Poi(\lambda + \mu)$.
    \end{example*}

    For the following results, it will be helpful to introduce the notation
    \begin{align*}
        p(1^-) = \lim_{z \to 1^-} p(z)
    \end{align*}

    \begin{proposition}
        Let $X$ be a discrete random variable with pgf $p(z)$, then
        \begin{align*}
            \EE[X] = p'(1^-)
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        \begin{itemize}
            \item Assume $\EE[X] < \infty$, and take $0 < z < 1$. We have
            \begin{align*}
                p'(z) = \sum_{r=1}^\infty rp_r z^{r-1} \leq \sum_{r=1}^\infty rp_r = \EE[X]
            \end{align*}
            Hence $p'$ is upper bounded by $\EE[X]$, and it is clearly increasing, so the limit exists and is $\leq \EE[X]$.

            Let $\eps > 0$. There exists $N$ such that
            \begin{align*}
                \sum_{r=1}^N rp_r \geq \EE[X] - \eps
            \end{align*}
            since this is what it means for the infinite sum to be equal to $\EE[X]$. Furthermore,
            \begin{align*}
                \lim_{z \to 1^-} p'(z) \geq \lim_{z \to 1^-} \sum_{r=1}^N rp_r z^{r-1} = \sum_{r=1}^N rp_r \geq \EE[X] - \eps
            \end{align*}
            \item If $\EE[X] = \infty$, then, for any $M \geq 0$, we have $N$ such that
            \begin{align*}
                \sum_{r=1}^N rp_r \geq M
            \end{align*}
            Then, as above,
            \begin{align*}
                \lim_{z \to 1^-} p'(z) \geq \lim_{z \to 1^-} \sum_{r=1}^N rp_r z^{r-1} = \sum_{r=1}^N rp_r \geq M
            \end{align*}
            so the limit is infinite.
        \end{itemize}
    \end{proof}
    
    \begin{proposition}
        Let $X$ be a discrete random variable with pgf $p(z)$, then
        \begin{align*}
            \EE[X(X-1)\dots(X-k+1)] = p^{(k)}(1^-)
        \end{align*}
    \end{proposition}
    
    \begin{corollary}
        Let $X$ be a discrete random variable with pgf $p(z)$, then
        \begin{align*}
            \Var(X) = p''(1^-) + p'(1^-) - (p'(1^-))^2
        \end{align*}
    \end{corollary}
    
    \begin{proposition}
        Let $X$ be a discrete random variable with pgf $p(z)$, then
        \begin{align*}
            \PP(X = n) = \frac 1{n!}\left.\frac{d^np}{dz^n}\right|_{z=0}
        \end{align*}
    \end{proposition}
    
    \begin{remark*}
        This is another way to see \cref{pgf-unique}.
    \end{remark*}
    
    \subsection{Sum of a random number of random variables}

    Let $X_1, X_2, \dots$ be i.i.d. random variables and let $N$ be an independent random variable with values in $\NN$. Set
    \begin{align*}
        S_n = X_1 + \dots + X_n
    \end{align*}
    that, is
    \begin{align*}
        S_n(\omega) = \sum_{i=1}^n X_i(\omega)
    \end{align*}
    Passing to the random variable $N$, we obtain
    \begin{align*}
        S_N(\omega) = S_{N(\omega)}(\omega) = \sum_{i=1}^{N(\omega)} X_i(\omega)
    \end{align*}
    
    \begin{lemma}
        Let $q(z) = \EE[z^N]$ be the pgf of $N$, and $p(z) = \EE[z^{X_i}]$ be the pgf of $X_i$. Then
        \begin{align*}
            r(z) = \EE[z^{S_N}] = q(p(z))
        \end{align*}
    \end{lemma}
    
    \begin{proof}[Proof 1]
        We have
        \begin{align*}
            r(z) = \EE[z^{S_N}] &= \sum_{n=0}^\infty \EE[z^{S_N} \cdot 1(N = n)] \\
            &= \sum_{n=0}^\infty \EE[z^{S_n} \cdot 1(N = n)] \\
            &= \sum_{n=0}^\infty \EE[z^{X_1+\dots+X_n} \cdot 1(N = n)] \\
            &= \sum_{n=0}^\infty \EE[z^{X_1+\dots+X_n}] \PP(N = n) \\
            &= \sum_{n=0}^\infty (p(z))^n \PP(N = n) \\
            &= q(p(z))
        \end{align*}
    \end{proof}
    
    \begin{proof}[Proof 2]
        We have
        \begin{align*}
            r(z) = \EE[z^{S_n}] = \EE[\EE[z^{S_N} \mid N]] = g(N)
        \end{align*}
        where
        \begin{align*}
            g(n) &= \EE[z^{S_N} \mid N = n] \\
            &= \EE[z^{S_n}] \\
            &= p(z)^n
        \end{align*}
        so that
        \begin{align*}
            r(z) &= \EE[p(z)^n] \\
            &= q(p(z))
        \end{align*}
    \end{proof}
    
    \begin{corollary}
        We have
        \begin{align*}
            \EE[S_n] &= \EE[N] \cdot \EE[X_1] \\
            \Var(S_n) &= \EE[N] \cdot \Var(X_1)+ \Var(N) \cdot \EE[X_1]^2
        \end{align*}
    \end{corollary}
    
    \begin{proof}
        The first result follows from
        \begin{align*}
            \EE[S_n] &= r'(1-) \\
            &= \lim_{z\to 1^-} r'(z) \\
            &= \lim_{z\to 1^-} q'(p(z)) \lim_{z\to 1^-} p'(z) 
        \end{align*}
        Clearly $p(z)$ approaches 1 from below as $z$ approaches 1 from below, so this is just
        \begin{align*}
            q'(1-)p'(1-) = \EE[N] \cdot \EE[X_1]
        \end{align*}
        as required.   
        
        The second result is left as an exercise.
    \end{proof}
    
    \section{Branching processes}
    
    Think of a branching process as a series of generations of offspring. We have $X_0 = 1$, there being one person in the first generation; then, in each generation, each individual produces $k$ offspring independently with a fixed distribution (in particular, the distribution of $X_1$)

    Let $Y_{k,n}$ with $k \geq 1$, $n \geq 0$ be i.i.d. random variables with the distribution of $X_1$, interpreted as the number of offspring produced by the $k^\text{th}$ individual in the $n^\text{th}$ generation. Hence
    \begin{align*}
        X_{n+1} = \begin{cases}
            Y_{1, n} + \dots + Y_{X_n, n} & X_n \geq 1 \\
            0 & X_n = 0
        \end{cases}
    \end{align*}
    
    \begin{proposition}
        \label{expected-branching}
        We have
        \begin{align*}
            \EE[X_n] = \EE[X_1]^n
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        We will proceed by induction. The base case is clear. Let us consider the conditional expectation $\EE[X_{n+1} \mid X_n = m]$ -- we have
        \begin{align*}
            \EE[X_{n+1} \mid X_n = m] &= \EE[Y_{1,n} + \dots + Y_{m,n} \mid X_n = m] \\
            &= \EE[Y_{1,n} + \dots + Y_{m,n}] && \text{(independence)} \\
            &= m\EE[X_1]
        \end{align*}
        so that $\EE[X_{n+1} \mid X_n] = X_n\EE[X_1]$. Hence $\EE[X_{n+1}] = \EE[\EE[X_{n+1} \mid X_n]] = \EE[X_n\EE[X_1]] = \EE[X_n] \EE[X_1]$, which completes the result by induction.        
    \end{proof}
    
    \begin{proposition}
        Let $G(z) = \EE[z^{X_1}]$, $G_n(z) = \EE[z^{X_n}]$, then
        \begin{align*}
            G_{n+1}(z) = G(G_n(z))
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Again, let us consider $\EE[z^{X_{n+1}} \mid X_n = m]$. We have
        \begin{align*}
            \EE[z^{X_{n+1}} \mid X_n = m] &= \EE[z^{Y_{1, n} + \dots + Y_{m, n}} \mid X_n = m] \\
            &= \EE[z^{Y_{1, n} + \dots + Y_{m, n}}] \\
            &= (G(z))^m
        \end{align*}
        Hence $G_{n+1}(z) = \EE[\EE[z^{X_{n+1}} \mid X_n]] = \EE[G(z)^{X_n}] = G_n(G(z))$
    \end{proof}
    
    \subsection{Extinction probability}

    Let $q_n = \PP(X_n = 0)$, and $q = \PP(X_n = 0 \text{ for some }n \geq 0)$. Letting $A_n$ be the event that $X_n = 0$ for some $n$, we can see $q = \PP(\cup A_n)$, and $A_n \subseteq A_{n+1}$, so the $A_n$ form an increasing sequence and we get
    \begin{align*}
        \PP(A_n) \to \PP\left(\bigcup A_n\right)
    \end{align*}
    by \cref{continuity} (continuity of probability measures).
    
    \begin{lemma}
        Let $G(z) = \EE[z^{X_1}]$, then $q_{n+1} = G(q_n)$ and $q = G(q)$.
    \end{lemma}
    
    \begin{proof}[Proof 1]
        We have
        \begin{align*}
            q_{n+1} &= \PP(X_{n+1} = 0) \\
            &= G_{n+1}(0) \\
            &= G(G_n(0)) \\
            &= G(q_n)
        \end{align*}
        Then, since $G$ is continuous and $q_n \to q$, we can simply take the limit on both sides to get $q=G(q)$ as required.
    \end{proof}
    
    \begin{proof}[Proof 2]
        We will condition $q_{n+1}$ on the number of individuals in the first generation, i.e. $X_1$. Supposing $X_1 = m$, let $X_n^{(1)}, \dots, X_n^{(m)}$ be i.i.d. branching processes with the same distribution as the original one. Hence
        \begin{align*}
            X_{n+1} = X_n^{(1)} + \dots + X_n^{(m)}
        \end{align*}
        So
        \begin{align*}
            q_{n+1} &= \PP(X_{n+1} = 0) \\
            &= \sum_m \PP(X_{n+1} = 0 \mid X_1 = m) \PP(X_1 = m) \\
            &= \sum_m \PP(X_n^{(1)} + \dots + X_n^{(m)} = 0 \mid X_1 = m) \PP(X_1 = m) \\
            &= \sum_m \left(\PP(X_n^{(1)} = 0)\right)^m \PP(X_1 = m) \\
            &= \sum_m q_n^m \PP(X_1 = m) \\ &= G(q_n)
        \end{align*}
        Once again, we take the limit on both sides.
    \end{proof}

    Recalling \cref{expected-branching}, we have that
    \begin{itemize}
        \item if $\EE[X_1] < 1$, $\EE[X_n] \to 0$ as $n \to \infty$, so we expect $q=1$;
        \item if $\EE[X_1] > 1$, $\EE[X_n] \to \infty$ as $n \to \infty$, so we expect $q<1$;
        \item if $\EE[X_1] = 1$, then $\EE[X_n] = 1$ for all $n$. But it is not clear what $q$ will be. Let us focus on this last case.
    \end{itemize}

    We can also interpret these results in the context of a graph of $G$. We know that the graph of $G$ will intersect $x$ at $x=1$, and we could either have another intersection in the range $0 \leq x < 1$, or no intersections.

    \begin{itemize}
        \item \emph{There is another intersection}. Then iterating $G$ starting from $G(0)$ will lead us to the other intersection. Furthermore, the gradient of $G$ at $1$, $G(1-) = \EE[X_1]$ must be greater than 1 for there to be such an intersection, as we expect;
        \item \emph{There is not another intersection}. Then iterating $G$ will take us all the way to 1, and the gradient of $G$ at 1, $G(1-) = \EE[X_1]$ must be less than or equal to 1 for there not to be any further intersections. We conclude that $q = 1$ in the case $\EE[X_1] = 1$.
    \end{itemize}
    
    \begin{proposition}
        Assume $P(X_1 = 1) < 1$. The extinction probability $q$ is the minimal non-negative solution to $G(x) = x$. Furthermore, $q < 1$ if and only if $\EE[X_1] > 1$.
    \end{proposition}
    
    \begin{proof}
        We know $q = G(q)$. Let $t$ be the smallest minimal solution of $x=G(x)$. We will show by induction that, for any $n$, $q_n \leq t$, and then take limits to establish $q \leq t \implies q = t$.

        For the base case, $0 = q_0 \leq t$. Suppose $q_n \leq t$, then
        \begin{align*}
            q_{n+1} = G(q_n) \leq G(t) = t
        \end{align*}
        since $G$ is increasing.      
        
        Now, we show $q < 1$ if and only if $\EE[X_1] > 1$. Suppose for now $g_0 + g_1 = \PP(X_1) \leq 1 < 1$.

        Let $H(z) = G(z) - z = \sum_{r=0}^\infty g_rz^r - z$, so $H(1) = 0$. First, we show that $H$ can have at most one root in $[0, 1)$. Observe
        \begin{align*}
            H''(z) = \sum_{r=0}^\infty r(r-1)g_rz^{r-1} > 0
        \end{align*}
        in $(0, 1)$, because $g_0 + g_1 < 1$, so there must be some later $r$ with a positive value of $g_r$, making the sum strictly larger than zero. Hence $H'(z)$ is strictly increasing, and hence can have at most one more root other than 1 -- otherwise, we would have a point $c$ with $H'(c) = 0$ by Rolle's theorem.

        We now split into two cases.
        \begin{itemize}
            \item $H$ has no root other than 1, so $q=1$. Then $H'(1-) = G'(1-) - 1 = \EE[X_1] - 1$. Furthermore, $H(0) = G(0) = g_0 \geq 0$ while $H(1) = 0$, so $H(z) \geq 0$ for all $z \in [0, 1]$. Now
            \begin{align*}
                H'(1-) = \lim_{z \to 1^-} \frac{H(z)-H(0)}{z-1} \leq 0
            \end{align*}
            since the denominator is $H(z) - 0 \geq 0$ while the denominator is $z-1 \leq 0$. Hence $H'(1-) \leq 0 \implies \EE[X_1] \leq 1$, as required.
            \item $H$ has one more root $r$, so $r$ is the extinction probability. We hence have $H(r) = 0$ and $H(1) = 0$ with $r<1$, so we have $z \in (r, 1)$ with $H'(z) = 0$ by Rolle. Hence $\EE[X_1] - 1 = H'(-1) > H'(z) = 0$, so $\EE[X_1] > 1$ as required. 
        \end{itemize}

        On the other hand, if $\PP(X_1 \leq 1) = 1$, then $\EE[X_1] = g_1$, so
        \begin{align*}
            G(z) &= g_0 + g_1z \\
            &= 1 - \EE[X_1] + \EE[X_1]z
        \end{align*}
        Then, if $G(z) = z$, we have $(1-\EE[X_1])z = 1-\EE[X_1]$ and, since $\EE[X_1] = g_1 < 1$, $1-\EE[X_1] \neq 0$ so $z=1$.
        
    \end{proof}
    
    \section{Continuous random variables}

    Recall \cref{random-variables}, that

    \begin{definition*}
        Given a probability space $(\Omega, \mathcal F, \PP)$, a \emph{random variable} is a function
        \begin{align*}
            X: \Omega \to \RR
        \end{align*}
        satisfying
        \begin{align*}
            \{\omega \in \Omega : X(\omega) \leq x\} \in \mathcal F
        \end{align*}
        for any $x$.
    \end{definition*}
    
    and \cref{probability-distribution-function}, that

    \begin{definition*}
        Suppose $X$ is a random variable. The \emph{probability distribution function} of $X$ is given by
        \begin{align*}
            F_X(x): \RR &\to [0, 1] \\
            x &\mapsto \PP(\{\omega \in \Omega: X(\omega) \leq x\}) = \PP(X \leq x)
        \end{align*}
    \end{definition*}
    
    When dealing with discrete random variables, it was possible to only consider $\PP(X = x)$ for some $x$. But now, we will have to work in more generality.

    Let us observe some important properties.
    \begin{enumerate}[label=\protect\circled{\arabic*}]
        \item if $x \leq y$, then $F(x) \leq F(y)$;
        \item for any $a < b$,
        \begin{align*}
            \PP(a < x \leq b) = F(b) - F(a)
        \end{align*}
    \end{enumerate}
