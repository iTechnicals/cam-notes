    \section{Groups}

    \subsection{Groups -- a reminder}

    \begin{definition}
        A \emph{group} is a triple $(G, \cdot, e)$ where $G$ is a set, 
        \begin{align*}
            \cdot: G \times G \to G
        \end{align*}
        is a function, and $e \in G$ such that
        \begin{itemize}
            \item \emph{Associativity}. $(a \cdot b) \cdot c = a \cdot (b \cdot c)$;
            \item \emph{Identity}. For all $a \in G$, $a \cdot e = e \cdot a = a$;
            \item \emph{Inverses}. For all $a \in G$, there exists $a^{-1} \in G$ such that $a \cdot a^{-1} = a^{-1} \cdot a = e$.
        \end{itemize}
        The size of $G$, $|G|$, is called the \emph{order} of the group. 
    \end{definition}
    
    \begin{proposition}[Uniqueness of inverses]
        Inverses are unique.
    \end{proposition}

    \begin{proof}
        Left as an exercise (see 1A).
    \end{proof}
    
    \begin{definition}[Subgroups]
        If $G$ is a group and $H \subseteq G$, then we say $H$ is a \emph{subgroup} if
        \begin{itemize}
            \item $e \in H$;
            \item for all $a$, $b \in H$, $a \cdot b \in H$;
            \item $(H, \cdot, e)$ is a group.
        \end{itemize}
        In order to say item 3, we must first establish 1 and 2.
    \end{definition}
    
    \begin{proposition}
        A non-empty subset $H \subseteq G$ is a subgroup if and only if, for all $h_1$, $h_2 \in H$, $h_1h_2^{-1} \in H$.
    \end{proposition}
    
    \begin{proof}
        Left as an exercise (see 1A).
    \end{proof}
    
    \begin{definition}
        A group $G$ is called \emph{abelian} if, for all $g_1$, $g_2 \in G$, $g_1g_2 = g_2g_1$.
    \end{definition}
    
    \begin{example}
        \begin{itemize}
            \item We have $(\ZZ, +)$, $(\QQ, +)$, $\RR$, $\CC$, $\ZZ/n\ZZ$.
            \item The symmetric groups $S_n$: permutations of $\{1, \dots, n\}$ under composition. More generally for any set $X$, we have $\Sym(X)$.
            \item The dihedral groups $D_{2n}$: the group of symmetries of a regular $n$-gon.
            \item The general linear group over $\RR$, $\GL(n, \RR)$ or $\GL_n(\RR)$, is the set of $\{f: \RR^n \to \RR^n \mid f \text{ is an }\RR\text{-linear bijection}\}$. Since $\GL(n, \RR)$ is made of functions, we can think of it a subset of $\Sym(\RR^n)$. (Equivalently, it's the set of $n$ by $n$ invertible matrices with elements in $\RR$).
        \end{itemize}
        Some examples of subgroups:
        \begin{itemize}
            \item The even permutations $A_n \leq S_n$ (the alternating group);
            \item Rotational symmetries of an $n$-goin, $C_n \leq D_{2n}$;
            \item Determinant one matrices $\SL(n, \RR) \leq \GL(n, \RR)$.
        \end{itemize}
        Other examples include $C_2 \times C_2$, $Q_8 = \{\pm 1, \pm i, \pm j, \pm k\}$.        
    \end{example}
    
    \begin{definition}
        Let $G$ be a group, $g \in G$ and $H \leq G$ a subgroup. Then the \emph{left coset}
        \begin{align*}
            gH = \{gh: h \in H\}
        \end{align*}
    \end{definition}
    Basic properties include that $g \in gH$, so cosets are non-empty. Also, cosets partition $G$ -- they have no non-trivial overlaps and every $g$ lies in some coset. Furthermore, the map
    \begin{align*}
        H &\to gH \\
        h &\mapsto gh
    \end{align*}
    defines a bijection.
    
    \begin{theorem}[Lagrange]
        If $G$ is a finite group and $H \leq G$ then
        \begin{align*}
            |G| = |H| \cdot |G:H|
        \end{align*}
        where $|G:H|$, the index of $H$ in $G$, is the number of cosets of $H$ in $|G|$.
    \end{theorem}
    
    \begin{definition}
        Let $G$ be a group and let $g \in G$. The \emph{order} of $g$, $\ord(g)$, is the smallest positive integer $n$ such that $g^n = e$, or infinity if no such $n$ exists.
    \end{definition}
    
    \begin{proposition}
        Let $g \in G$. Then, $\ord(g) \mid |G|$.
    \end{proposition}
    
    \begin{proof}
        Let $g \in G$ and consider $H = \{e, g, g^2, \dots, g^{n-1}\}$ where $n = \ord(g)$. This is a subgroup: it is non-empty, and $g^r \cdot g^{-s} = g^{r-s} \in H$ (after potentially multiplying by $g^n = e$). Also, the elements of $H$ are distinct -- if $g^i = g^j$ for $0 \leq i < j \leq n$, then $g^{j-i} = e$, contradicting the minimality of $\ord(g)$. Hence $|H| = ord(g)$ -- now apply Lagrange.
    \end{proof}
    
    \subsection{Normal subgroups, quotients and homomorphisms}

    Let $H \leq G$. When are two cosets of $H$ in $G$ equal? If $gH = g'H$, then $g \in g'H$, so $g = g'h$ for some $h \in H$. In other words, $gH = g'H \iff (g')^{-1}g \in H$.

    We denote the set of left cosets of $H \leq G$ by $G/H = \{gH: g \in G\}$. Does $G/H$ carry a natural group structure?

    \lecture{Lecture 2 -- 24/01/26}

    We propose the `natural' group structure:
    \begin{align*}
        g_1H \cdot g_2H = (g_1g_2)H
    \end{align*}
    
    But this rule may depend on the coset representatives we pick. We must check that it doesn't. Suppose $g_2H = g_2'H$, so $g_2' = g_2h$ for some $h \in H$. Then
    \begin{align*}
        g_1H \cdot g_2'H = (g_1g_2')H = g_1g_2hH = g_1g_2H
    \end{align*}
    
    On the other hand, if $g_1H = g_1'H$, so $g_1' = g_1h$ for $h \in H$. Then
    \begin{align*}
        g_1'H \cdot g_2H = g_1hg_2H
    \end{align*}
    
    and that is not automatically equal to $g_1g_2H$ -- we need $g_2^{-1}hg_2 \in H$ to hold for all $g_2 \in G$, $h \in H$.

    \begin{definition}[Normal subgroups]
        A subgroup $H \leq G$ is \emph{normal} if, for all $g \in G$, $h \in H$, $ghg^{-1} \in H$. We write this as $H \norm G$.
    \end{definition}
    
    \begin{remark}
        When $H \norm G$, then $G/H$ inherits a group structure by the rule above. We will refer to this as the \emph{quotient group of $G$ by $H$}.
    \end{remark}
    
    \begin{definition}[Homomorphisms]
        Let $G$ and $H$ be groups. A homomorphism from $G$ to $H$ is a function $f: G \to H$ such that, for all $g_1$, $g_2 \in G$, $f(g_1 g_2) = f(g_1) f(g_2)$.
    \end{definition}
    
    Results such as $f(e_G) = e_H$ follow immediately.
    
    \begin{lemma}
        If $f: G \to H$ is a homomorphism, then $f(g^{-1}) = f(g)^{-1}$.
    \end{lemma}
    
    \begin{proof}
        Calculate $f(g \cdot g^{-1})$ in two ways.
        \begin{enumerate}[label=\protect\circled{\arabic*}]
            \item $f(gg^{-1}) = f(e_G) = e_H$;
            \item $f(gg^{-1}) = f(g) f(g^{-1})$
        \end{enumerate}
        so $f(g) f(g^{-1}) = e_H$ and the result follows.        
    \end{proof}
    
    \begin{definition}
        Let $f: G \to H$ be a homomorphism. The \emph{kernel} of $f$ is
        \begin{align*}
            \ker f = \{g \in G : f(g) = e_H\}
        \end{align*}
        and the \emph{image} of $f$ is
        \begin{align*}
            \im f = \{h \in H: h = f(g) \text{ for some } g\}
        \end{align*}     
    \end{definition}
    
    \begin{proposition}
        Let $f: G \to H$ be a homomorphism. Then $\ker f \norm G$ and $\im f \leq H$.
    \end{proposition}
    
    \begin{proof}
        Observe $e \in \ker f$ so $\ker f$ non-empty. Now, if $x, y \in \ker f$, then
        \begin{align*}
            f(xy^{-1}) = f(x)f(y)^{-1} = e
        \end{align*}
        so $xy^{-1} \in \ker f$. For normality, let $x \in G$ and $g \in \ker f$. Now
        \begin{align*}
            f(xgx^{-1}) = f(x) e f(x)^{-1} = e
        \end{align*}
        so $xgx^{-1} \in \ker f$ and hence $\ker f \norm G$.
        
        For the image, observe that, if $a$, $b \in \im f$ with $a = f(x)$, $b = f(y)$, then $ab^{-1} = f(xy^{-1}) \in \im f$.
    \end{proof}
    
    \begin{remark*}
        Observe $f: G \to G/H$ taking $g \mapsto gH$ is a homomorphism. This is a nice way of expressing its `naturalness'.
    \end{remark*}
    
    \begin{definition}
        A homomorphism $f: G \to H$ is an \emph{isomorphism} if it is a bijection. Two groups $G$ and $H$ are said to be \emph{isomorphic}, $G \cong H$, if there exists an isomorphism between them.
    \end{definition}
    
    \begin{theorem}[First isomorphism theorem]
        Let $f: G \to H$ be a homomorphism, then $\ker f$ is normal in $G$ and the homomorphism
        \begin{align*}
            G/\ker f &\to \im f \\
            g \ker f &\mapsto f(g)
        \end{align*}
        is an isomorphism. In particular, $G/\ker f \cong \im f$.                
    \end{theorem}
    
    \begin{proof}
        We have already shown the normality result. Consider the map
        \begin{align*}
            \varphi: G/\ker f &\to \im f \\
            g\ker f &\mapsto f(g)
        \end{align*}
        \begin{itemize}
            \item \emph{$\varphi$ is well-defined}. If $g \ker f = g'\ker f$, then $g'g^{-1} \in \ker f$, so $f(g'g^{-1})= e$ which gives $f(g) = f(g')$ as required.
            \item \emph{$\varphi$ is surjective}. Let $h \in \im f$. Then $h = f(g)$ for some $g \in G$, and so $h = \varphi(g \ker f)$.
            \item \emph{$\varphi$ is injective}. Suppose $\varphi(g \ker f) = \varphi(g' \ker f)$, then $f(g) = f(g')$. But then $g'g^{-1} \in \ker f$, so $g \ker f = g' \ker f$.
        \end{itemize}
    \end{proof}
    
    \begin{example*}
        Consider the homomorphism
        \begin{align*}
            \exp: (\CC, +) &\to (\CC^*, \times) \\
            z &\mapsto e^z
        \end{align*}
        Then $\ker \exp = 2\pi i\ZZ$, so $\CC^* \cong \CC/2\pi i \ZZ$.
    \end{example*}
    
    \begin{theorem}[Second isomorphism theorem]
        Let $H \leq K$ and $K \norm G$. Then
        \begin{align*}
            HK = \{hk : h \in H, k \in K\}
        \end{align*}
        is a subgroup of $G$. Furthermore, $H \cap K$ is normal in $H$, and there is an isomorphism $HK/K \cong H/H \cap K$.
    \end{theorem}
    
    \begin{proof}
        We will show the results in order.
        \begin{itemize}
            \item \emph{$HK$ is a subgroup}. It contains $e_G$, so is non-empty. Now consider $x = hk$ and $y = h'k'$. We'll show $yx^{-1} \in HK$. Observe
            \begin{align*}
                yx^{-1} &= h'k'(hk)^{-1} \\
                &= h'k'k^{-1}h^{-1} \\
                &= (h'h^{-1})h(k'k^{-1})h^{-1} \\
                &= h'h^{-1}k''
            \end{align*}
            by normality of $K$.
            \item \emph{$H \cap K$ is a kernel}. Consider
            \begin{align*}
                \varphi: H &\to G/K \\
                h &\mapsto hK
            \end{align*}
            by restricitng the domain from $G \to G/K$ -- the kernel is $H \cap K$.
            \item \emph{Final statement}. By the first isomorphism theorem, we have
            \begin{align*}
                \im \varphi \cong H/(H \cap K)
            \end{align*}
            so we must show $\im \varphi \cong HK/K$. But $\im \varphi$ is the set of $K$-cosets in $G$ that are representable by elements of $H$, i.e. $\{hK : h \in H\} \subseteq G/K = HK/K$.
        \end{itemize}
    \end{proof}
    
    \subsubsection{Correspondence between subgroups of $G$ and $G/K$}

    Let $p: G \to G/K$ be the quotient homomorphism. We have a bijection
    \begin{align*}
        \{\text{subgroups of } G \text{ containing } K\}& \leftrightarrow \{\text{subgroups of } G/K\} \\
        K \norm L \leq G &\mapsto L/K \leq G/K \\
        \{g \in G : gK \in A\} &\mapsfrom A \leq G/K
    \end{align*}
    
    The forward bijection is just applying the projection map on a subset, and, since the quotient map is a homomorphism, its image is a subgroup; and the backward bijection is actually just applying the preimage of $p$. It is an exercise to check that the forward and backward directions are inverses.

    \begin{remark*}
        In fact the correspondence matches normal subgroups on the two sides.
    \end{remark*}
    
    \begin{theorem}[Third isomorphism theorem]
        Let $K \leq L \leq G$ be normal subgroups of $G$. Then there exists an isomorphism $(G/K)/(L/K) \cong G/L$.
    \end{theorem}
    
    \begin{proof}
        Define a map
        \begin{align*}
            \varphi: G/K &\to G/L \\
            gK &\mapsto gL
        \end{align*}
        \begin{itemize}
            \item \emph{$\varphi$ is well-defined}. If $gK = g'K$, then $g'g^{-1} \in K \subseteq L \implies gL = g'L$.
            \item \emph{$\varphi$ is a homomorphism}. Clear.
        \end{itemize}
        We have $\ker \varphi = \{gK: gL = L\} = \{gK: g \in L\} = L/K$. Hence we are done by the first isomorphism theorem.
    \end{proof}
    
    \begin{definition}[Simplicity]
        A group $G$ is called \emph{simple} if its only normal subgroups are $G$ and $\{e\}$.
    \end{definition}
    
    \begin{proposition}
        Let $G$ be an abelian group. Then $G$ is simple if and only if $G \cong C_p$ for some prime $p$.
    \end{proposition}
    
    \begin{proof}
        If $G \cong C_p$, then any non-identity element generates $G$ by, for example, Lagrange. So $G$ is simple.

        If $G$ is simple and abelian, then take any $g \neq e \in G$ and consider $\gen g$. This is a subgroup, and since $G$ is abelian, it is automatically normal. Hence, by simplicity, $\gen g = G$, and so $G$ is cyclic.

        If $G$ is infinite and cyclic, then $G \cong \ZZ$, but $\ZZ$ is not simple, so this is a contradiction. Hence $G$ is finite and $G \cong C_m$. If $q$ divides $m$, take the subgroup generated by $g^{m/q}$, where $g$ generates $G$. Normality is automatic, so $q=1$ or $q=m$ by simplicity -- in other words, $m$ is prime.
    \end{proof}
    
    \begin{theorem}
        Let $G$ be a finite group. Then, there exist subgroups
        \begin{align*}
            G = H_1 \triangleright H_2 \triangleright H_3 \triangleright \cdots \triangleright H_n = \{e\}
        \end{align*}
        such that $H_i/H_{i+1}$ is simple for all $i$.        
    \end{theorem}
    
    \begin{proof}
        If $G$ is simple, $H_2 = \{e\}$, and we're done. Otherwise, let $H_2$ be a proper normal subgroup of maximal order in $G$. We claim $G/H_2$ is simple. If not, consider the quotient map $\pi: G \to G/H_2$.

        If $N \norm G/H_2$ is a proper normal subgroup, by the correspondence theorem, we can find a proper normal subgroup in $G$ containing $H_2$. But this contradicts the maximality of $H_2$.

        Now we can just iterate -- note that $|H_{i+1}| < |H_i|$, so we will eventually terminate.
    \end{proof}
    
    \begin{remark*}
        This sequence is not necessarily unique.
    \end{remark*}
    
    \subsection{Group actions and permutations}

    \begin{definition}
        Let $X$ be a set. Let $\Sym(X)$ be the set of permutations on $X$. The identity is the identity function, and the group law is given by composition. If $X = \{1, \dots, n\}$, then we write $\Sym(X) = S_n$.
    \end{definition}
    
    \begin{itemize}
        \item It's convenient to write $\sigma \in S_n$ as a product of disjoint cycles.
        \item Every $\sigma \in S_n$ is a product of transpositions, and the number of such transpositions gives rise to the \emph{sign} of $\sigma$.
        \begin{align*}
            \sign: S_n &\to \{\pm 1\} \\
            \sigma &\mapsto \begin{cases}
                +1 & \sigma \text{ is a product of an even number of transpositions} \\
                -1 & \sigma \text{ ... odd ...}
            \end{cases}
        \end{align*}
        $\sign$ is a homomorphism, and is surjective when $n \geq 3$.
        
    \end{itemize}
    
    Let $X$ be a set and $G$ a group, and consider a homomorphism $G \to \Sym(X)$. This is called a \emph{permutation representation} of $G$.

    \begin{definition}[Group actions]
        An \emph{action} of a group $G$ on a set $X$ is a function
        \begin{align*}
            G \times X &\to X \\
            (g, x) &\mapsto g \cdot x
        \end{align*}
        such that
        \begin{itemize}
            \item $e \cdot x = x$ for all $x \in X$;
            \item $g_1 \cdot (g_2 x) = (g_1g_2) \cdot x$ for all $g_1, g_2 \in G$ and $x \in X$.
        \end{itemize}
        We say $G \acts X$.
    \end{definition}
    
    We should note the bijection between permutation representations and group actions.

    \begin{proposition}
        The map $a$ is a bijection.
        \begin{align*}
        a: \{\text{homomorphisms } G \to \Sym(X)\} &\longrightarrow \{\text{actions } G \acts X\} \\
        \varphi: G \to \Sym(X) &\longmapsto a(\varphi): G \times X \to X: (g, x) \mapsto \varphi(g) \cdot x
    \end{align*}
    \end{proposition}
    
    \begin{proof}
        Let us simply construct an inverse. Let $\cdot: G \times X \to X$ be an action. Define
        \begin{align*}
            \varphi: G &\to \Sym(X) \\
            g &\mapsto \varphi(g) := [g \cdot: X \to X]
        \end{align*}
        This is a permutation as $(g^{-1}) \cdot: X \to X$ is the inverse. Indeed,
        \begin{align*}
            \varphi(g^{-1})(\varphi(g)(x)) &= g^{-1} \cdot (g \cdot x) \\
            &= (g^{-1}g)x \\
            &= x
        \end{align*}
        To see that this rule gives a homomorphism is simple.
    \end{proof}
    
    We have some notation:
    \begin{itemize}
        \item $G^X = \im \phi \leq \Sym(X)$;
        \item $G_X = \ker \phi \triangleleft G$
    \end{itemize}
    It is a special case of the first isomorphism theorem that $G^X \cong G/G_X$.
    
    \begin{example}
        Let $G$ be the symmetry group of the unit cube, and let $X$ be the set of the cube's diagonals. Clearly $G$ acts on $X$. We have hence
        \begin{align*}
            \phi: G \to \Sym(X) \cong S_4
        \end{align*}
        We can see $\im \phi \cong S_4$, since we can swap any two diagonals -- then these transpositions generate the whole symmetric group. Furthermore, $\ker \phi = \{e, s\}$, where $s$ sends every vertex to its opposite. As an immediate consequence, we conclude $|G| = 48$.
    \end{example}
    
    \begin{example}
        For any group $G$, we can make $G$ act on itself (as a set) by
        \begin{align*}
            G \times G &\to G \\
            (g, g_1) &\mapsto gg_1
        \end{align*}
        which gives $\phi: G \to \Sym(G)$. The kernel $\ker \phi = \{e\}$, and so $\phi$ is injective -- we conclude that $G$ is isomorphic to a subgroup of $\Sym(G)$ by the first isomorphism theorem.
    \end{example}
    
    \begin{example}
        Let $H \leq G$ and $X = G/H$. Then $G \acts X$ by
        \begin{align*}
            (g, g_1H) \to gg_1H
        \end{align*}
        We get a homomorphism $\phi: G \to \Sym(X)$. Consider the kernel -- it is clear that $\ker \phi \subseteq H$. But more is true. If $g in \ker \phi$, then
        \begin{align*}
            gg_1H &= g_1H \\
            \implies g_1^{-1}gg_1 &\in H \\
            \implies \ker \phi &\subseteq \bigcap_{g_1 \in G}g_1Hg_1^{-1}
        \end{align*}
        that is, $g$ is contained in every conjugate of $H$. Furthermore, this containment is reversible: suppose $g \in \bigcap_{g_1 \in G} g_1Hg_1^{-1}$. Then $gg_1H = g_1H$, so $g \in \ker \phi$. We conclude
        \begin{align*}
            \ker \phi = \bigcap_{g_1 \in G} g_1Hg_1^{-1}
        \end{align*}
        If $H$ were normal, we would just have H on the right-hand side -- but when $H$ is not normal, we have an interesting way to make a smaller normal subgroup, $\ker \phi$, out of $H$.
    \end{example}
    
    \begin{theorem}
        Let $G$ be finite and let $H \leq G$ be of index $n$. Then, there exists a normal $K \triangleleft G$ with $K \subseteq H$ such that $G/K$ is isomorphic to a subgroup of $S_n$. Thus $|G/K|$ divides $n!$, and $|G/K| \geq n$.
    \end{theorem}
    
    \begin{proof}
        Consider $\phi: G \to \Sym(G/H)$ from the example. The kernel, call it $K$, is normal. We have shown it is contained in $H$ already -- applying the first isomorphism theorem, we have
        \begin{align*}
            G/K \cong \im \phi \leq \Sym(G/H) \cong S_n
        \end{align*}
        Lagrange gives $|G/K|$ divides $n!$, and $K \leq H$, so $|G/K| \geq |G/H| = n$.
    \end{proof}
    
    \begin{corollary}
        Let $G$ be a non-abelian simple group. Let $H \leq G$ be a proper subgroup with index $n > 1$. Then $G$ is isomorphic to a subgroup of $A_n$ with $n \geq 5$.
    \end{corollary}

    \begin{itemize}
        \item If $H$ contains an element of the form $\sigma = (1 \; 2 \; \dots \; \gamma)\tau$ with $\tau$ disjoint from $(1 \; 2 \; \dots \; \gamma)$ and $\gamma \geq 4$, then normality of $H$ gives that $\delta^{-1}\sigma\delta \in H$, so furthermore $\sigma^{-1}\delta^{-1}\sigma\delta \in H$. Since $\tau$ misses 1, 2, 3, it commutes with $\delta$, and with $(1 \; 2 \; \dots \; \gamma)$ by assumption.

        Then, expanding $\sigma^{-1}\delta^{-1}\sigma\delta$ out, we get $(2 \; 3 \; \gamma)$, so we have our 3-cycle.

        \item If $H$ contains an element of the form $\sigma = (1 \; 2 \; 3)(4 \; 5 \; 6)\tau$ in a disjoint cycle decomposition, take $\delta = (1 \; 2 \; 4)$ and calculate $\sigma^{-1}\delta^{-1}\sigma\delta$ -- this is a 5-cycle, bringing us into case 1.
        
        \item If $H$ contains $\sigma = (1 \; 2 \; 3)\tau$ where $\tau$ is a product of disjoint transpositions (if $\tau$ contained anything longer, we could relabel and apply cases 1 or 2). Then $\sigma^2$ is a 3-cycle.
        
        \item If $H$ contains $\sigma = (1 \; 2)(3 \; 4)\tau$ where $\tau$ is a product of disjoint transpositions, let $\delta = (1 \; 2 \; 3)$ and consider $\mu = \sigma^{-1}\delta^{-1}\sigma\delta = (1 \; 4)(2 \; 3)$. Then let $\nu = (1 \; 5 \; 2)\mu (1 \; 2 \; 5) = (1 \; 3)(4 \; 5)$ -- since $\mu\nu = (1 \; 2 \; 3 \; 4 \; 5)$, we are in case 1.
    \end{itemize}

    \subsection{Finite $p$-groups}

    Let $p$ be a prime. Groups of order $p^n$ for some $n$ are called \emph{$p$-groups}.

    \begin{theorem}
        Let $G$ be a $p$-group. Then the centre of $G$ is non-trivial.
        \label{centre-nontrivial}
    \end{theorem}
    
    \begin{proof}
        Let $G$ act on itself by conjugation. The centre $Z(G)$ is the set of all size one orbits. Since the orbits partition $G$, we have
        \begin{align*}
            |G| = p^n = |Z(G)| + \sum \text{sizes of conjugacy classes of size} >1
        \end{align*}
        We know conjugacy classes have sizes dividing $p^n$. Hence, if a conjugacy class has size larger than 1, it is a multiple of $p$. Then $Z(G)$ is a difference of multiples of $p$ and the result follows
    \end{proof}
    
    \begin{theorem}
        A group of size $p^2$ is abelian.
    \end{theorem}
    
    \begin{lemma}
        If $G/Z(G)$ is cyclic, then $G$ is abelian.
        \label{cyclic-abelian}
    \end{lemma}
    
    \begin{proof}[Proof of lemma]
        We have a coset $xZ(G)$ in $G/Z(G)$ that generates $G/Z(G)$, so every coset has the form $x^mZ(G)$. This means every $G$ can be written as $g=x^mz$ for some $m$ and $z \in Z(G)$. From this the result immediately follows.
    \end{proof}
    
    \begin{proof}[Proof of theorem]
        By \cref{centre-nontrivial}, $Z(G)$ is either $G$ or has size $p$. In the former case we are done -- in the latter case, $G/Z(G) \cong C_p$, which is cyclic, so, by \cref{cyclic-abelian}, $G$ is abelian.
    \end{proof}
    
    \begin{theorem}
        Let $G$ be a $p$-group of size $p^n$. Then $G$ has a subgroup of size $k$ for all $0 \leq k \leq n$.
    \end{theorem}
    
    \begin{proof}
        We induct on $n$.
        \emph{Base case}. $n=1$, the result is clear.s
        \emph{Inductive step}. Suppose $n > 1$. If $k=0$, take $\{e\}$. Otherwise, note $Z(G)$ is non-trivial. Let $x \in Z(G)$ with $x \neq e$, and its order is a power of $p$ -- if we replace $x$ with some power of $x$, say $x'$, we can produce an element of order $p$. Then $\gen x'$ has order $p$ and is normal.

        Now consider $G/\gen x'$. This is a group of size $p^{n-1}$, and so it contains a subgroup, say $L$. of size $p^{k-1}$ by induction. By the correspondence of subgroups, we find $K \leq G$ containing $\gen x'$ such that $K/\gen x' \cong L$. But then $|K| = p^k$, as required.
    \end{proof}
    
    \subsection{Finite abelian groups}

    \begin{theorem}[Classification of finite abelian groups]
        \label{classification-finite-abelian}
        Let $G$ be a finite abelian group. There exist positive integers $d_1, \dots, d_r$ such that
        \begin{align*}
            G \cong C_{d_1} \times \dots \times C_{d_r}
        \end{align*}
        Furthermore, we can choose the $d_i$ such that $d_{i+1} \mid d_i$, in which case the expression is unique.
    \end{theorem}
    
    \begin{example}
        Consider, for example, abelian groups of order 8 -- they are exactly $C_8$, $C_4 \times C_2$, and $C_2 \times C_2 \times C_2$.
    \end{example}
    
    \begin{lemma}[Chinese remainder theorem]
        \label{crt}
        If $n$ and $m$ are coprime, then $C_n \times C_m \cong C_{nm}$.
    \end{lemma}
    
    \begin{proof}
        It suffices to find an order $mn$ element of $C_n \times C_m$. If $g \in C_n$ has order $n$ and $h \in C_m$ has order $m$, then consider $(g, h) \in C_n \times C_m$. Suppose it has order $k$, then $g^k = h^k = e$, so that $n \mid k$ and $m \mid k$. But then $nm \mid k$, and $k \mid nm$ by Lagrange, so $k=mn$ as required.
    \end{proof}

    \begin{corollary}
        Let $G$ be finite and abelian, then we have
        \begin{align*}
            G \cong C_{l_1} \times \dots \times C_{l_n}
        \end{align*}
        where each $l_i$ is a prime power.        
    \end{corollary}
    
    \begin{proof}
        The result follows from \cref{classification-finite-abelian} and \cref{crt}.
    \end{proof}
    
    \subsection{Sylow theorems}

    Let $G$ be a finite group of size $p^am$ where $p \nmid m$ and $p$ is some prime. A \emph{Sylow subgroup} of $G$ is a subgroup of order $p^a$.

    \begin{theorem}[Sylow theorems]
        Let $G$ be a finite group and $p$ a prime.
        \begin{enumerate}
            \item The set $\Syl_p(G) = \{P \leq G: P \text{ is Sylow-}p\}$ is non-empty;
            \item any $H$, $H' \in \Syl_p(G)$ are conjugate;
            \item if $n_p = |\Syl_p(G)|$, then $n_p \equiv 1 \pmod p$, and $n_p \mid |G|$ (so $n_p \mid m$).
        \end{enumerate}
    \end{theorem}
    
    \begin{corollary}
        $\Syl_p(G) = \{P\}$ if and only if $P$ is normal.
    \end{corollary}
    
    \begin{proof}
        For any $g \in G$, $gPg^{-1}$ is isomorphic as a group to $P$, so it is also a Sylow $p$-subgroup.
    \end{proof}
    
    \begin{corollary}
        \label{non-abelian-simple}
        Let $G$ be non-abelian and simple and suppose $p \mid |G|$. Then $|G|$ divides $\frac{n_p!}{2}$ and $n_p \geq 5$.
    \end{corollary}
    
    \begin{proof}
        The conjugation action of $G$ on $\Syl_p(G)$ gives a homomorphism
        \begin{align*}
            \phi: G \to \Sym(\Syl_p(G)) \cong S_{n_p}
        \end{align*}
        By simplicity of $G$, $\ker \phi = \{e\}$ or $G$. If $\ker \phi = G$, then $gPg^{-1} = P$ for all $g \in G$, so $P$ is normal. Since $P \neq \{e\}$, $P = G$. But $Z(P) \norm P$ non-trivially, which is a contradiction. Hence $\phi$ is injective, so $G$ isomorphic to a subgroup of $S_{n_p}$. Consider now
        \[\begin{tikzcd}
            G & {S_{n_p}} & {\{\pm1\}}
            \arrow["\phi", from=1-1, to=1-2]
            \arrow["\sign", from=1-2, to=1-3]
        \end{tikzcd}\]
        If the composition is surjective, then $\ker(\sign \circ \phi)$ is an index 2 subgroup of $G$. But $G$ is simple, so this is impossible, (since $G \neq C_2$). In other words, $\im \phi \subseteq A_{n_p}$, and the result follows by Lagrange. Finally, all non-abelian subgroups of $A_2$, $A_3$ and $A_4$ are not simple.
    \end{proof}
    
    \begin{example*}
        Suppose $G$ has size $11 \times 12 = 132$. If $G$ is simple, then none of the $n_p$'s are equal to 1. Furthermore, there are 12 Sylow-11 subgroups, since $n_11 \equiv 1 \pmod 11$ and $n_11 \mid 12$.

        Similarly, $n_3 \equiv 1 \pmod 3$, and $n_3 \mid 44$, so $n_3 = 4$ or $22$ -- by \cref{non-abelian-simple}, $n_3 = 22$. Any two Sylow 11-subgroups intersect only at $\{e\}$, so we already have 120 elements. We also have at least 44 elements of order 3, which is impossible, so no such simple $G$ exists.
    \end{example*}
    
    \begin{proof}[Proof of Sylow part 1]
        We will try and produce a Sylow $p$-subgroup as a stabiliser of an element in an appropriate action.

        Let $|G| = n = p^am$, of course $p \nmid m$. Let $\Omega = \{X \subseteq G : |X| = p^a\}$. Let $G$ act on $\Omega$ by $g \cdot \{g_1, \dots, g_{p^a}\} = \{gg_1, \dots, gg_{p^a}\}$.

        \emph{Claim 1.} We have $|\Omega| = \binom n {p^a} = m \not\equiv 0\pmod p$.

        \emph{Proof.} Expand $(1+x)^n$ mod $p$.

        \emph{Claim 2.} Suppose $U \in \Omega$ and $H \leq G$ stabilises $U$, then $|H| \mid |U|$.

        \emph{Proof.} We have $hU = U$ for all $h \in H$. In other words, for all $u \in U$, the coset $Hu$ is contained in $U$. Furthermore, every $u \in U$ lies in some coset of $H$, and the cosets are disjoint, so the cosets partition $U$. We get $|H| \mid |U|$.

        We now know $|\Omega| \neq 0 \pmod p$, and $|\Omega| = |O_1| + \dots + |O_r|$ where the $O_i$ are the orbits of the action. Since $p \nmid |\Omega|$, there exists an orbit, say $\Theta$, whose size is coprime to $p$.

        Let $T \in \Theta$. By the orbit-stabiliser theorem, $|G| = |\Theta|\cdot|\Stab(T)|$, so $p^am = |\Theta||\Stab(T)|$. By claim 2, $|\Stab(T)|$ divides $p^a$, but there are no factors of $p$ in $\Theta$, so we are done.
    \end{proof}

    \subsection{Basic definitions}

    Let $R$ be a ring. We have
    \begin{align*}
        0 + 0 &= 0 \\
        \implies r \cdot 0 + r \cdot 0 &= r \cdot 0 \\
        \implies r \cdot 0 &= 0
    \end{align*}
    Furthermore, if $1 = 0$, we have $r = 1 \cdot r = 0 \cdot r = 0$, so we have the \emph{zero ring}.

    \begin{definition}[Ring direct product]
        Let $R$, $S$ be rings. The \emph{direct product} $R \times S$ is a ring with operations and identities defined pointwise.
    \end{definition}
    
    \begin{definition}[Polynomials]
        Let $R$ be a ring. A \emph{polynomial} in $X$ with coefficients in $R$ is an expression of the form
        \begin{align*}
            f(X) = a_0 + a_1X + \dots + a_nX^n
        \end{align*}
        where each $a_i \in \RR$, and the $X^i$ are formal symbols.  As convention, we will identify $f(X)$ with $f(X) + 0 \cdot X^{n+1}$ -- then, the largest $n$ with $a_n \neq 0$ is the \emph{degree} of $f(x)$ (by convention, we say the degree of $0$ is $-\infty$). Finally, a polynomial is \emph{monic} if it ha s degree $n$ and $a_n=1$.
    \end{definition}
    
    \begin{definition}[Polynomial rings]
        Let $R$ be a ring. The \emph{polynomial ring} $R[X]$ is the set of all polynomials with coefficients in $R$, with obvious operations and identities.
    \end{definition}
    
    \begin{remark*}
        Note that, for every $f \in R[X]$, we get a function $R \to R$ which takes $r$ to $f(r)$. But this is not the same as the polynomial itself.
    \end{remark*}
    
    \begin{definition}[Rings of formal power series]
        The ring of \emph{formal power series} over $R$ in the variable $X$ is
        \begin{align*}
            R[[X]] = \left\{\sum_{i=0}^\infty r_iX^i : r_i \in R\right\}
        \end{align*}
        with obvious operations and identities.
    \end{definition}
    
    \begin{example}
        Consider the element $1-x \in R[X]$. Is it a unit?

        Suppose we have $(1-x)g(x) = 1$, where $g(x) = a_0 + \dots + a_nx^n$, with $a_n \neq 0$. Then
        \begin{align*}
            (1-x)g(x) = a_0 + (a_1-a_0)x + \dots + (a_n-a_{n-1})x^n - a_nx^{n+1}
        \end{align*}
        and then, by comparing the $x^{n+1}$ coefficient, we can see this is not 1.
        
        On the other hand, if we view $1-x \in R[[X]]$, this has an inverse:
        \begin{align*}
            (1-x)(1+x+x^2+\cdots)=1
        \end{align*}
        In fact, every power series has an inverse as long as $r_0$ is invertible.        
    \end{example}
    
    \begin{definition}[Rings of Laurent series]
        Write $R[X, X^{-1}]$ for the ring of \emph{Laurent series} with coefficients in $R$, that is
        \begin{align*}
            f = \sum_{i \in \ZZ} a_ix^i
        \end{align*}
        with $a_i \in \RR$ and all but finitely many $a_i = 0$.        
    \end{definition}
    
    \subsection{Homomorphisms, ideals and quotients}

    \begin{definition}[Ring homomorphisms]
        Let $R$ and $S$ be rings. A function $f: R \to S$ is a \emph{ring homomorphism} if
        \begin{itemize}
            \item $f(r_1 + r_2) = f(r_1) + f(r_2)$;
            \item $\implies f(0) = 0$;
            \item $f(r_1r_2) = f(r_1)f(r_2)$;
            \item $\centernot\implies f(1) = 1$.
        \end{itemize}
        The kernel of $f$ is $\ker f = \{r \in R : f(r) = 0\}$, and the image of $f$ is $\im f = \{s \in S: s = f(r) \text{ for some }r \in R\}$.
    \end{definition}
    
    \begin{definition}[Ring isomorphisms]
        If $f: R \to S$ is a bijective ring homomorphism, then it is called an isomorphism. Furthermore, its inverse is also a ring homomorphism.
    \end{definition}
    
    \begin{definition}
        A subset $I \subseteq R$ is called an \emph{ideal}, written $I \norm R$, if
        \begin{itemize}
            \item $I$ is a subgroup additively;
            \item if $a \in I$ and $b \in R$, then $ab \in I$.
        \end{itemize}
        Usually ideals are not subrings, since they can only contain 1 if $I = R$.        
    \end{definition}
    
    \begin{lemma}
        If $\phi: R \to S$ is a homomorphism, then $\ker \phi \norm R$.
    \end{lemma}
    
    \begin{proof}
        Since $\phi$ is a group homomorphism, we know $\ker \phi$ is a subgroup.

        Furthermore, if $a \in \ker \phi$ and $b \in R$, then $\phi(ab) = \phi(a)\phi(b) = 0 \cdot \phi(b) = 0$.
    \end{proof}
    
    \begin{remark*}
        If $a \in R$ is a unit, and $I \norm R$ is an ideal, then $u \in I \implies I = R$ -- since $1 = uv$, $1 \in I$, and so $I = R$.
    \end{remark*}
    
    \begin{example}
        Observe that $n\ZZ$ is an ideal of $\ZZ$ for any $n$. In fact, every ideal of $\ZZ$ has this form. Indeed, if $I \neq \{0\}$, let $n \in \ZZ$ be the minimal positive element of $I$. By the Euclidean algorithm, we can write $m \in I$ as $m = qn + r$, with $0 \leq r < n$, and $qn \in I$, so $r \in I$ -- this contradicts minimality of $n$.
    \end{example}
    
    \begin{definition}
        Let $A \subseteq R$ be a subset. The \emph{ideal generated by $A$}
        is
        \begin{align*}
            \gen A = \left\{\sum_{a \in A}r_aa : r_a \in \RR, r_a = 0 \text{ for all but finitely many }a\right\}
        \end{align*}
    \end{definition}
    
    \begin{definition}
        An ideal is called \emph{principal} if there exists $r \in R$ such that $I = \gen r$.
    \end{definition}
    
    \begin{example}
        Consider the subset of $R[x]$ given by
        \begin{align*}
            \{f(x) \in R(x): f(0) = 0\}
        \end{align*}
        which is generated by $x \in R[x]$ -- hence, this is a principal ideal.        
    \end{example}
    
    \begin{definition}[Quotients]
        Let $I \norm R$ be an ideal. The \emph{quotient ring} $R/I$ is the set of cosets of $I$ in $R$ with $0$ and $1$ given by $0_R + I$ and $1_R + I$, with the rules
        \begin{align*}
            (r_1 + I) + (r_2 + I) &= (r_1 + r_2) + I \\
            (r_1 + I) \cdot (r_2 + I) &= r_1r_2 + I
        \end{align*}
    \end{definition}
    
    \begin{proposition}
        The quotient ring is a ring. Furthermore, the function
        \begin{align*}
            f: R &\to R/I \\
            r &\mapsto r + I
        \end{align*}
        is a ring homomorphism.
    \end{proposition}
    
    \begin{proof}
        This is obviously a group homomorphism. Furthermore, multiplication is well-defined: suppose $r_1 + I = r_1' + I$, $r_2 + I = r_2' + I$. We then have $r_i - r_i' \in I$, say $r_i - r_i' = a_i$. We then have
        \begin{align*}
            r_1'r_2' = (r_1+a_1)(r_2+a_2) = r_1r_2 + \underbrace{r_1a_2 + r_2a_1 + a_1a_2}_{\in I}
        \end{align*}
        We conclude that the coset in $I$ of $r_1'r_2'$ is the same as that of $r_1r_2$.

        To check the other ring properties is straightforward.        
    \end{proof}
    
    \begin{example}
        \begin{itemize}
            \item We have $n\ZZ \norm \ZZ$. The quotient ring $\ZZ/n\ZZ$ consists of cosets of $n\ZZ$, i.e. $0 + n\ZZ$, $1 + n\ZZ, \dots, (n-1) + n\ZZ$. The ring structure is modular arithmetic.
            \item Take $\gen x \norm \CC[x]$. The elements of the quotient $\CC[x]/\gen x$ can be represented by elements of the form $(a_0 + a_1x + \dots + a_nx^n) + \gen x$ -- but all the terms $a_1x, \dots, a_nx^n$ lie in $\gen x$, so every element is uniquely represented by $a_0$. Hence $\CC[X]/\gen x \cong \CC$.
        \end{itemize}
    \end{example}
    
    \begin{proposition}[Euclidean algorithm for polynomial rings over fields]
        Let $K$ be a field, and take $f$, $g \in K[x]$. There exist polynomials $r$, $q \in K[x]$ such that
        \begin{align*}
            f = gq + r
        \end{align*}
        and $\deg r < \deg g$.        
    \end{proposition}
    
    \begin{proof}
        Let $n = \deg f$ and $m = \deg g$, so
        \begin{align*}
            f = \sum_{i=0}^n a_ix^i \qquad g = \sum_{i=0}^m b_ix^i
        \end{align*}
        with $a_i, b_i \in K$, $a_n \neq 0$. If $n < m$, set $q = 0$ and $r = f$. On the other hand, if $n \geq m$, we proceed by induction on $n$. In particular, let $f_1 = f - a_n b_m^{-1} x^{n-m}$ (note we are using the fact that $K$ is a field). Now, $\deg f_1 < n$. If $n = m$, we can write
        \begin{align*}
            f = (a_nb_m^{-1}x^{n-m})g + f_1
        \end{align*}
        Otherwise, if $n > m$< we can write by induction
        \begin{align*}
            f_1 = gq_1 + r_1
        \end{align*}
        with $\deg r_1 < m$. Then
        \begin{align*}
            f = (a_nbm^{-1}x^{n-m}+q)g + r_1
        \end{align*}
    \end{proof}
    
    \begin{corollary*}
        All ideals in $K[x]$ for $K$ a field are principal.
    \end{corollary*}
    
    \begin{proof}
        We have proven that Euclidean algorithm $\implies$ all ideals are principal, since we did it for $\ZZ$. The result follows.
    \end{proof}
    
    \begin{theorem}[First isomorphism theorem for rings]
        Let $\phi: R \to S$ be a ring homomorphism. Then, the map
        \begin{align*}
            f: R/\ker \phi &\to \im \phi \\
            r + \ker \phi &\mapsto \phi(r)
        \end{align*}
        is a well-defined ring isomorphism.        
    \end{theorem}
    
    \begin{proof}
        Well-definedness, bijectivity and additivity all follow from the group isomorphism theorem. It remains to check multiplicativity. Observe
        \begin{align*}
            f((r + \ker \phi)(t + \ker \phi)) &= f(rt + \ker \phi) \\
            &= \phi(rt) \\
            &= \phi(r) \phi(t) \\
            &= f(r + \ker \phi)f(t + \ker \phi)
        \end{align*}
    \end{proof}
    
    \begin{example}
        Consider the homomorphism
        \begin{align*}
            \phi: \RR[x] &\to \CC \\
            f(x) &\mapsto f(i)
        \end{align*}
        The map is clearly surjective, and the kernel is exactly those polynomials $f$ satisfying $f(i) = 0$. However, since coefficients of $f$ are real, $-i$ is also a root. In conclusion, $x^2 + 1 \mid f$, and we conclude $\ker \phi = \gen{x^2+1}$. Hence, by the first isomorphism theorem, $\CC \cong \RR[x]/\gen{x^2+1}$.        
    \end{example}
    
    \begin{proposition}[Second isomorphism theorem for rings]
        Let $R \leq S$ and $J \norm S$. Then $J \cap R$ is an ideal in $R$, and
        \begin{align*}
            (R+J)/J = \{r + J: r \in R\} \leq S/J
        \end{align*}
        is a subring. Furthermore,
        \begin{align*}
            R/(R \cap J) \cong (R+J)/J
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Define a function
        \begin{align*}
            \phi: R &\to S/J \\
            r &\mapsto r + J
        \end{align*}
        so $\ker \phi = \{r \in R: r + J = 0\} = R \cap J$, so this is an ideal by the first isomorphism theorem. Also, the image is $\im \phi = \{r + J: r \in R\} = (R + J)/J$. The result follows from the first isomorphism theorem.        
    \end{proof}

    \begin{proposition}
        Let $R$ be a ring. Then $R$ is a field if and only if the only ideals of $R$ are $\{0\}$ and $R$.
    \end{proposition}
    
    \begin{proof}
        \begin{itemize}
            \item[$\implies$] If $R$ is a field and $I \norm R$, say $I \neq \{0\}$, then $I$ contains a unit, say $u \in I$. But then $1 \in I$ since $1 = uv$. We conclude $I = R$.
            \item[$\impliedby$] If the only ideals are $\{0\}$ and $R$, take $0 \neq r \in R$. Then $\gen r$ is $R$. So $1 \in R \implies 1 = rs$. So $r$ is a unit and hence $R$ is a field. 
        \end{itemize}
    \end{proof}
    
    \begin{definition}[Maximal ideals]
        An ideal $R \neq I \norm R$ is called \emph{maximal} if, for any $J \norm R$, $I \subseteq J \subseteq R$ implies $I = J$ or $J = R$.
    \end{definition}
    
    \begin{proposition}
        \label{maximal-field}
        An ideal $I \norm R$ is maximal if and only if $R/I$ is a field.
    \end{proposition}
    
    \begin{proof}
        $R/I$ is a field if and only if its only ideals are $\{0\}$ and $R/I$ itself. Now apply the ideal correspondence result.
    \end{proof}
    
    \begin{definition}[Prime ideals]
        An ideal $R \neq I \norm R$ is \emph{prime} if, whenever $ab \in I$, either $a$ or $b \in I$.
    \end{definition}
    
    \begin{example}
        An ideal $n\ZZ \norm \ZZ$ is a prime ideal if and only if $n$ is 0 or a prime.
    \end{example}
    
    \begin{proposition}
        \label{prime-integral}
        An ideal $I \norm R$ is prime if and only if $R/I$ is an integral domain.
    \end{proposition}
    
    \begin{proof}
        \begin{itemize}
            \item[$\implies$] Suppose $I$ is prime and let $a + I$ and $b + I \in R + I$ be cosets. Suppose $(a + I)(b + I) = ab + I = 0 + I$ if and only if $ab \in I$. But, since $I$ is prime, either $a$ or $b \in I$, so either $a + I$ or $b + I = 0 + I$.
            \item[$\impliedby$] If $R/I$ is an integral domain, then $ab \in I$ satisfies $ab + I = (a + I)(b + I) = 0 + I$ in $R/I$, so either $a \in I$ or $b \in I$, so $I$ is prime.
        \end{itemize}
    \end{proof}
    
    \begin{corollary}
        Every maximal ideal in a ring is prime.
    \end{corollary}
    
    \begin{proof}
        Since every field is an integral domain, the result follows from \cref{maximal-field} and \cref{prime-integral}.
    \end{proof}
    
    \begin{remark*}
        The converse clearly does not hold.
    \end{remark*}
    
    \begin{proposition}
        The characteristic of an integral domain is either zero or a prime number.
    \end{proposition}
    
    \begin{proof}
        Consider the unique homomorphism $\ZZ \to R$. Letting the characteristic be $n$, we have that $\gen n \norm \ZZ$ is the kernel, so $\ZZ/n\ZZ \leq R$ by the first isomorphism theorem. But then $\ZZ/n\ZZ$ is an integral domain, and the result follows.
    \end{proof}
    
    \subsection{Factorisation in integral domains}

    Throughout this section, $R$ is an integral domain.

    \begin{definition}[Division]
        Let $a$, $b \in R$. We say $a$ \emph{divides} $b$, written $a \mid b$, if there exists $c \in R$ such that $ac = b$. Equivalently, we can say $\gen b \subseteq \gen a$.
    \end{definition}
    
    \begin{definition}[Associates]
        We say $a$, $b \in R$ are \emph{associates} if $a = bc$ for some unit $c \in R$. Equivalently, we can say $a \mid b$ and $b \mid a$, or $\gen a = \gen b$.
    \end{definition}
    
    \begin{definition}[Irreducible elements]
        An element $r \in R$ is \emph{irreducible} if $r \neq 0$, $r$ is not a unit, and, if $r = xy$, either $x$ or $y$ is a unit.
    \end{definition}
    
    \begin{definition}[Prime elements]
        An element $r \in R$ is \emph{prime} if $r \neq 0$, $r$ is not a unit, and, if $r \mid xy$, either $r \mid x$ or $r \mid y$.
    \end{definition}
    
    \begin{proposition}
        Let $r \in R$. Then, if $r \neq 0$, it is a prime element if and only if $\gen r$ is a prime ideal.
    \end{proposition}
    
    \begin{proof}
        \begin{itemize}
            \item[$\implies$] Let $r$ be a prime element, and suppose $ab \in \gen r$. Then $r \mid ab$, so $r \mid a$ or $r \mid b$, hence $a \in \gen r$ or $b \in \gen r$ as required.
            \item[$\impliedby$] Let $\gen r$ be a prime ideal. By definition, $r$ is not a unit. Say $r \mid ab$, so $ab \in \gen r$. Since $\gen r$ is prime, either $a \in \gen r$ or $b \in gen r$, so either $r \mid a$ or $r \mid b$. 
        \end{itemize}
    \end{proof}
    
    \begin{proposition}
        Let $r \in R$ be prime. Then $r$ is irreducible.
    \end{proposition}
    
    \begin{proof}
        Let $r \in R$ be prime, and suppose $r=ab$. Since $r \mid r$, $r \mid ab$, so, since $r$ is prime, $r \mid a$ or $r \mid b$ -- wlog suppose $r \mid a$. Hence $a = rc$ with $c \in R$, so $r = ab = rcb$. Since we're in an integral domain, we can cancel to get $bc = 1$, so $b$ is a unit as required.
    \end{proof}
    
    \begin{remark*}
        The converse is false -- for example, $R = \ZZ[\sqrt{-5}]$.
    \end{remark*}
