    \section{Preliminaries}

    \subsection{Vector notation}

    We will in general work in $\RR^3$, but sometimes $\RR^2$. In fact, most of our results will generalise to $\RR^n$.

    We will write column vectors to save space when we mean row vectors, and we will use dot notation for differentiation (even though $t$ isn't really time).

    If $\vec a$ and $\vec b$ are vectors, then the following hold:
    \begin{align}
        \dot{\vec a(t)} = \frac{d \vec a}{dt} &= \frac{d}{dt}(a_1, a_2, a_3) = (\dot a_1, \dot a_2, \dot a_3) \\
        \frac{d}{dt}(\vec a \cdot \vec b) &= \dot {\vec a} \cdot \vec b + \vec a \cdot \dot {\vec b} \\
        \frac{d}{dt}(\vec a \times \vec b) &= \dot {\vec a} \times \vec b + \vec a \times \dot{\vec b}
    \end{align}
    
    and these results can be shown using suffix notation. Integration is also componentwise. We have therefore
    \begin{align*}
        \int_{t_1}^{t_2} \dot {\vec a}(t) \; dt = \vec a(t_2) - \vec a(t_1)
    \end{align*}
    
    The modulus of a vector will be denoted by the same letter as the vector, but not bolded/underlined. So, for example, $r=|\vec r|$ (but $x$ means $x_1$, not $|\vec x|$).

    Notationally, we might write $\partial_i$ for $\pd {}{x_i}$.

    \subsection{The chain rule}

    Recall that, if $f$ is a function of $\vec x$ which is itself a function of $t$, then
    \begin{proposition}
        \begin{align*}
            \frac{d}{dt} f(\vec x(t)) &= \pd{f}{x} \frac{dx}{dt} + \pd{f}{y} \frac{dy}{dt} + \pd{f}{z} \frac{dz}{dt} \\
            &= \pd {f}{x_i} \frac{dx_i}{dt}
        \end{align*} \label{1.1}
    \end{proposition}
    
    
    If instead $\vec x = \vec x(\vec u)$, where $\vec u = (u_1, u_2, u_3) = (u, v, w)$, then
    \begin{align*}
        \pd{f}{u} &= \pd{f}{x} \pd{x}{u} + \pd{f}{y} \pd{y}{u} + \pd{f}{z} \pd{z}{u}
    \end{align*}
    and similarly for $\pd{f}{v}$ and $\pd{f}{w}$, i.e.
    \begin{align*}
        \pd {f}{u_i} = \pd {f}{x_i} \pd {x_i}{u_i}
    \end{align*}
    
    \subsection{Taylor's theorem}

    Recall that for a small vector displacement $\delta \vec x$,
    \begin{align*}
        f(\vec x + \delta \vec x) &= f(x + \delta x, y + \delta y, z + \delta z) \\
        &= f(x, y, z) + \pd f x \delta x + \pd f y \delta y + \pd f z \delta z + \frac{1}{2!} \left(\delta x \pd{}{x} + \delta y \pd{}{y} + \delta z \pd {}{z}\right)^2f + \cdots
    \end{align*}
    or, in suffix notation,
    \begin{proposition}
        \label{1.2} \[
            f(\vec x + \delta \vec x) = f(\vec x) + \pd{f}{x_i} \delta x_i + O(|\delta \vec x|^2)
        \]
    \end{proposition}
    
    
    
    
    \subsection{Infinitesimals}
    
    In one dimension, we have from Taylor's theorem
    \begin{align*}
        \delta f = f(x + \delta x) - f(x) = f'(x) \delta x + O(\delta x^2)
    \end{align*}
    Taking $\delta x \to 0$, $\delta x$ and $\delta f$ become $dx$ and $df$, and we get the exact equality
    \begin{align*}
        df = f'(x) \D x
    \end{align*}
    If we want, we can even think of infinitesimals as numbers such that their squares are zero.

    Note that infinitesimals have theiir own product and quotient rules, namely
    \begin{align*}
        d(fg) &= g \D f + f \D g \\
        d\left(\frac f g\right) &= \frac{g \D f - f \D g}{g^2}
    \end{align*}

    We also get
    \begin{proposition}
        \label{1.3} \[
            df = \pd{f}{x_i} \; dx_i
        \]
    \end{proposition}
    
    
    The modulus of an infinitesimal vector can be calculated using, for example
    \begin{align*}
        |d \vec x| = |dx| \sqrt{1 + \left(\frac{dy}{dx}\right)^2 + \left(\frac{dy}{dx}\right)^2}
    \end{align*}
    
    If $dx$ and $dy$ are independent infinitesimals then $dx \D y$ is an infinitesimal area. But if $y$ depends on $x$ in some way then $dx \D y$ vanishes, since it's proportional to $dx^2$.
    
    \section{Vector differential operators}
    
    \begin{remark*}
        What is an operator? It is a map between vector spaces of functions.
    \end{remark*}

    \subsection{Del}

    The fundamental differential operator is \emph{del}:
    \begin{align*}
        \grad = \begin{pmatrix}
         \pd{}{x} \\
         \pd{}{y} \\
         \pd{}{z} \\
        \end{pmatrix}
    \end{align*}
    
    We can also write
    \begin{align*}
        \grad = \vec e_1 \pd{}{x_1} + \vec e_2 \pd{}{x_2} + \vec e_3 \pd{}{x_3} = \vec e_i \pd{}{x_i}
    \end{align*}
    where $\{\vec e_1, \vec e_2, \vec e_3\}$ is the canonical basis for $\RR^3$.
    
    \subsection{Gradient}

    When del is applied to a scalar field $f(\vec x)$, it is known as the \emph{gradient} operator:
    \begin{align*}
        \Aboxed{\grad f = \begin{pmatrix}
         \pd{f}{x} \\
         \pd{f}{y} \\
         \pd{f}{z} \\
        \end{pmatrix}}
    \end{align*}
    This can also be denoted $\mathrm{grad} f$.
    
    In suffix notation, we have
    \begin{align*}
        (\grad f)_i &= \pd{f}{x_i} \\
        \iff \grad f &= \pd{f}{x_i} \vec e_i
    \end{align*}
    
    \begin{example*}
        \begin{itemize}
            \item If $f(\vec x) = |\vec x|^2 = x^2 + y^2 + z^2$, then $\grad f = (2x, 2y, 2z) = 2\vec x$. Alternatively, $\pd{}{x_i}(x_jx_j) = 2x_i \delta_{ij} = 2x_i$ (the $\delta_{ij}$ comes from $\pd{x_j}{x_i}$).
            \item $\grad|\vec x - \vec x_0|^2 = 2(\vec x - \vec x_0)$. This is a direct consequence of the above and a very simple application of the chain rule.
            \item $\grad(\vec a \cdot \vec x) = \grad(a_1x+a_2y+a_3z) = a_1a_2a_3 = \vec a$. Alternatively, $\pd{}{x_i}(a_jx_j) = a_i\delta_{ij} = a_i$.
        \end{itemize}
    \end{example*}
    
    $\grad f$ in 3D is analogous to $f'$ in 1D, and we can evaluate it at points.

    \begin{example*}
        If $f(\vec x) = x + y^2 + z^4$, then $\grad f(\vec a) = \grad f(\vec b)$ if and only if $\vec a - \vec b$ is parallel to the $x$-axis. Observe $\grad f(\vec x) = (1, 2y, 4x^3)$, so
        \begin{align*}
            (1, 2a_2, 4a_3^3) &= (1, 2b_2, 4b_3^3) \\
            \iff (a_2, a_3) &= (b_2, b_3) \\
            \iff \vec a - \vec b &= (a_1 - b_1, 0, 0)
        \end{align*}
    \end{example*}
    
    We have some new opportunities for notation now -- for example, we can rewrite (1.1) as

    \begin{proposition}
        \label{2.1} \[
            \frac{d}{dt} f(\vec x(t)) = \dot{\vec x} \cdot \grad f
        \]
    \end{proposition}
    
    and (1.3) as

    \begin{proposition}
        \label{2.2} \[
            df = \grad f \cdot \D x
        \]
    \end{proposition}
    
    
    
    Equation (2.2) also gives a way of defining $\grad f$ that is \emph{coordinate-independent}, i.e. it doesn't require us to choose a particular basis at any point.
    
    
    \begin{example*}
        If $f(x) = |\vec x|^2$, then
        \begin{align*}
            df = f(\vec x + d\vec x) - f(\vec x) &= \vec x \cdot \vec x + 2 \vec x \cdot d \vec x + d \vec x \cdot d \vec x - |\vec x|^2 \\
            &= 2\vec x \cdot d \vec x
        \end{align*}
        so it must be the case that $\grad f = 2x$.        
    \end{example*}
    
    Finally, we may write $\grad_{\vec x}$, $\grad_{\vec y}$, etc., to denote differentiation w.r.t. a particular vector variable.
    
    \subsection{Divergence}

    \begin{definition*}
        The \emph{divergence} of a vector field is given by del dotted with the field:
        \begin{align*}
            \div \vec F = \begin{pmatrix}
             \pd{}{x} \\
             \pd{}{y} \\
             \pd{}{z} \\
            \end{pmatrix} \cdot \begin{pmatrix}
             F_1 \\
             F_2 \\
             F_3 \\
            \end{pmatrix} = \pd{F_1}{x} + \pd{F_2}{y} + \pd{F_3}{z} = \pd{F_i}{x_i}
        \end{align*}        
    \end{definition*}

    More formally, we can derive this expression by

    \begin{align*}
        \div \vec F &= \left(\vec e_i \pd{}{x_i}\right) \cdot (F_j \vec e_j) \\
        &= \vec e_i \cdot \pd{}{x_i}(F_j \vec e_j) \\
        &= \vec e_i \cdot \pd{F_j}{x_i} \vec e_j \\
        &= \pd{F_j}{x_i} \delta_{ij} = \pd{F_i}{x_i}
    \end{align*}
    
    What is the divergence? It measures the amount by which a field is `expanding' around $\vec x$. We will get a better idea later on when we derive a coordinate-independent definition.

    \begin{example*}
        \begin{itemize}
            \item Take $F(\vec x) = \vec x$. Then $\div F = \pd{x}{x} + \pd{y}{y} + \pd{z}{z} = 3$. Suffix notation also works! DIAGRAM
            \item Take $F(\vec x) = (-y, x, 0)$. Then $\div F = -\pd{y}{x} + \pd{x}{y} = 0$. DIAGRAM
            \item If $A$ is a 3 by 3 matrix, we might want to calculate
            \begin{align*}
                \div (A \vec x) = \pd{x}{x_i} (A_{ij}x_j) = A_{ij} \delta_{ij} = A_{ii} = \Tr A
            \end{align*}
        \end{itemize}
    \end{example*}
    
    \subsection{Curl}

    \begin{definition*}
        The \emph{curl} of a vector field is given by del crossed with the field:
        \begin{align*}
            \curl \vec F = \begin{pmatrix}
             \pd{}{x} \\
             \pd{}{y} \\
             \pd{}{z} \\
            \end{pmatrix} \times \begin{pmatrix}
             F_1 \\
             F_2 \\
             F_3 \\
            \end{pmatrix} = \begin{pmatrix}
             \pd{F_3}{y} - \pd{F_2}{z} \\
             \pd{F_1}{z} - \pd{F_3}{x} \\
             \pd{F_2}{x} - \pd{F_1}{y} \\
            \end{pmatrix}
        \end{align*}
        In suffix notation, we have
        \begin{align*}
            (\curl F)_i = \lv{ijk} \pd{}{x_j} F_k
        \end{align*}
        
        
    \end{definition*}
    
    The curl, notably, is only defined in 3D, since we only have a cross product in 3D, unlike gradient or divergence. It measures, in a certain sense, the \emph{rotation} of the field about a point $\vec x$.

    \begin{example*}
        \begin{itemize}
            \item With $F(\vec x) = \vec x$, we have
            \begin{align*}
                \curl F = \begin{pmatrix}
                 \pd{}{x} \\
                 \pd{}{y} \\
                 \pd{}{z} \\
                \end{pmatrix} \times \begin{pmatrix}
                 x \\
                 y \\
                 z \\
                \end{pmatrix} = \vec 0
            \end{align*}
            Alternatively, $(\curl F)_i = \lv{ijk} \pd{}{x_j} x_k = \lv{ijk}\delta_{jk} = 0$.
            \item With $F(\vec x) = (-y, x, 0)$, $\curl F = (0, 0, 2)$.
            \item If $A$ is a 3 by 3 symmetric matrix,
            \begin{align*}
                (\curl (A \vec x))_i = \lv{ijk} \pd{}{x_j} A_{kl}x_l = \lv{ijk} A_{kj}
            \end{align*}
            But $\lv{ijk}$ is antisymmetric while $A_{kj}$ is symmetric, so the terms cancel in pairs and we get $\curl (A \vec x) = \vec 0$.
        \end{itemize}
    \end{example*}
    
    \subsection{Directional derivatives}

    Given a \emph{vector} field $\V u$ we can define a scalar \emph{operator}
    \begin{align*}
        \V u \cdot \grad = u_1 \pd{}{x_1} + u_2 \pd{}{x_2} + u_3 \pd{}{x_3} = u_i \pd{}{x_i}
    \end{align*}
    We can then apply this operator to a \emph{scalar} field $f$:
    \begin{align*}
        (\V u \cdot \grad)f = u_1 \pd{f}{x_1} + u_2 \pd{f}{x_2} + u_3 \pd{f}{x_3} = u_i \pd{f}{x_i}
    \end{align*}
    But, happily, it turns out
    \begin{align*}
        (\V u \cdot \grad)f = \left(u_i \pd{}{x_i}\right)f = u_i \pd{f}{x_i} = \V u \cdot (\grad f)
    \end{align*}
    and so we can omit the brackets.
    

    However, we can also apply $\V u \cdot \grad$ to a \emph{vector} field $\V F$:
    \begin{align*}
        (\V u \cdot \grad) \V F = u_1 \pd{}{x_1} \V F + u_2 \pd{}{x_2} \V F + u_3 \pd{}{x_3} \V F
    \end{align*}
    so that
    \begin{align*}
        ((\V u \cdot \grad) \V F)_i = u_j \pd{F_i}{x_j}
    \end{align*}
    and this gives us a \emph{vector} field back. Disturbingly, the brackets here \emph{are} important, since $\V u \cdot (\grad \V F)$ is the gradient of a vector which isn't defined. However, we omit them anyway (since there's no ambiguity).
    
    \begin{example*}
        Take $\V u = (0, x, y)$ and $\V F = (yz, 0, x)$. Then $\V u \cdot \grad = x \pd{}{y} + y \pd{}{z}$, so $(\V u \cdot \grad) \V F = (xz + y^2, 0, 0)$.
    \end{example*}
    
    \begin{definition*}
        Given a unit vector $\hvec n$, the \emph{directional derivative} $D_{\hvec n}f$ of $f$ at $\V x$ is defined to be the one-dimensional derivative of $f$ along the line from $\V x$ parallel to $\hvec n$. That is, if we make $g(t) = f(\V x + t \hvec n)$, then
        \begin{align*}
            D_{\hvec n}f(\V x) = g'(0) = \hvec n \cdot \grad f(\V x)
        \end{align*}
        by \cref{2.1}. Then $D_{\hvec n} = \hvec n \cdot \grad$, and so the operator $\V u \cdot \grad$ is also called the directional derivative (even for non-unit vectors $\V u$).    
    \end{definition*}
    
    \subsection{Level surfaces and the gradient}

    \begin{definition*}
        In 3D, a \emph{level surface} of a function $f(\V x)$ is a two-dimensional surface satisfying $f(\V x) = c$ for some constant $c$. In 2D, these are just curves which we refer to as \emph{contours}.
    \end{definition*}
    
    \begin{proposition*}
        The gradient of $f$ is always normal to a level surface.
    \end{proposition*}
    
    \begin{proof}
        Starting at $\V x$ on the surface, consider a displacement $d\V x$ within the surface. Because $f = c$, $df = 0$ for this displacement, so, by \cref{2.2}, $\grad f \cdot d \V x = df = 0$ for all $d \V x$ lying in the surface. So $\grad f$ is perpendicular to all of these and hence is perpendicular to the surface.
    \end{proof}
    
    \begin{example*}
        Take $f(\V x) = |\V x|^2$, so $\grad f = 2 \V x$. Since this is simply radially outwards, we conclude the level surfaces are spheres centred at the origin.
    \end{example*}
    
    \begin{proposition*}
        $\grad f$ always points in the direction in which $f$ grows the most rapidly.
    \end{proposition*}
    
    \begin{proof}
        Consider $D_{\hvec n}f = \hvec n \cdot \grad f$ for an arbitrary unit vector $\hvec n$. Clearly $\hvec n \cdot \grad f$ is maximised when $\hvec n$ and $\grad f$ are parallel, and in particular this maximum is $|\grad f|$.
    \end{proof}
    
    
    \begin{remark*}
        We can think of level surfaces in two different ways, and endow them with different grads. Suppose we have a ground level $z = h(x, y)$ (imagine hills). Working in 2D, we have that $\grad h$ points in the direction that is steepest uphill (or the projection of that onto the $x$-$y$ plane). On the other hand, if we consider $f(x, y, z) = z - h(x, y)$ (which defines the exact same surface), then the level surface $f = 0$ has $\grad f$ pointing normal to the surface.
    \end{remark*}
    
    \subsection{Various vector differential identities}

    The following important identities can be proven using suffix notation.

    \begin{proposition}
        \[
            \grad(fg) = f \grad g + g \grad f
        \]
    \end{proposition}
    
    \begin{proposition}
        \[
            \grad(f(g(\V x))) = f'(g(\V x))\grad g
        \]
    \end{proposition}

    \begin{proposition}
        \label{2.5}\[
            \grad\cdot (f \V F) = f \grad \cdot \V F + \V F \cdot \grad f
        \]
    \end{proposition}

    \begin{proof}
        We have
        \begin{align*}
            \grad \cdot (f \V F) &= \pd{}{x_i}(fF_i) \\
            &= \pd{f}{x_i}F_i + f\pd{F_i}{x_i} \\
            &= \grad f \cdot \V F + f \grad \cdot \V F
        \end{align*}
    \end{proof}
    
    \begin{proposition}
        \[
            \grad \times (f \V F) = f \grad \times \V F + \grad f \times \V F
        \]
    \end{proposition}

    The above propositions should be memorised -- it is not necessary to memorise the following propositions, but one should be able to produce proofs of them.
    
    \begin{proposition}
        \[
            \grad(\V F \cdot \V G) = \V F \times (\grad \times \V G) + (\V F \cdot \grad) \V G + \V G \times (\grad \times \V F) + (\V G \cdot \grad)\V F
        \]
    \end{proposition}
    
    \begin{proof}
        Counterintuitively, let us start with
        \begin{align*}
            (\V F \times (\grad \times \V G))_i &= \lv{ijk}F_j\lv{klm}\pd{}{x_l}G_m \\
            &= (\delta_{il}\delta_{jm}-\delta_{im}\delta_{jl})F_j \pd{G_m}{x_l} \\
            &= F_j \pd{G_j}{x_i} - F_j \pd{G_i}{x_j}
        \end{align*}
        Adding to a similar expression for $(\V G \times (\grad \times \V F))_i$, we get
        \begin{align*}
            &\;F_j \pd{G_j}{x_i} - F_j \pd{G_i}{x_j} + G_j \pd{F_j}{x_i} - G_j \pd{F_i}{x_j} \\
            =&\; \underbrace{F_j \pd{G_j}{x_i} + G_j \pd{F_j}{x_i}}_{\pd{}{x_i}(F_jG_j)} - F_j \pd{G_i}{x_j} - G_j \pd{F_i}{x_j} \\
            =&\; (\grad(\V F \cdot \V G))_i - ((\V F \cdot \grad)\V G)_i - ((\V G \cdot \grad)\V F)_i
        \end{align*}
        and the result follows.
    \end{proof}

    \begin{proposition}
        \[
            \grad \cdot (\V F \times \V G) = -\V F \cdot (\grad \times \V G) + \V G \cdot (\grad \times \V F)
        \]
    \end{proposition}
    
    \begin{proposition}
        \[
            \grad \times (\V F \times \V G) = (\grad \cdot \V G) \V F - (\grad \cdot \V F)\V G + (\V G \cdot \grad)\V F - (\V F \cdot \grad)\V G
        \]
    \end{proposition}
    
    \begin{remark*}
        We must not use the formulae $\V a \cdot (\V b \times \V c) = (\V a \times \V b) \cdot \V c$ or $\V a \times (\V b \times \V c) = (\V a \cdot \V c) \V b - (\V a \cdot \V b)\V c$, since our derivations assumed certain commutativity and anti-commutativity properties that don't hold for operators.
    \end{remark*}
    
    We have one more very useful identity.

    \begin{proposition}
        \[
            \pd{r}{x_i} = \frac{x_i}{r}
        \]
    \end{proposition}
    
    \begin{proof}
        We have $r^2 = x_jx_j$, so $2r \pd{r}{x_i} = 2x_j \pd{x_j}{x_i} = 2x_j \delta_{ij} = 2x_i$.
    \end{proof}
    
    \begin{corollary}
        \label{2.11}\[
            \grad f(r) = f'(r) \V e_r
        \]
        where $\V e_r = \V r / r$.
    \end{corollary}
    
    \begin{proof}
        We have
        \begin{align*}
            \pd{}{x_i} f(r) = f'(r) \pd{r}{x_i} = f'(r) \frac{x_i}{r}
        \end{align*}
    \end{proof}
    
    \begin{example*}
        Using \cref{2.5} and \cref{2.11}, we have
        \begin{align*}
            \grad \cdot (r^3 \V r) &= r^3 \grad \cdot \V R + \V r \cdot \grad r^3 \\
            &= 3r^3 + \V r \cdot 3r^2 \V e_r \\
            &= 3r^3 + \V r \cdot 3 r \V r \\
            &= 6r^3
        \end{align*}
    \end{example*}
    
    To deal with the modulus of a vector field, we can do
    \begin{align*}
        \pd{}{x_i} |\V F| = \pd{}{x_i} \sqrt{F_jF_j} = \frac{1}{2|\V F|} \pd{}{x_i} (F_jF_j) = \frac{1}{|\V F|}F_j \pd{F_j}{x_i}
    \end{align*}
    
    \subsection{The Laplacian}

    \begin{definition*}
        The \emph{Laplacian} of a scalar field is
        \begin{align*}
            \grad^2 f &= \grad \cdot (\grad f) \\
            &= \pd{^2f}{x^2} + \pd{^2f}{y^2} + \pd{^2f}{z^2} \\
            &= \pd{^2f}{x_i\partial x_i}
        \end{align*}
    \end{definition*}
    It may also be denoted $\Delta f$.
    
    \begin{example*}
        \begin{itemize}
            \item If $f(\V x) = \alpha x^2 + \beta y^2 + \gamma z^2$, then
            \begin{align*}
                \left(\pd{^2}{x^2} + \pd{^2}{y^2} + \pd{^2}{z^2}\right)(\alpha x^2 + \beta y^2 + \gamma z^2) = 2(\alpha + \beta + \gamma)
            \end{align*}
            \item $\grad^2 |\V x|^2 = \grad^2(x^2+y^2+z^2) = 6$.
            \item If $f(\V x) = (\V a \cdot \V x)^2$,
            \begin{align*}
                \grad^2 f &= \grad \cdot (\grad f) \\
                &= \grad \cdot (2(\V a \cdot \V x)\grad(\V a \cdot \V x)) \\
                &= \grad (2(\V a \cdot \V x)\V a) \\
                &= 2(\V a \cdot \V x)\grad \cdot \V a + 2 \V a \cdot \grad(\V a \cdot \V x) \\
                &= 2 \V a \cdot \V a \\
                &= 2|\V a|^2
            \end{align*}
            Alternatively, and perhaps more easily,
            \begin{align*}
                \grad^2 f &= \grad^2(a_1^2x^2 + a_2^2y^2 + a_3^2z^2 + 2a_1a_2xy + 2a_1a_3xz + 2a_2a_3yz) \\
                &= 2(a_1^2 + a_2^2 + a_3^2) \\
                &= 2|\V a|^2
            \end{align*}
        \end{itemize}
    \end{example*}
    
    Functions that satisfy $\grad^2 f = 0$ (Laplace's equation) are called \emph{harmonic}.

    We often have spherically symmetric functions, i.e. those for which $f = f(r)$. In that case,

    \begin{align*}
        \grad^2 f(r) &= \grad \cdot (\grad f(r)) \\
        &= \grad \cdot (f'(r) \V e_r) \\
        &= \pd{}{x_i} \left(\frac{f'(r)}{r} x_i\right) \\
        &= \frac{rf''(r) - f'(r)}{r^2} \frac{x_i}{r} x_i + \frac{f'(r)}{r} \delta_{ii} \\
        &= f''(r) + \frac{2f'(r)}{r}
    \end{align*}
    This can also be written $\frac{1}{r^2} \frac{d}{dr} (r^2 f'(r))$ or $\frac{1}{r} \frac{d^2}{dr^2}(rf)$.

    \begin{remark*}
        As we can see, the only thing that depends on the dimension is $\delta_{ii}$. So, if we were in 2D, we would have $f''(r) + \frac{f'(r)}{r}$ instead.
    \end{remark*}
    For example, $\grad^2 |\V x|^2 = \grad^2f$ with $f(r) = r^2$, so $\grad^2 f = 2 + \frac{4r}{r} = 6$.

    For a vector field $\V F(\V x)$, we can't define $\grad^2 \V F$ as $\grad \cdot (\grad \V F)$, since $\grad \V F$ is undefined. Instead, we say
    \begin{align*}
        \grad^2 \V F = (\grad^2 F_1, \grad^2 F_2, \grad^2 F_3)
    \end{align*}
    An alternative definition that is coordinate-independent is
    \begin{align*}
        \grad^2 \V F = \grad(\grad \cdot \V F) - \grad \times (\grad \times \V F)
    \end{align*}
    but the first definition is preferable for calculations. To show they are equivalent, we have
    \begin{align*}
        (\curl(\curl \V F))_i &= \lv{ijk} \pd{}{x_j} \lv{klm} \pd{F_m}{x_l} \\
        &= \pd{^2F_j}{x_i \partial x_j} - \pd{^2F_i}{x_j \partial x_j} \\
        &= \pd{}{x_i} (\grad \cdot \V F) - \grad^2(F_i)
    \end{align*}
    
    \subsection{Irrotational and conservative vector fields}

    Consider a vector field $\V F$ defined in a domain $\mathscr D \subseteq \RR^3$.

    \begin{definition*}
        If $\V F$ satisfies $\curl \V F = \V 0$ in $\mathscr D$, it is \emph{irrotational} there.
    \end{definition*}

    \begin{definition*}
        If instead there exists a single-valued scalar field $f$ defined in $\mathscr D$ such that $\V F = \grad f$, then $\V F$ is said to be \emph{conservative}, and $f$ is known as a \emph{scalar potential} for $\V F$. Sometimes the sign of $f$ is changed, so $\V F = -\grad f$.
    \end{definition*}
    
    \begin{proposition}
        A vector field is conservative if and only if it is irrotational.
    \end{proposition}
    
    \begin{proof}
        \begin{itemize}
            \item[$\implies$] If $\V F = \grad f$, then $\curl \V F = \curl (\grad f)$. But
            \begin{align*}
                (\curl (\grad f))_i = \lv{ijk} \pd{}{x_i} \pd{f}{x_k}
            \end{align*}
            and $\lv{ijk}$ is antisymmetric while $\pd{^2f}{x_jx_k}$ is symmetric, so the terms cancel in pairs.
            \item[$\impliedby$] The converse holds only if $\mathscr D$ is \emph{simply connected}. The result will be shown in a later section.            
        \end{itemize}
    \end{proof}
    
    \begin{definition*}
        A \emph{simply connected domain} $\mathscr D$ is a connected domain in which any closed curve can be shrunk continuously to a point.
    \end{definition*}
    
    \begin{example*}
        \begin{itemize}
            \item $\RR^3$ is simply connected.
            \item $\RR^3 \setminus \{(x, 0, 0): x \in \RR\}$ is not simply connected - take a curve wrapped around the $x-axis$.
            \item $\RR^3 \setminus \{(x, 0, 0): x \geq 0\}$ is simply connected.
            \item $\RR^3 \setminus \{\V 0\}$ is simply connected.
        \end{itemize}
        Importantly, consider $\V F(\V x) = \frac{1}{x^2+y^2}(-y, x, 0)$. We can see that $\V F$ is irrotational. We can also see that $f(\V x) = \arctan(y/x)$ satisfies $\V F = \grad f$ -- but $f$ is not a single-valued function -- or, if we choose the principal value, it's not continous. Indeed, the domain of $\V F$ is not simply connected, since it is undefined on the $z$-axis.
    \end{example*}

    \subsection{Solenoidal/incompressible vector fields}

    \begin{definition*}
        A vector field $\V F$ is \emph{solenoidal} (or \emph{divergence-free}) in a domain $\mathscr D$ if $\div \V F = 0$ there.
    \end{definition*}
    
    \begin{remark*}
        The nomenclature arises from magnetic fields, which are always solenoidal (this is actually equivalent to the non-existence of magnetic monopoles). In fluid dynamics, on the other hand, such fields are called incompressible.
    \end{remark*}
    
    If there exists a vector field $\V A$ in $\mathscr D$ such that $\V F = \curl \V A$, then $\V F$ is certainly solenoidal because
    \begin{align*}
        \div(\curl \V A) = \pd{}{x_i} \lv{ijk} \pd{A_k}{x_j} = 0
    \end{align*}
    Such an $\V A$ is called a \emph{vector potential}. The converse also holds in a suitable domain $\mathscr D$. 
    
    \begin{remark*}
        For the converse to hold, we need $\mathscr D$ to be simply connected, and we also need \emph{surfaces} in $\mathscr D$ to be able to be shrunk to a point. But this is complicated and non-examinable.
    \end{remark*}
    
    \section{Orthogonal curvilinear coordinate systems}

    \subsection{General curvilinear coordinate systems}

    \begin{definition*}
        A \emph{curvilinear coordinate system} in 3D is a parameterisation $(u, v, w)$ of $\RR^3$ -- in other words, a surjective map $\V x(u, v, w)$ from a subset of $\RR^3$ to $\RR^3$.
    \end{definition*}
    
    A well-known example in $\RR^2$ is plane polar coordinates.

    \begin{remark*}
        Ideally, the map $\V x$ would be invertible, but in practice most coordinate systems fail to be invertible at points -- for example, we can't invert at the origin in polar coordinates.
    \end{remark*}
    We define the \emph{scale factors}
    \begin{align*}
        h_u = \left|\pd{\V x}{u}\right|,\quad h_v = \left|\pd{\V x}{v}\right|,\quad h_w = \left|\pd{\V x}{w}\right|
    \end{align*}
    and corresponding unit vectors
    \begin{align*}
        \V e_u = \frac 1{h_u} \pd{\V x}{u}, \quad \V e_v = \frac 1{h_v} \pd{\V x}{v}, \quad \V e_w = \frac 1{h_w} \pd{\V x}{w}
    \end{align*}
    \begin{definition*}
        An \emph{orthogonal curvilinear coordinate system} is one in which $\V e_u$, $\V e_v$, $\V e_w$ are mutually orthogonal, and we normally re-order them to form a right-handed set. This allows them to be used as a basis.
    \end{definition*}
    Importantly, this basis is \emph{local} -- the basis vectors depend on position. For example, a vector field $\V F(\V x)$ can be expressed at $\V x$ in terms of its components relative to the basis vectors at $\V x$ -- so $\V F = F_u \V e_u + F_v \V e_v + F_w \V e_w$. But even if $\V F$ is a constant vector, \emph{its components won't be}.
    
    By extending \cref{1.3} to the vector function $\V x(u, v, w)$, we have the following result:
    \begin{proposition}
        \label{3.1}\begin{align*}
            d\V x &= \pd{\V x}{u} \D u + \pd{\V x}{v} \D v + \pd{\V x}{w} \D w \\
            &= h_u \D u \V e_u + h_v \D v \V e_v + h_w \D w \V e_w
        \end{align*}
    \end{proposition}
    whereas in Cartesian coordinates it would simply be $\D u \V e_u + \D v \V e_v + \D w \V e_w$. Similarly, the position vector is \emph{not} given by $u \V e_u + v \V e_v + w \V e_w$ -- there is a difference between the coordinates $(u, v, w)$ and the position vector $\V x(u, v, w)$.

    Consider a displacement that keeps $u$ fixed, but allows $v$ and $w$ to vary. Then, from \cref{3.1}, $d\V x$ is a linear combination of $\V e_v$ and $\V e_w$, so $d\V x \perp \V e_u$ for all possible $d\V x$. Hence $\V e_u$ is perpendicular to any surface of constant $u$, and similarly for $\V e_v$ and $\V e_w$. So surfaces of constant $u$, constant $v$ and constant $w$ all intersect orthogonally at any given point.

    \subsection{Spherical polar coordinates}

    \begin{definition*}
        \emph{Spherical polar coordinates} are given by
        \begin{align*}
            x &= r \sin \theta \cos \phi \\
            y &= r \sin \theta \sin \phi \\
            z &= r \cos \theta
        \end{align*}
        where $r \geq 0$, $\theta \in [0, \pi]$, and $\phi$ can be restricted to $[0, 2\pi)$ -- DIAGRAM!! We can invert this using

            \tan \phi &= y/x
        \end{align*}
    \end{definition*}

        Some people order the coordinates as $(r, \phi, \theta)$ instead. Some swap the definitions of $\phi$ and $\theta$. Some do both. Some replace $\theta$ with $\frac{\pi}{2} - \theta$. This can be extremely confusing.

        \label{3.4} For spherical polar coordinates, \begin{align*}

    \section{Differential geometry of curves, surfaces and volumes}

    \subsection{Curves and arc length}

    \begin{definition*}
        A \emph{curve} $C$ is a map $\V x: [a, b] \subset \RR \to \RR^3$, in other words $\V x(t) = (x(t), y(t), z(t))$ for $a \leq t \leq b$. We say that the curve is \emph{parameterised} by $t$. A curve is \emph{closed} if $\V x(a) = \V x(b)$.
    \end{definition*}
    
    In this course we will require that curves be continuous and piecewise smooth (infinitely differentiable everywhere except a finite number of points). DIAGRAM: mostly smooth but a few spiky points.

    Normally, we will use the word curve to refer to the \emph{image} of the map $C$, not the map itself.

    \begin{definition*}
        The \emph{arc length} of a curve is given by
        \begin{align*}
            s = \int ds = \int |\dot{\V x}(t)| \D t
        \end{align*}        
    \end{definition*}
    This is motivated by
    \begin{align*}
        \delta s = |\delta \V x| &= \sqrt{\delta x^2 + \delta y^2 + \delta z^2} \\
        &= \sqrt{\left(\frac{\delta x}{\delta t}\right)^2+\left(\frac{\delta y}{\delta t}\right)^2+\left(\frac{\delta z}{\delta t}\right)^2} \\
        &= \left|\frac{\delta \V x}{\delta t}\right|\delta t
    \end{align*}
    which, taking the limit, gives
    \begin{proposition}
        \label{4.1 }\[
            \frac{ds}{dt} = |\dot{\V x}|
        \]
    \end{proposition}
    
    Thus we obtain $s$ as a function of $t$ which we can, at least theoretically, invert to reparameterise a curve in terms of its arc length, which could be quite useful -- the so-called \emph{natural parameterisation}.
    
    \begin{itemize}
        \item Given two curves $C_1$ and $C_2$ with matching end-points, we define the curve $C_1 + C_2$ to be the two curves joined end-to-end.
        \item The curve $-C$ is the curve $C$ traversed in the opposite direction -- $\V x_{-C}(t) = \V x_C(a+b-t)$.
    \end{itemize}
    
    \begin{example*}
        \begin{itemize}
            \item Consider $\V x(t) = (at^2, 2at, 0)$, $t \in [-1, 1]$. This is a parameterisation (the standard one) of the parabola $y^2=4ax$ between $y= \pm 2a$.
            \item $\V x(t) = (a \cos t, b \sin t, 0)$, $t \in [0, \pi]$ is a parameterisation of the upper half of the ellipse $x^2/a^2+y^2/b^2=1$.
            \item $\V x(t) = (\cos t, \sin t, t)$ is a helix. In cylindrical polars, $\rho = 1$, $\phi = t$, $z = t$, in other words the curve winds its way around the unit cylinder.
        \end{itemize}
        When drawing our curves, we should add an arrow to indicate the direction in which $t$ increases.
    \end{example*}
    
    \begin{example*}[Bonus non-examinable!]
        Let $R(t) = 2 + \cos \alpha t$ and $\V x(t) = (R(t) \cos \beta t, R(t) \sin \beta t, \sin \alpha t)$. This curve winds its way around the torus given in cylindrical polars by $(\rho-2)^2 + z^2 = 1$. If $\alpha/\beta \in \QQ$, the curve closes on itself -- in fact, if $\alpha/\beta = p/q$ where $p$ and $q$ are coprime integers, then we have the \emph{torus knot} $K_{q, p}$. But if $\alpha/\beta \notin \QQ$, the curve covers the entire torus.
    \end{example*}
    
    \subsection{The Frenet-Serret equations}

    We will assume in this section that $\V x$ is parameterised by arc length $s$. If it is not, then the following results \emph{do not apply}. For clarity, we will use primes to denote differentiation with respect to this canonical parameterisation. We define
    \begin{proposition}
        \begin{align*}
            \V t &= \V x' && \text{the \emph{tangent} vector} \\
            \V n &= \frac 1 {|\V x''|}\V x'' && \text{the \emph{principal normal} vector} \\
            \V b &= \V t \times \V n && \text{the \emph{binormal} vector}
        \end{align*}
    \end{proposition}

    \begin{proposition}[Frenet-Serret equations]
        \begin{align*}
            \V t' &= \kappa \V n \\
            \V n' &= \tau \V b - \kappa \V t \\
            \V b' &= -\tau \V n
        \end{align*}
        where $\kappa = |\V x''|$ and $\tau = -\V b' \cdot \V n$ are scalars, not necessarily constant, called the \emph{curvature} and \emph{torsion} of the curve respectively.
    \end{proposition}
    
    \begin{proof}
        First, $ds = |d \V x|$, so $|\frac{d \V x}{ds}| = 1$ -- hence $|\V t| = 1$. Now $\kappa = |\V x''| = |\V t'|$, so $\V n = \V t'/\kappa$ by definition. Hence $|\V n| = 1$ and $\V t' = \kappa \V n$ (later on we'll worry about $\kappa = 0$).

        Differentiating $\V t \cdot \V t = 1$ gives $2\V t \cdot \V t' = 0$, so $\V t \cdot \V n = 0$, that is $\V t \perp \V n$. Then, by the definition $\V b = \V t \times \V n$, we can be sure $\{\V t, \V n, \V b\}$ is orthonormal and right-handed. This set (which depends on the position along the curve) is known as the \emph{moving trihedral}.

        Now $\V b \cdot \V b = 1$, so $\V b \cdot \V b' = 0$, and $\V t \cdot \V b = 0$, so 
        \begin{align*}
            0 &= \V t' \cdot \V b + \V t \cdot \V b' \\
            &= \kappa \V n \cdot \V b + \V t \cdot \V b' \\
            &= \V t \cdot \V b'
        \end{align*}
        Therefore $\V b' \perp \V t$ and $\V b' \perp \V b$, so it must be that $\V b' \parallel \V n$ and, from the definition of $\tau$, $\V b' = -\tau \V n$.

        Finally, $\V n = \V b \times \V t$, so
        \begin{align*}
            \V n' = \V b' \times \V t + \V b \times \V t' = \tau \V n \times \V t + \V b \times \kappa \V n = \tau \V b - \kappa \V t
        \end{align*}
        as required.

        Let's now worry about $\kappa = 0$. In that case, $\V n$, $\V b$ and $\tau$ are also all undefined. If $\kappa = 0$ everywhere, the curve is just a straight line.
    \end{proof}
    
    \begin{definition*}
        The \emph{radius of curvature} is defined as $R = \kappa^{-1}$.
    \end{definition*}

Then a circle of radius $R$ centred at $\V x + R \V n$ is the best possible approximation of the curve at $\V x$ by a circle -- it has the same values of $\V x$, $\V x'$ and $\V x''$ as the curve itself (as well as $\kappa$, of course).

    \subsection{Surfaces, surface elements and the normal}

    \begin{definition*}
        A \emph{surface} is a map $\V x: U \subseteq \RR^2 \to \RR^3$. We typically use $u$ and $v$ to refer to the arguments of $\V x$.
    \end{definition*}
    
    As with curves, we will normally refer to the image of the map, not the map itself.
    
    \begin{example*}
        We might wish to make a hemispherical surface, $H_R = \{\V x: |\V x| = R, z < 0\}$. We can use spherical polar coordinates to parameterise this as
        \begin{align*}
            \V x(\theta, \phi) = \begin{pmatrix}
             R \sin \theta \cos \phi \\
             R \sin \theta \sin \phi \\
             R \cos \theta  \\
            \end{pmatrix}
        \end{align*}
        with $\theta \in (\pi/2, \pi]$, $\phi \in [0, 2\pi)$.
    \end{example*}
    
    \begin{lemma*}
        Given a parallelogram with two sides $\V a$ and $\V b$, its area is $|\V a \times \V b|$.
    \end{lemma*}
    
    Consider now a region lying between curves of constant $u$ and curves of constant $v$. What is the area of such a region?

    Suppose we have two such curves meeting at $\V x(u, v)$, and nearby curves at constant $u + \delta u$ and $v + \delta v$. The part of $S$ bounded by these curves forms, at lowest order, a very small parallelogram, with sides given by
    \begin{align*}
        \V x(u + \delta u, v) - \V x(u, v) &= \pd{\V x}{u}\delta u \\
        \V x(u, v + \delta v) - \V x(u, v) &= \pd{\V x}{v}\delta v 
    \end{align*}
    Hence, by the lemma, the area of the region is given by
    \begin{proposition}
        \label{4.4} \[
            dS = \left|\pd{\V x}{u} \times \pd{\V x}{v}\right|\D u \D v
        \]
    \end{proposition}
    This infinitesimal parallelogram is referred to as a \emph{surface element}.
    \begin{remark*}
        Importantly, the normal to the surface at $\V x$ is parallel to $\pd{\V x}{u} \times \pd{\V x}{v}$.
    \end{remark*}

    \begin{definition*}
        A \emph{volume} is a map $\V x: U \subseteq \RR^3 \to \RR^3$. We typically use $u$, $v$ and $w$ to refer to the arguments of $\V x$.
    \end{definition*}
    
    \begin{example*}
        To parameterise a sphere, we can do a similar thing to before, but let $r \in [0, R]$.
    \end{example*}
    
    Similarly, we can divide our volumes into regions lying between \emph{surfaces} of constant $u$, $v$ and $w$. To lowest order, these regions are \emph{parallelepipeds} with sides $\pd{\V x}{u} \delta u$, $\pd{\V x}{v} \delta v$ and $\pd{\V x}{w} \delta w$. Hence the volume element is given by
    \begin{proposition}
        \label{4.5} \[

        \]
    \end{proposition}

    \subsection{Line integrals}

    \begin{definition*}
        Given a curve $C$ parameterised by $t \in [a, b]$, the \emph{line integral} of a vector field $\V F$ along $C$ is
        \begin{align*}
            \Aboxed{\int_C \V F \cdot d \V x = \int_a^b \V F(\V x(t)) \cdot \frac{d \V x}{dt} \D t}
        \end{align*}
        An alternative but equivalent definition is to return to a Riemann sum: we define
        \begin{align*}
            \int_C \V F \cdot d \V x &= \lim_{\Delta \to 0}\sum_{n=0}^{N-1} \V F(\V x_n) \cdot \delta \V x_n
        \end{align*}
    \end{definition*}
    
    \begin{remark*}
        We can also take a line integral on a scalar field, or take a line integral while crossing (!) $\V F$ with $d \V x$ instead of dotting.
    \end{remark*}
    
    If $C$ is a closed curve, then the line integral around the whole curve has a special notation
    \begin{align*}
        \oint_C \V F \cdot d \V x
    \end{align*}
    and this is sometimes known as the \emph{circulation} of $\V F$ around $C$. It also doesn't matter where we start along such a curve, so long aswe go all the way around.

    We have some elementary properties of line integrals:
    \begin{itemize}
        \item $\int_{-C} \V F \cdot d \V x = -\int_C \V F \cdot d \V x$;
        \item $\int_{C_1+C_2} \V F \cdot d \V x = \int_{C_1} \V F \cdot d \V x + \int_{C_2} \V F \cdot d \V x$.
    \end{itemize}
    
    \begin{example*}
        \begin{itemize}
            \item Consider $\V F = (y, -x, 0)$, and $C$ the straight line joining $(-1, 0, 0)$ to $(1, 0, 0)$. Parameterise the curve as $\V x = (t, 0, 0)$ for $t \in [-1, 1]$. Then $\V F(\V x(t)) = (0, -t, 0)$, and $\dot{\V x} = (1, 0, 0)$, so $\int_{C_1} \V F \cdot d \V x = 0$.
            \item Consider $\V F = (3x^2, 2y, 0)$ on the same line. We obtain $\V F \cdot \dot{\V x} = (3t^2, 0, 0) \cdot (1, 0, 0)$ and so $\int_{C_1} \V F \cdot d \V x = \int_{-1}^1 3t^2 \D t = 2$.
            \item Consider $\V F = (y, -x, 0)$ along the curve $C_2$ joining $(-1, 0, 0)$ to $(1, 0, 0)$ parameterised as $\V x = (t-1, t(t-2), t(t^2-4))$ for $t \in [0, 2]$. We have
            \begin{align*}
                \V F &= (t(t-2), 1-t, 0) && \V x = (1, 2t-2, 3t^2-4)
            \end{align*}
            so
            \begin{align*}
                \int_{C_2} \V F \cdot d \V x &= \int_0^2 t(t-2) - 2(t-1)^2 \D t \\
                &= -\frac 8 3
            \end{align*}
            \item Consider $\V F = (3x^2, 2y, 0)$ along $C_2$. We have
            \begin{align*}
                \int_{C_2} \V F \cdot d \V x &= \int_0^2 (3(t-1)^2, 2t(t-2), 0) \cdot (1, 2t-2, 3t^2-4) \D t \\
                &= \int_0^2 4t^3 - 9t^2 + 2t - 3 \D t \\
                &= 2
            \end{align*}
            \item What is the work done by a force field $\V F = (y^2e^x, e^y + 2ye^x, 0)$ moving a particle along $C_3$, which is the ellipse $x^2/a^2 + y^2/b^2 = 1$ in the $(x, y)$-plane starting at $(a, 0, 0)$ and going anticlockwise to $(0, b, 0)$?    
            
            First, we need to parameterise the curve by $\V x(\theta) = (a \cos \theta, b \sin \theta, 0)$ for $\theta \in [0, \pi/2]$. Then
            \begin{align*}
                \V F(\V x(\theta)) = (b^2 \sin^2 \theta \, e^{a \cos \theta}, e^{b \sin \theta} + 2b \sin \theta \, e^{a \cos \theta}, 0)
            \end{align*}
            so that
            \begin{align*}
                \V F \cdot \frac{d \V x}{d \theta} = -ab^2 \sin^3 \theta \, e^{a \cos \theta} + b \cos \theta \, e^{b \sin \theta}+ 2b^2 \sin \theta \cos \theta \, e^{a \cos \theta}
            \end{align*}
            so
            \begin{align*}
                \int_{C_3} \V F \cdot d \V x &= [e^{b \sin \theta}]^{\pi/2}_0 + \int_0^{\pi/2} b^2 \sin \theta \, e^{a \cos \theta} (-a + 2 \cos \theta + a \cos^2 \theta) \D \theta \\
                &= e^b - 1 + \int_1^0 b^2 e^{au} (a^2-2u-au^2) \D u \qquad (u = \cos \theta) \\
                &= e^b - 1 + [b^2e^{au}(1-u^2)]^0_1 \\
                &= e^b - 1 + b^2
            \end{align*}
            Around the whole ellipse, starting at $(a, 0, 0)$ and going anticlockwise all the way round, we would find $\oint \V F \cdot d \V x = 0$.
        \end{itemize}
    \end{example*}
    
    \begin{example*}
        Let's do one in cylindrical polar coordinates! Consider $\V F = z \V e_\phi$ on the path $\rho = z^2$, $\phi = \pi z$ from $z = 0$ to $z=1$. Here, we have
        \begin{align*}

            &= (2z \V e_\rho + \pi \rho \V e_\phi + \V e_z) \D z

        \end{align*}

    \subsection{Path independence}

    \begin{definition*}[Exact differential forms -- reminder]
        We say that a differential form $\V F \cdot d \V x$ is \emph{exact} if there exists a single-valued scalar field $f(\V x)$ such that $\V F \cdot d \V x = \D f$ for all $d \V x$.
    \end{definition*}
    
    From \cref{2.2}, we know that $df = \grad f \cdot d \V x$ -- so $\V F \cdot d \V x$ is exact if and only if there exists a single-valued function $f$ such that $\V F = \grad f$, i.e. if $\V F$ is a conservative vector field.

    If $\V F$ is a conservative vector field, and $C$ is a path from $\V a$ to $\V b$, then
    \begin{align*}
        \int_C \V F \cdot d \V x = \int_C df = f(\V b) - f(\V a)
    \end{align*}
    and this is notably independent of the particular path. So we can, if we like, write
    \begin{align*}
        \int_{\V a}^{\V b} \V F \cdot d \V x
    \end{align*}
    
    In particular, we have the important result that, if $C$ is a closed curve,
    \begin{align*}
        \oint_C \V F \cdot d \V x = 0
    \end{align*}
    for any conservative $\V F$, since $\V b = \V a$ and $f$ is single-valued.

    Now we have the awesome result for vector fields $\V F$ in a simply-connected domain:

    \begin{enumerate}[label=\protect\circled{\arabic*}]
        \item $\V F$ is a conservative vector field;
        \item there exists a single-valued function $f(\V x)$ such that $\V F = \grad f$;
        \item $\V F \cdot d \V x$ is an exact differential;
        \item $\V F$ is irrotational;
        \item $\grad \times \V F = \V 0$;
        \item $\pd{F_i}{x_j} = \pd{F_j}{x_i}$ for all $i, j$;
        \item for any $\V a$, $\V b$ in the domain, $\int_{\V a}^{\V b} \V F \cdot d \V x$ is path-independent;
        \item $\oint_C \V F \cdot d \V x = 0$ for all closed curves $C$ in the domain.
    \end{enumerate}
    To prove $\circled 8 \implies \circled 7$ and $\circled 7 \implies \circled 2$ is non-trivial and we'll have to wait until section 6.4.

    \subsection{Area integrals}

    To integrate over a region $A \subset \RR^2$, we cover $A$ with a large number of small disjoint subsets $\delta A_i$. Then
    \begin{align*}
        \int_A f(x, y) \D A = \lim_{\Delta \to 0} \sum_{i=1}^N f(x_i, y_i) \delta A_i
    \end{align*}
    where in particular $\Delta$ is the maximum linear dimension of any of the $\delta A_i$.

    Informally, we can think of $\int_A f(x, y) \D A$ as the volume under the graph of $z = f(x, y)$, in analogy with the 1D case.

    Note that $A$ refers to the region, as well as its area: we have
    \begin{align*}
        A = \int_A 1 \D A
    \end{align*}
    
    The most obvious way to divide $A$ up is with rectangles parallel to the $x$ and $y$ axes, each with area $\delta x \delta y$. Summing over a horizontal strip with a fixed value of $y$, with constant height $\delta y$ and letting $\delta x \to 0$, we obtain
    
    \begin{align*}
        \left(\int_{X(y)} f(x, y) \D x\right)\delta y
    \end{align*}
    where, for each $y$, $X(y) = \{x: (x, y) \in A\}$. We then sum these over the range of $y$, taking $\delta y \to 0$, to obtain
    \begin{align*}
        \int_{y_{\min}}^{y_{\max}}\left(\int_{X(y)} f(x, y) \D x\right)\D y
    \end{align*}
    But we could have equally well done this the other way to obtain 
    \begin{align*}
        \int_{x_{\min}}^{x_{\max}}\left(\int_{Y(x)} f(x, y) \D y\right)\D x
    \end{align*}
    
    \begin{proposition*}[Fubini's theorem]
        For any reasonable function $f$ over a reasonable $A$, both of these are equal to $\int_A f(x, y) \D A$.
    \end{proposition*}
    
    \begin{remark*}
        Only very basic conditions are needed. For example, if $f$ is piecewise continuous and the boundary of $A$ is a finite piecewise smooth curve, then Fubini's theorem holds.
    \end{remark*}
    
    Other notations for $\int_A f(x, y) \D A$ include $\int_A f(x, y) \D x \D y$, $\int_A f(\V x) d^2 \V x$, $\iint_A f(x, y) \D A$, $\int_A f(x, y) \D S$ and even $\int dx \int dy\, f(x, y)$.

    \begin{example*}
        Let us integrate $f(x, y) = xy^2$ over $A$, a triangle with vertices at $(0, 0)$, $(1, 1)$ and $(1, -1)$ (DIAGRAM). We have
        \begin{align*}
            \int_A f(x, y) \D A &= \int_0^1 \int_{-x}^x xy^2 \D y dx \\
            &= \int_0^1 x\left[\frac 1 3 y^3\right]^x_{-x} \D x \\
            &= \int_0^1 \frac 2 3 x^4 \D x = \frac 2 {15}
        \end{align*}
        Alternatively, we could have done
        \begin{align*}
            \int_A xy^2 \D A &= \int_{-1}^1 \int_{|y|}{1} xy^2 \D x dy \\
            &= \int_{-1}^1 y^2 \left[\frac 1 2 x^2\right]_{|y|}^1 \\
            &= \int_{-1}^1 \frac 1 2 y^2 (1-y^2) \D y \\
            &= \frac 2 {15}
        \end{align*}
        Importantly, the inside integral should have limits specified in terms of $x$ or $y$.       
    \end{example*}
    
    In the special case that $A$ is a rectangle $[a, b] \times [c, d]$ and $f$ is separable, i.e. $f(x, y) = g(x)h(y)$, then
    \begin{align*}
        \int_A f(x, y) \D A = \left(\int_a^b g(x) \D x\right)\left(\int_c^d h(y) \D y\right)
    \end{align*}
    
    \subsection{Change of variables for area integrals}

    \begin{proposition*}
        Let $(u, v) \mapsto (x, y)$ be a bijection from a region $A'$ in the $(u, v)$-plane to a region $A$ in the $(x, y)$-plane. Then
        \begin{align*}
            \int_A f(x, y) \D x dy = \int_{A'} F(u, v) \left|\pd{(x,y)}{(u, v)}\right| \D u dv \\
        \end{align*}
        where $F(u, v) = f(x(u, v), y(u, v))$ and
        \begin{align*}
            J = \pd{(x,y)}{(u, v)} = \det \begin{pmatrix}
            \pd x u & \pd x v \\
            \pd y u & \pd y v \\
            \end{pmatrix}
        \end{align*}
        is called the \emph{Jacobian} of the transformation. Importantly, it is the \emph{modulus} of the Jacobian that appears in the integral.
    \end{proposition*}
    
    \begin{remark*}
        We often call capital $F$ little $f$ instead, although this is an obvious abuse of notation.
    \end{remark*}

    \begin{remark*}
        The change of variables may fail to be a bijection for one or two points -- c.f. polar coordinates. But this is ok.
    \end{remark*}    
    
    \begin{proof}
        Let us divide $A$ up using curves of constant $u$ and $v$. The area of each region is given by
        \begin{align*}
            \left|\pd{\V x}{u} \times \pd{\V x}{v}\right|\D u \D v &= \left|\begin{pmatrix}
             \pd x u \\
             \pd y u \\
             0 \\
            \end{pmatrix} \times \begin{pmatrix}
             \pd x v \\
             \pd y v \\
             0 \\
            \end{pmatrix}\right|\D u \D v \\
            &= \left|\pd x u \pd y v - \pd x v \pd y u\right|\D u \D v \\
            &= |J| \D u \D v
        \end{align*}
        Then we simply need to add up all the infinitesimal areas $f(\V x(u, v)) |J| \D u \D v$.
    \end{proof}
    
    \begin{remark*}
        A full proof of this result requires first a proof of the cross product expression for the infinitesimal area.
    \end{remark*}

    \subsection{Properties of the Jacobian}

    Suppose that $J_1$ is the Jacobian matrix for the transformation $(u, v) \mapsto (x, y)$, and $J_2$ is the Jacobian matrix for $(\xi, \eta) \mapsto (u, v)$. Then, consider the product
    \begin{align*}
        J_1J_2 &= \begin{pmatrix}
            \pd x u & \pd x v \\
            \pd y u & \pd y v \\
        \end{pmatrix}\begin{pmatrix}
            \pd u \xi & \pd u \theta \\
            \pd v \xi & \pd v \theta \\
        \end{pmatrix} \\
        &= \begin{pmatrix}
            \pd x u \pd u \xi + \pd x v \pd v \xi & \pd x u \pd u \theta + \pd x v \pd v \theta \\
            \pd y u \pd u \xi + \pd y v \pd v \xi & \pd y u \pd u \theta + \pd y v \pd v \theta \\
        \end{pmatrix} \\
        &= \begin{pmatrix}
            \pd x \xi & \pd x \theta \\
            \pd y \xi & \pd y \theta \\
        \end{pmatrix}
    \end{align*}
    In other words, the Jacobian of a composition of maps is the product of the Jacobians:
    \begin{align*}
        \pd{(x, y)}{(u, v)} \cdot \pd{(u, v)}{(\xi, \eta)} = \pd{(x, y)}{(\xi, \eta)}

    \begin{align*}

    \end{align*}

    \subsection{Volume integrals}

    Volume integrals are really the same as area integrals, just in one more dimension (in fact these notions will generalise to $\RR^n$). We cover $V$ with a large number of small disjoint volumes, and define
    \begin{align*}
        \int_V f(\V x) \D V = \lim_{\Delta \to 0} \sum_{i=1}^N f(\V x_i) \delta V_i
    \end{align*}
    where $\Delta$ is the maximum linear dimension of all the $\delta V_i$.

    Alternatively, we have the iterated integral definition, in particular
    \begin{align*}
        \int_{x_{\min}}^{x_\text{max}}\left(\int_{Y(x)}\left(\int_{Z(x,y)}f(x, y, z) \, dz\right) \, dy\right) \, dx
    \end{align*}
    Notations include $\int_V f(\V x) \D V$, $\int_V f(x, y, z) \D x dy \, dz$, $\iiint_V f(x, y, z) \D V$, $\int_V f(\V x) d^3 \V x$ and $\int dx \int dy \int dz f(x, y, z)$.

    \begin{example*}
        Let $V$ be the cuboid $x \in [-a, a]$, $y \in [-b, b]$, $z \in [-c, c]$. Let us calculate $\int_V \V F \D V$ where $\V F$ is the vector field $(z^2, 0, yz)$.

    \subsection{Change of variables for volume integrals}
    
    \begin{proposition*}
        Let $(u, v) \mapsto (x, y)$ be a bijection from a region $A'$ in the $(u, v)$-plane to a region $A$ in the $(x, y)$-plane. Then
        \begin{align*}
            \int_A f(x, y, z) \D x dy \, dz = \int_{A'} F(u, v, w) \left|\pd{(x,y,z)}{(u, v,w)}\right| \D u dv \, dw \\
        \end{align*}
        where $F(u, v, w) = f(x(u, v, w), y(u, v, w), z(u, v, w))$ and
        \begin{align*}
            J = \pd{(x,y,z)}{(u, v,w)} = \det \begin{pmatrix}
            \pd x u & \pd x v & \pd x w \\
            \pd y u & \pd y v & \pd y w \\
            \pd z u & \pd z v & \pd z w
            \end{pmatrix}
        \end{align*}
        is the Jacobian.
    \end{proposition*}
    
    \begin{proof}
        The proof is almost identical. We have
        \begin{align*}
            dV = \left|\pd{\V x}u \cdot \left(\pd{\V x}v \times \pd{\V x}w\right)\right|\D u dv \, dw = \left|\det\left(\pd{\V x}u \middle| \pd{\V x}v \middle| \pd{\V x}w\right)\right| = |J| \D u dv \, dw
        \end{align*}
    \end{proof}

    \lecture{Lecture 12 -- 18/02/26}
    
    All properties of the Jacobian established in two variables also hold for three variables.

    The most common changes of variables are to either cylindrical or spherical polar coordinates. We have
    \begin{align*}
        \Aboxed{J_\text{cyl} &= \det {\begin{pmatrix}
         \cos \phi & -\rho \sin \phi & 0 \\
         \sin \phi & \rho \cos \phi & 0 \\
         0 & 0 & 1 \\
        \end{pmatrix}} = \rho} \\
        \Aboxed{J_\text{sph} &= \det {\begin{pmatrix}
         \sin \theta \cos \phi & r \cos \theta \cos \phi & -r \sin \theta \sin \phi \\
         \sin \theta \sin \phi & r \cos \theta \sin \phi & r sin \theta \cos \phi \\
         \cos \theta & -r \sin \theta & 0 \\
        \end{pmatrix}} = r^2 \sin \theta}
    \end{align*}
    Furthermore, $\rho$ and $r^2 \sin \theta$ are always positive (recall $\theta \in [0, \pi]$), so we don't need to worry about the absolute value.

    \begin{example*}
        Let us calculate the volume of the sphere $r \leq R$. We have
        \begin{align*}
            \int_{r \leq R} 1 \D V &= \int_{r=0}^R \int_{\theta=0}^\pi \int_{\phi=0}^{2\pi} 1 \cdot r^2 \sin \theta \D r d\theta \, d\phi \\
            &= \int_0^R r^2 \D r \int_0^\pi \sin \theta \D \theta \int_0^{2\pi} d\phi \\
            &= \frac 4 3 \pi R^3
        \end{align*}
    \end{example*}
    
    \begin{example*}
        That was easy! Let's instead calculate the volume of a sphere with a cylinder of radius $a$ along the $z$-axis removed from it. We should use cylindrical polars with the cylinder occupying $\rho \leq a$ and the sphere $\rho^2 + z^2 \leq R^2$. We obtain
        \begin{align*}
            V &= \int_{\rho=a}^R \int_{\phi=0}^{2\pi} \int_{z=-\sqrt{R^2-\rho^2}}^{\sqrt{R^2-\rho^2}} 1 \cdot \rho \D z d\phi \, d\rho \\
            &= 2\pi \int_a^R 2\rho \sqrt{R^2-\rho^2} \D \rho \\
            &= \frac 4 3 \pi (R^2-a^2)^{3/2}
        \end{align*}
    \end{example*}
    
    \subsection{Classification of surfaces}

    A surface $S$ can be categorised as either \emph{closed} or \emph{open}. An open surface is one which has a boundary curve, denoted $\pr S$. The surface is said to \emph{span} the curve. A \emph{closed} surface is one with no boundary curve -- $\pr S = \emptyset$.

    \begin{example*}
        The hemispherical shell $H_R = \{\V x: |\V x| = R, z < 0\}$ is an open surface with $\pr H_R = \{\V x : x^2 + y^2 = R^2, z = 0\}$, i.e. a circle.

        The spherical shell $S_R = \{\V x: |\V x| = R\}$ is a closed surface, as it has no boundary.
    \end{example*}
    
    We can further categorise surfaces as \emph{orientable} (having two sides) or \emph{non-orientable} (having only one side). We shall only consider orientable surfaces.
    
    The important thing about closed orientable surfaces is that they have an inside and an outside, while an open or non-orientable surface does not.

    These definitions also only apply to finite surfaces -- whenever we are considering surfaces of infinite extent, we will treat them as the limit of finite surfaces.
    
    \subsection{Surface integrals}

    Integrating over a two-dimensional surface $S$ embedded in $\RR^3$ is unsurprisingly similar to an area integral. We divide $S$ into small subsets $\delta S_i$, and define
    \begin{align*}
        \int_S f(\V x) \D S = \lim_{\Delta \to 0} \sum_{i=1}^n f(\V x_i) \delta S_i
    \end{align*}
    Importantly, these $\delta S_i$ are not flat, unlike in the case of an area integral. Alternative notations include $\iint_S f(\V X) \D S$ and, if $S$ is closed, $\oiint_S f(\V x) \D S$.

    At each point on an orientable surface there will be two unit normals, pointing in opposite directions. Either normal can be denoted $\hvec n$ -- the one we choose depends on the particular application.

    If $\V F$ is a vector field, we might want to consider $\int_S \V F(\V x) \cdot \hvec n(\V x) \D S$. This is known as the \emph{flux} through $S$. We also define
    \begin{align*}
        d \V S = \hvec n \D S
    \end{align*}
    so that the flux of $\V F$ through $\V S$ is
    \begin{align*}
        \int_S \V F \cdot d \V S
    \end{align*}

    \begin{proposition}[The magic formula]
        If $S$ is parameterised by $u$ and $v$, the vector $d \V S$ is given by
        \begin{align*}
            d \V S = \pm \pd{\V x}u \times \pd{\V x}v \D u dv
        \end{align*}
    \end{proposition}
    
    \begin{proof}
        Using \cref{4.4}, we have
        \begin{align*}
            \hvec n = \pm \frac{\pd{\V x}u \times \pd{\V x}v}{\left|\pd{\V x}u \times \pd{\V x}v\right|}, \qquad dS = \left|\pd{\V x}u \times \pd{\V x}v\right|\D u dv
        \end{align*}
        from which the result follows.
    \end{proof}
    
    Using these results reduces a surface integral to an area integral in the flat $(u, v)$-plane.
    
    \begin{example*}
        Let us find the surface area of the hemispherical shell $H_R = \{\V x: |\V x| = R, z < 0\}$. Let us parameterise $H_R$ by
        \begin{align*}
            \V x(\theta, \phi) = \begin{pmatrix}
             R \sin \theta \cos \phi \\
             R \sin \theta \sin \phi \\
             R \cos \theta \\
            \end{pmatrix}
        \end{align*}
        with $\theta \in (\frac{\pi}2, \pi]$, $\phi \in [0, 2\pi)$. Then
        \begin{align*}
            \pd{\V x}\theta \times \pd{\V x}\phi = R \begin{pmatrix}
             \cos \theta \cos \phi \\
             \cos \theta \sin \phi \\
             -\sin \theta \\
            \end{pmatrix} \times R \begin{pmatrix}
             -\sin \theta \sin \phi \\
             \sin \theta \cos \phi \\
             0 \\
            \end{pmatrix} = R^2 \begin{pmatrix}
             \sin^2 \theta \cos \phi \\
             \sin^2 \theta \sin \phi \\
             \sin \theta \cos \theta \\
            \end{pmatrix}
        \end{align*}
        so
        \begin{align*}
            dS = R^2 \sin \theta \D \theta d\phi
        \end{align*}
        Hence, the surface area is
        \begin{align*}
            \int_{H_R} 1 \D S &= \int_{\theta = \frac{\pi}{2}}^\pi \int_{\phi=0}^{2 \pi} R^2 \sin \theta \D \theta d \phi \\
            &= R^2 \int_{\frac{\pi}{2}}^\pi \sin \theta \D \theta \int_0^{2\pi} 1 \D \phi \\
            &= 2\pi R^2
        \end{align*}
    \end{example*}

    \section{Integral theorems}

    In general, integral theorems aim to reduce the dimension of an integral, much as the fundamental theorem of calculus does for normal integrals.

    \subsection{Green's theorem in the plane}

    \begin{theorem}[Green's theorem in the plane]
        Given an area $A \subset \RR^2$ with piecewise smooth boundary $C = \pr A$ and continuously differentiable functions $P(x, y)$ and $Q(x, y)$,
        \begin{align*}
            \int_A \left(\pd Q x - \pd P y \right) \D x dy = \oint_C (P \D x + Q \, dy)
        \end{align*}
        where $C$ is traversed in the positive sense (normally meaning anti-clockwise).
    \end{theorem}
    
    \begin{proof}
        Initially, suppose that $A$ is a simple shape such that, for each $y$, there is a single range of values for $x$, $x^-(y) \leq x \leq x^+(y)$; likewise, for each $x$, $y^-(x) \leq y \leq y^+(x)$. Then, we may divide $C$ into $C^+$ and $C^-$, which satisfy respectively $x = x^+(y)$ and $x = x^-(y)$ for every $y$ in the range. Now
        \begin{align*}
            \int_A \pd Q x \D x dy &= \int_{y_\text{min}}^{y_\text{max}} \int_{x^-(y)}^{x^+(y)} \pd Q x \D x dy \\
            &= \int_{y_\text{min}}^{y_\text{max}} Q(x^+(y), y) - Q(x^-(y), y) \D y \\
            &= \int_{C^+} Q(x, y) \D y + \int_{C^-} Q(x, y) \D y \\
            &= \oint_C Q \D y
        \end{align*}
        Note that the minus sign is lost since $C^-$ goes from $y_\text{max}$ to $y_\text{min}$, unlike $C^+$. The exact same approach will work for the $\pd P y$ integral with some care, which will suffice for the result.
        
        This completes the proof of Green's theorem for simple shapes. If, on the other hand, $A$ is a `complicated' shape, we claim we can break it up into simple shapes $A_i$, and apply Green's theorem to each of these.
    \end{proof}
    
    \begin{remark*}
        In particular, we note that $\int_A \pd Q x \D x dy = \sum_i \int_{A_i} \pd Q x \D x dy$, since the $A_i$ are disjoint. Furthermore, $\oint_{\pr A} Q \D y = \sum_i \oint_{\pr A_i} Q \D y$, because, although each $\pr A_i$ contains segments not part of $\pr A$, those contributions exactly cancel out. Hence the result follows.
    \end{remark*}
    
    \lecture{Lecture 14 -- 23/02/26}
    
    \begin{example*}
        Let us consider the ellipse $x^2/a^2 + y^2/b^2 \leq 1$ with $P = -\frac 1 2 y$, $Q = \frac 1 2 x$. Then
        \begin{align*}
            \pi ab = \int_A 1 \D x dy = \int_A \left(\pd Q x - \pd P y\right) \D x dy
        \end{align*}
        and
        \begin{align*}
            \oint_C P \D x + Q \D y &= \int_0^{2\pi} \left(-\frac 1 2 b \sin t, \frac 1 2 a \cos t\right) \cdot \left(-a \sin t, b \cos t\right) \D t \\
            &= \pi ab
        \end{align*}
    \end{example*}
    
    \subsubsection{Multiple boundary curves}

    We may also consider areas with multiple boundary curves -- in particular, internal boundary curves as well as external boundary curves. In this case, it is necessary to traverse internal boundaries \emph{clockwise} as opposed to counter-clockwise.

    To illustrate this, let us consider an area $A$ with outer boundary $C_1$ traversed counter-clockwise and one inner boundary $C_2$ traversed clockwise.

    Let $A_2$ be the interior of $C_2$, and $A_1 = A \cup A_2$, so $A_1$ is the entire interior of $C_1$. Then $\int_{A_1} = \int_A + \int_{A_2}$, so $\int_A = \int_{A_1} - \int_{A_2} = \oint_{C_1} - \oint_{-C_2}$, by Green's theorem -- in particular, we integrate over $-C_2$ since $C_2$ is clockwise. Then clearly $\int_A = \oint_{C_1} + \oint_{C_2}$ as required.
    
    \begin{remark*}
        This argument unfortunately assumes that the integrand is defined in $A_1$ (the area in the hole), which is of course not true in general. In the general case, we can put a cut in $A$ between the interior and exterior boundaries -- in that case, it's clear that, for the outer boundary to be traversed counter-clockwise, the inner boundary is traversed clockwise. Then, taking the cut arbitrarily small, the result follows.
    \end{remark*}
    
    \subsection{Stokes' theorem}

    \begin{theorem}[Stokes' theorem]
        If $\V F(x)$ is a continuously differentiable vector field and $S$ is a piecewise smooth orientable open surface with piecewise smooth boundary curve $C = \pr S$,
        \begin{align*}
            \int_S \curl \V F \cdot d\V S = \oint_C \V F \cdot d\V x
        \end{align*}
        where $C$ is traversed in the positive sense with regard to the normal on $S$.  
    \end{theorem}
    
    \begin{remark*}
        Note that Green's theorem in the plane is a 2D case of Stokes' theorem. Let $S$ be a flat surface with $z=0$ and take $\V F = (P, Q, 0)$ -- the result follows.
    \end{remark*}
    
    What does `the positive sense with regard to the normal on $S$' mean? Firstly, note that, since $S$ is open, we must specify a normal to give any notion of direction. Then, we say that if $\V t$ is the tangent vector to the curve $C$ and $\V n$ is the normal to $S$ at the same surface, then $\V t \times \V n$ should point away from the surface. Equivalently, if one walks along $C$ with one's feet on $C$ and one's head in the direction of $N$, the surface should be on one's \emph{left}. DIAGRAM!

    Alternatively, one has the right-hand rule -- if one's thumb points along $\hvec n$, then one's curled fingers indicate the direction of $C$. 

    \begin{proof}[Sketch proof, non-examinable]
        We start from Green's theorem, parameterising the surface $S$ using $(u, v) \in D \subset \RR^2$. Then the right hand side becomes
        \begin{align*}
            \oint_C \V F \cdot d\V x = \oint_{\pr D} \V F \cdot \left(\pd{\V x} u \D u + \pd{\V x}v \D v\right)
        \end{align*}
        Letting $P(u, v) = \V F(\V x(u, v)) \cdot \pd{\V x}u$ and $Q(u, v) = \V F(\V x(u, v)) \cdot \pd{\V x}v$ and apply Green's theorem in the $(u, v)$-plane to obtain
        \begin{align*}
            \int_D \left(\pd Q u - \pd P v\right) \D u \D v
        \end{align*}
        Then, one can use suffix notation and the chain rule to show
        \begin{align*}
            (\curl \V F) \cdot \left(\pd{\V x}u \times \pd{\V x}v\right) = \pd{}{u} \left(\V F \cdot \pd{\V x}v \right) - \pd{}{v} \left(\V F \cdot \pd{\V x}u\right)
        \end{align*}
        from which the result follows.
    \end{proof}
    
    \begin{example*}
        Consider the open surface $S$ given by $z = (x^2+y^2)^2$, $0 \leq z \leq 1$ and the vector field $\V F = (x^2y, 0, 0)$. We use cylindrical polars, so $S$ becomes $z=\rho^4$, so that $S$ is parameterised by $\V x = (\rho \cos \phi, \rho \sin \phi, \rho^4)$. Applying the magic formula,
        \begin{align*}
            d\V S = -\begin{pmatrix}
             \cos \phi \\
             \sin \phi \\
             4\rho^3 \\
            \end{pmatrix} \times \begin{pmatrix}
             -\rho \sin \phi \\
             \rho \cos \phi \\
             0 \\
            \end{pmatrix}\D\rho\D\phi = \begin{pmatrix}
             4\rho^4\cos\phi \\
             4\rho^4\sin\phi \\
             -\rho \\
            \end{pmatrix}\D\rho\D\phi
        \end{align*}
        where we have chosen the negative sign to get the downward-facing normal. Now $\curl \V F = (0, 0, -x^2) = (0, 0, -\rho^2 \cos^2 \phi)$, so
        \begin{align*}
            \int_S \curl \V F \cdot d\V S = \int_{\rho=0}^1 \int_{\phi=0}^{2\pi} \rho^3 \cos^2 \phi \D \phi \D \rho = \frac \pi 4
        \end{align*}
        Now for the line integral. $C = \pr S$ is the curve $x^2+y^2=1$, $z=1$, that is $\rho = 1$, $z=1$, $\phi \in [0, 2\pi)$. To determine the direction, apply the right hand rule -- we see that $C$ must be traversed in the direction of decreasing $\phi$, or clockwise when viewed from above. Hence we have
        \begin{align*}
            \oint_C \V F \cdot d\V x &= \int_{2\pi}^0 x^2y \frac{d\V x}{d\phi} \D \phi \\
            &= \int_0^{2\pi} \cos^2\phi \sin^2\phi \D \phi
        \end{align*}       
    \end{example*}
    
    \subsubsection{Closed surfaces}

    Stokes' theorem also applies to closed surfaces $S$, which have $\pr S = \emptyset$. Hence
    \begin{align*}
        \oiint_S \curl \V F \cdot d\V S = 0
    \end{align*}
    
    \subsubsection{Multiple boundary curves}
    
    We may again have the case of two or more boundary curves. We have to apply the right hand rule for the directions of $\V n$ and the curve separately for \emph{each} boundary.

        On $z = 0$, we instead need to move in the direction of increasing $\phi$. We thus parameterise $C_1$ as $\V x(t) = (\cos t, \frac 1 2 \sin t, 0)$ with $\V F = (\frac 1 2 \sin t, 0, \cos^2 t + \frac 1 2 \sin^2 t)$ for $t$ increasing from 0 to $\pi$. Hence
        \begin{align*}
            \int_{C_1} \V F \cdot d \V X = \int_0^{2\pi} \left(-\frac 1 2 \sin^2 t\right) \D t = -\frac{\pi} 2
        \end{align*}
        To verify Stokes' theorem, we observe
        \begin{align*}
            \frac{3\pi} 2 = 2 \pi - \frac{\pi} 2
        \end{align*}
    \end{example*}

    \subsection{The divergence theorem}

    \begin{theorem}[Divergence theorem]
        Let $\V F(\V x)$ be a continuously differentiable vector field and $V$ a volume enclosed by a piecewise smooth orientable \emph{closed} surface $S = \pr V$. Then
        \begin{align*}
            \int_V \div \V F d V = \int_S \V F \cdot d \V S
        \end{align*}
        where the normal to $S$ points outwards from $V$.        
    \end{theorem}
    The theorem is also known as Gauss's theorem, or sometimes Green's theorem.

    \begin{proof}[Sketch proof, non-examinable]
        To prove this we once again start by assuming $V$ is a simple shape so that, for each fixed $y$ and $z$, we have a single range of $x$, that is $x^-(y, z) < x < x^+(y, z)$ (DIAGRAM). $S$ can then be divided into a right-hand part $S^+$ on which $x = x^+(y, z)$ and a left-hand part $S^-$ on which $x = x^-(y, z)$. Now
        \begin{align*}
            \int_V \pd{F_1} x \D V &= \int_z \int_y \left(\int_{x^-(y, z)}^{x^+(y, z)} \pd{F_1} x \D x\right) \D y \D z \\
            &= \int_z \int_y \left(F_1(x^+(y, z), y, z) - F_1(x^-(y, z), y, z)\right) \D y \D z \\
            &= \int_{S^+} F_1(\V x) \D y \D z - \int_{S^-} F_1(\V x) \D y \D z
        \end{align*}
        We parameterise $S^+$ using $y$ and $z$, so $\V x = (x^+(y, z), y, z)$ on $S^+$, so
        \begin{align*}
            d\V S = \pm \begin{pmatrix}
             \pd{x^+}{y} \\
             1 \\
             0 \\
            \end{pmatrix} \times \begin{pmatrix}
             \pd{x^+}{z} \\
             0 \\
             1 \\
            \end{pmatrix} \D y \D z = \pm \begin{pmatrix}
             1 \\
             -\pd{x^+}y \\
             -\pd{x^+}z \\
            \end{pmatrix} \D y \D z
        \end{align*}
        and therefore $\V e_1 \cdot d \V S = \pm \D y \D z$. Since we have chosen the outward normal, we want the plus sign. Similarly, on $S^-$, $\V e_1 \cdot d \V S = -\D y \D z$, so
        \begin{align*}
            \int_V \pd{F_1} x \D V = \int_{S^+} F_1 \V e_1 \cdot d\V S + \int_{S^-} F_1 \V e_1 \cdot d\V S = \int_S F_1 \V e_1 \cdot d\V S
        \end{align*}
        Analogous calculations can clearly be carried out for $\int_V \pd{F_2} y dV$ and $\int_V \pd{F_3} z dV$, yielding $\int_S F_2 \V e_2 \cdot d \V S$ and $\int_S F_3 \V e_3 \cdot d \V S$ respectively. Addition yields the result.

        For more complicated shapes, we split the shape up as in the proof of Green's theorem in the plane.    
    \end{proof}
    
    \begin{example}
        Let us calculate the flux of the vector field $\V F = (y, x, z^6)$ through the surface of the sphere $V_R = \{\V x: |\V x| < R\}$? By the divergence theorem, it is $\int_{V_R} 6z^5 \D V$, which vanishes by oddness in $z$.

        But that's no fun. Let us calculate the flux directly. We've already calculated the surface element for a sphere, namely
        \begin{align*}
            d\V S = R^2 \sin \theta \begin{pmatrix}
             \sin \theta \cos \phi \\
             \sin \theta \sin \phi \\
             \cos \theta \\
            \end{pmatrix} \D \theta \D \phi = R^2 \sin \theta \V e_r \D \theta \D \phi
        \end{align*}
        so that
        \begin{align*}
            \int_{\pr V_R} \V F \cdot d \V S &= \int_{\pr V_R} \begin{pmatrix}
             R \sin \theta \sin \phi \\
             R \sin \theta \cos \phi \\
             R^6 \cos^6 \theta \\
            \end{pmatrix} \cdot R^2 \sin \theta \begin{pmatrix}
             \sin \theta \cos \phi \\
             \sin \theta \sin \phi \\
             \cos \theta \\
            \end{pmatrix} \D \theta \D \phi \\
            &= \int_{\theta=0}^{\pi} \int_{\phi=0}^{2\pi} \left(2R^3 \sin^2 \theta \sin \phi \cos \phi + R^8 \cos^7 \theta\right) \sin \theta \D \theta \D \phi
        \end{align*}
        Both terms vanish by oddness. Specifically, we have $\int_0^{2\pi} \sin \phi \cos \phi \D \phi = 0$ (the range can be repalced with $\int_{-\pi}^{\pi}$ and the integrand is odd) and $\int_0^{\pi} \cos^7 \theta \sin \theta \D \theta = 0$ (because the integrand is odd around $\frac{\pi}2$). Hence the divergence theorem is confirmed.        
    \end{example}
    
    \begin{remark*}
        In such integrals, one should be on the lookout for integral tricks. For example,
        \begin{align*}
            \int_0^{2\pi} \sin^{2m+1} \phi \D \phi = \int_0^{2\pi} \cos^{2m+1} \phi \D \phi &= 0 \\
            \int_0^{2\pi} \sin^{2m+1} \phi \cos^n \phi \D \phi = \int_0^{2\pi} \sin^m \phi \cos^{2n+1} \phi \D \phi &= 0 \\
            \int_0^{\pi} \sin^m \theta \cos^{2n+1} \theta \D \theta &= 0
        \end{align*}
        because they're all odd about something or other.
    \end{remark*}
    
    \begin{example*}[hard]
        Let us find the flux of $\V F(\V x) = \V x$ through the surface
        \begin{align*}
            y^2 + z^2 = \beta^2(x+1)^2
        \end{align*}
        where $\beta$ is a positive constant, $1 \leq x \leq 2$, and the normal points away from the $x$-axis (DIAGRAM). Afterwards, we will confirm the result using the divergence theorem.
        
        S is in fact an open cone with apex at $(-1, 0, 0)$ and the $x$-axis as its axis of symmetry. We would like to use cylindrical polars, but rotated such that
        \begin{align*}
            x = x, \qquad y = \rho \cos \phi, \qquad z = \rho \sin \phi
        \end{align*}
        We then parameterise the cone with
        \begin{align*}
            \V x = (x, \beta(x+1) \cos \phi, \beta(x+1) \sin \phi)
        \end{align*}
        Hence
        \begin{align*}
            d \V S = \pm \begin{pmatrix}
             1 \\
             \beta \cos \phi \\
             \beta \sin \phi \\
            \end{pmatrix} \times \begin{pmatrix}
             0 \\
             -\beta(x+1)\sin\phi \\
             \beta(x+1)\cos\phi \\
            \end{pmatrix} \D x \D \phi = \pm \begin{pmatrix}
             \beta^2(x+1) \\
             -\beta(x+1)\cos\phi \\
             -\beta(x+1)\sin\phi \\
            \end{pmatrix} \D x \D \phi
        \end{align*}
        We need the minus sign to ensure that $d\V S$ points away from the $x$-axis -- see diagram. So the flux is given by
        \begin{align*}
            \mc F_S &= \int_{x=1}^2 \int_{\phi=0}^{2\pi} \begin{pmatrix}
             x \\
             \beta(x+1)\cos\phi \\
             \beta(x+1)\sin\phi \\
            \end{pmatrix} \cdot \begin{pmatrix}
             -\beta^2(x+1) \\
             \beta(x+1)\cos\phi \\
             \beta(x+1)\sin\phi \\
            \end{pmatrix} \D x \D \phi \\
            &= \int_{x=1}^2 \int_{\phi=0}^{2\pi} \beta^2(x+1) \D x \D \phi = 5\pi \beta^2
        \end{align*}
        
        
        
        
    \end{example*}
